} from './deltaSyncService';
import { DELTA_COLLECTION, CLOCK_SKEW_OVERLAP_MS, MAX_APPLIED_IDS, LAST_SYNC_KEY } from './deltaConstants';
import { applyIncomingDeltas } from './deltaApplyService';
import { safeSetItemLocal } from './storage';

let inFlightPullPromise: Promise<number> | null = null;
let lastPullCompletedAt = 0;

/**
 * Kéo các delta thay đổi từ server về client kể từ lastSyncTime và áp dụng vào Zustand
 * Tiết kiệm hơn 80% Read so với việc onSnapshot cả collection
 * B4: In-flight deduplication prevents concurrent identical queries across tabs/effects
 */
export async function pullAndApplyDeltas(yearFilter?: string, force: boolean = false): Promise<number> {
  if (inFlightPullPromise) {
    return inFlightPullPromise;
  }
  const now = Date.now();
  if (!force && now - lastPullCompletedAt < 2000) {
    return 0;
  }

  let pullSucceeded = false;
  inFlightPullPromise = (async () => {
    const lastSyncTime = getLastDeltaSyncTime();
    const restoreCutoff = getRestoreCutoffTimestamp();

    // Device offline longer than delta retention -> deltas are gone, do a full resync.
    if (await resyncIfStale(lastSyncTime, LAST_SYNC_KEY)) return 0;

    const colRef = collection(db, 'artifacts', APP_ID, 'public', 'data', DELTA_COLLECTION);

    try {
      const deltas: TakaDelta[] = [];
      let currentSince = Math.max(lastSyncTime - CLOCK_SKEW_OVERLAP_MS, restoreCutoff);
      let iteration = 0;
      const MAX_PAGES = 5;

      while (iteration < MAX_PAGES) {
        iteration++;
        const q = query(
          colRef,
          where('serverTime', '>', Timestamp.fromMillis(currentSince)),
          orderBy('serverTime', 'asc'),
          limit(500)
        );

        let snapshot: any;
        try {
          snapshot = await getDocsFromServer(q);
        } catch (serverErr) {
          // Nếu mạng đang reconnect lúc iOS mới thức dậy, KHÔNG đánh dấu pullSucceeded để lần gọi ngay sau đó chạy được ngay
          console.warn('[deltaSync] getDocsFromServer not reachable yet (device offline or reconnecting):', serverErr);
          return 0;
        }

        if (!snapshot || snapshot.empty) {
          break;
        }

        let pageMaxTime = currentSince;
        snapshot.forEach((d: any) => {
          const item = d.data() as TakaDelta;
          const itemId = item.id || d.id;
          if (appliedDeltaIds.has(itemId)) {
            if (deltaTime(item) > pageMaxTime) pageMaxTime = deltaTime(item);
            return;
          }
          appliedDeltaIds.add(itemId);
          if (appliedDeltaIds.size > MAX_APPLIED_IDS) {
            const oldest = appliedDeltaIds.values().next().value;
            if (oldest !== undefined) appliedDeltaIds.delete(oldest);
          }
          deltas.push(item);
          if (deltaTime(item) > pageMaxTime) {
            pageMaxTime = deltaTime(item);
          }
        });

        currentSince = pageMaxTime;
        if (snapshot.size < 500) {
          break;
        }
      }

      pullSucceeded = true;

      if (deltas.length === 0) {
        // Server confirmed 0 new deltas: advance sync cursor safely
        const safeRecentTime = Math.max(lastSyncTime, Date.now() - 60000);
        safeSetItemLocal(LAST_SYNC_KEY, String(safeRecentTime));
        return 0;
      }

      return applyIncomingDeltas(deltas);
    } catch (err) {
      console.warn('[deltaSync] Error pulling deltas:', err);
      return 0;
    } finally {
      if (pullSucceeded) {
        lastPullCompletedAt = Date.now();
      }
      inFlightPullPromise = null;
    }
  })();

  return inFlightPullPromise;
}

let syncIntervalTimer: any = null;
let isHibernating = false;
let lastActivityTime = Date.now();
const IDLE_TIMEOUT_MS = 5 * 60 * 1000; // 5 minutes inactivity threshold

/**
 * Khởi động Background Delta Sync Poller với Adaptive Tab Hibernation & Inactivity Sleep
 * - Khi ẩn tab hoặc người dùng không hoạt động > 5 phút: Tạm dừng định thời hoàn toàn (0 Idle Read).
 * - Khi quay lại tab hoặc tương tác: Lập tức sync tức thì (0ms) và kích hoạt lại interval.
 */
export function startDeltaSyncService(intervalMs: number = 30 * 60 * 1000): () => void {
  if (typeof window === 'undefined') return () => {};

  const stopRealtime = startDeltaRealtimeListener();

  let lastTriggerAt = 0;
  const triggerSync = (force: boolean = false) => {
    const now = Date.now();
    if (!force && now - lastTriggerAt < 10000) return;
    lastTriggerAt = now;
    const year = useAppStore.getState().globalYearFilter;
    pullAndApplyDeltas(year, force).catch(() => {});
  };

  const stopPolling = () => {
    if (syncIntervalTimer) {
      clearInterval(syncIntervalTimer);
      syncIntervalTimer = null;
    }
  };

  const startPolling = () => {
    stopPolling();
    syncIntervalTimer = setInterval(() => {
      if (typeof document !== 'undefined' && document.visibilityState === 'hidden') {
        hibernate();
        return;
      }
      if (Date.now() - lastActivityTime > IDLE_TIMEOUT_MS) {
        hibernate();
        return;
      }
      triggerSync(false);
    }, intervalMs);
  };

  const hibernate = () => {
    if (isHibernating) return;
    isHibernating = true;
    stopPolling();
  };

  const wakeAndSync = () => {
    lastActivityTime = Date.now();
    isHibernating = false;
    triggerSync(true); // Ép đồng bộ ngay khi thức giấc (bỏ qua throttle)
    startPolling();
  };

  // 1. Initial boot sync & start timer
  triggerSync(true);
  startPolling();
  // Khi syncResilience khôi phục xong mạng (sau disableNetwork -> enableNetwork), ép pull delta ngay lập tức với force=true!
  const stopNetworkRecovery = initNetworkRecovery(() => {
    triggerSync(true);
    // Phát sự kiện để deltaRealtimeListener gắn lại (re-attach) trên socket mới
    window.dispatchEvent(new CustomEvent('taka_network_recovered'));
  });

  // 2. Tab Visibility Listener (Page Visibility API - Hibernation)
  const handleVisibilityChange = () => {
    if (document.visibilityState === 'visible') {
      wakeAndSync();
    } else {
      hibernate();
    }
  };

  // 3. Online network reconnection & Window Focus (iPad Split-View / App Switcher)
  const handleOnline = () => {
    wakeAndSync();
  };
  const handleFocus = () => {
    if (document.visibilityState === 'visible') {
      wakeAndSync();
    }
  };

  document.addEventListener('visibilitychange', handleVisibilityChange);
  window.addEventListener('online', handleOnline);
  window.addEventListener('focus', handleFocus);

  return () => {
    stopPolling();
    stopNetworkRecovery();
    stopRealtime();
    document.removeEventListener('visibilitychange', handleVisibilityChange);
    window.removeEventListener('online', handleOnline);
    window.removeEventListener('focus', handleFocus);
  };
}

export async function triggerImmediateDeltaSync(year?: string): Promise<number> {
  lastActivityTime = Date.now();
  isHibernating = false;
  return pullAndApplyDeltas(year || useAppStore.getState().globalYearFilter, true);
}

import { collection, onSnapshot, orderBy, query, where, Timestamp } from 'firebase/firestore';
import { onAuthStateChanged } from 'firebase/auth';
import { auth, db, APP_ID } from './firebase';
import { applyIncomingDeltas, filterUnappliedDeltas, getLastDeltaSyncTime, deltaTime, pullAndApplyDeltas } from './deltaSyncService';
import type { TakaDelta } from '../types';

/**
 * Real-time delta channel.
 *
 * One onSnapshot listener on taka_deltas replaces 25s polling:
 *  - idle cost = 0 reads (previously >= 1 read per poll, ~1100+/user/day)
 *  - changes arrive in ~1s instead of 25s - 3 min
 *  - billed once per delta doc; cached deltas (persistent cache) are free
 * The listener is detached when the tab has been hidden > 2 min and re-attached (since = lastSyncTime) on return.
 */

const OVERLAP_MS = 3 * 60 * 1000; // clock-skew window, duplicates filtered by delta id
const DETACH_AFTER_HIDDEN_MS = 2 * 60 * 1000;
const RETRY_MS = 30 * 1000;

export function startDeltaRealtimeListener(): () => void {
  if (typeof window === 'undefined') return () => {};

  let unsub: (() => void) | null = null;
  let buffer: TakaDelta[] = [];
  let flushTimer: ReturnType<typeof setTimeout> | null = null;
  let detachTimer: ReturnType<typeof setTimeout> | null = null;
  let retryTimer: ReturnType<typeof setTimeout> | null = null;
  let signedIn = !!auth.currentUser;
  let stopped = false;

  const flush = () => {
    flushTimer = null;
    const batch = filterUnappliedDeltas(buffer);
    buffer = [];
    if (batch.length === 0) return;
    batch.sort((a, b) => deltaTime(a) - deltaTime(b));
    try {
      applyIncomingDeltas(batch);
    } catch (err) {
      console.warn('[deltaRealtime] apply failed:', err);
    }
  };

  const detach = () => {
    if (unsub) {
      try { unsub(); } catch {}
      unsub = null;
    }
  };

  const attach = () => {
    if (stopped || unsub || !signedIn) return;
    const since = getLastDeltaSyncTime() - OVERLAP_MS;
    const q = query(
      collection(db, 'artifacts', APP_ID, 'public', 'data', 'taka_deltas'),
      where('serverTime', '>', Timestamp.fromMillis(since)),
      orderBy('serverTime', 'asc')
    );
    unsub = onSnapshot(
      q,
      (snap) => {
        snap.docChanges().forEach((c) => {
          if (c.type === 'added') buffer.push(c.doc.data({ serverTimestamps: 'estimate' }) as TakaDelta);
        });
        if (buffer.length > 0 && !flushTimer) flushTimer = setTimeout(flush, 20);
      },
      (err) => {
        console.warn('[deltaRealtime] listener error, retrying:', err);
        detach();
        if (!stopped && !retryTimer) {
          retryTimer = setTimeout(() => {
            retryTimer = null;
            attach();
          }, RETRY_MS);
        }
      }
    );
  };

  if (signedIn) attach();

  let hiddenAt = 0;
  const onVisibility = () => {
    if (document.visibilityState === 'hidden') {
      hiddenAt = Date.now();
      if (detachTimer) clearTimeout(detachTimer);
      detachTimer = setTimeout(detach, DETACH_AFTER_HIDDEN_MS);
    } else {
      if (detachTimer) { clearTimeout(detachTimer); detachTimer = null; }
      const hiddenFor = hiddenAt ? Date.now() - hiddenAt : Infinity;
      hiddenAt = 0;
      // Ẩn ngắn (dưới 15 giây, ví dụ chuyển tab): kết nối vẫn sống và listener vẫn nhận delta realtime,
      // không cần tạo lại listener hay kéo delta (tránh đọc lại cửa sổ 3 phút).
      if (unsub && hiddenFor < 15000) return;
      // Ẩn lâu: ép tạo lại socket để phục hồi kết nối WebChannel bị iOS/Safari treo
      detach();
      attach();
      pullAndApplyDeltas(undefined, true).catch(() => {});
    }
  };

  const onNetworkRecovered = () => {
    if (document.visibilityState === 'visible') {
      detach();
      attach();
      pullAndApplyDeltas(undefined, true).catch(() => {});
    }
  };

  const handlePageShow = (e: PageTransitionEvent) => {
    if (e.persisted) {
      detach();
      attach();
      pullAndApplyDeltas(undefined, true).catch(() => {});
    }
  };

  const handleOnline = () => {
    detach();
    attach();
    pullAndApplyDeltas(undefined, true).catch(() => {});
  };

  const handleFocus = () => {
    // Listener còn sống thì đã nhận delta realtime (Firestore SDK tự nối lại và bù delta bị lỡ)
    if (unsub) return;
    pullAndApplyDeltas(undefined, true).catch(() => {});
  };

  const unsubAuth = onAuthStateChanged(auth, (user) => {
    signedIn = !!user;
    if (user) attach();
    else detach();
  });

  document.addEventListener('visibilitychange', onVisibility);
  window.addEventListener('taka_network_recovered', onNetworkRecovered);
  window.addEventListener('pageshow', handlePageShow as any);
  window.addEventListener('online', handleOnline);
  window.addEventListener('focus', handleFocus);

  return () => {
    stopped = true;
    unsubAuth();
    document.removeEventListener('visibilitychange', onVisibility);
    window.removeEventListener('taka_network_recovered', onNetworkRecovered);
    window.removeEventListener('pageshow', handlePageShow as any);
    window.removeEventListener('online', handleOnline);
    window.removeEventListener('focus', handleFocus);
    if (flushTimer) clearTimeout(flushTimer);
    if (detachTimer) clearTimeout(detachTimer);
    if (retryTimer) clearTimeout(retryTimer);
    detach();
  };
}

import { useAppStore } from '../store/useAppStore';
import { usePerformanceStore } from '../store/usePerformanceStore';
import { TakaDelta, TakaDeltaItem, Project, Task, Cost, DelegationGroup } from '../types';
import { localDb } from './localDb';
import { applyDeltasToLocalDatabase } from './backgroundSyncWorker';
import { triggerBatchChangeFlash } from './changeFlashService';
import { broadcastAppliedDeltas } from './broadcastSyncService';
import { applyFieldPatch } from './deltaPatchService';
import { getTasksForProject, calculateProjectComputedDates } from './projectDatesService';
import { resolveInvalidations } from './deltaInvalidateService';
import { 
  deltaTime, 
  getLastDeltaSyncTime, 
  getRestoreCutoffTimestamp, 
  markRestoreVersionCompleted
} from './deltaSyncService';
import { CLIENT_SESSION_ID, DELTA_COLLECTION, LAST_SYNC_KEY } from './deltaConstants';
import { safeSetItemLocal } from './storage';
import { markTasksDeleted } from './taskDeletedTracker';

/**
 * Applies a list of (already de-duplicated) deltas to Zustand + Dexie. Shared by the real-time listener and the
 * safety poll. Returns the number of deltas processed.
 */
export function applyIncomingDeltas(deltas: TakaDelta[]): number {
  if (!deltas || deltas.length === 0) return 0;

  // Ghi nhận vào Performance Store để giám sát
  usePerformanceStore
    .getState()
    .trackGetDoc(DELTA_COLLECTION, false, deltas.length, `Pulled ${deltas.length} deltas`);

  const lastSyncTime = getLastDeltaSyncTime();
  let maxTime = lastSyncTime;
  const store = useAppStore.getState();

  // Map chứa các thay đổi gộp để cập nhật nguyên tử vào Zustand
  let updatedProjects = [...(store.projects || [])];
  let updatedTasks = [...(store.tasks || [])];
  let updatedCosts = [...(store.costs || [])];
  let updatedDelegations = [...(store.delegationGroups || [])];

  let projectsChanged = false;
  let tasksChanged = false;
  let costsChanged = false;
  let delegationsChanged = false;

  const invalidations: TakaDeltaItem[] = [];

  // Các dự án có task thay đổi trong lô delta này (để chỉ tính lại đúng các dự án đó)
  const touchedProjectIds = new Set<string>();
  const touchedProjectCodes = new Set<string>();
  const noteTouchedProject = (t: any) => {
    if (!t) return;
    if (t.projectId) touchedProjectIds.add(String(t.projectId));
    if (t.projectCode) touchedProjectCodes.add(String(t.projectCode).trim().toLowerCase());
  };

  const applyItem = (
    docId: string,
    collectionNameRaw: string,
    action: 'create' | 'update' | 'delete',
    data: any,
    year?: string
  ) => {
    // Một số nơi ghi delta với tên đường dẫn thật `taka_delegation_groups`
    const collectionName =
      collectionNameRaw === 'taka_delegation_groups' ? 'taka_delegations' : collectionNameRaw;
    if (
      collectionName === 'taka_delegations' &&
      !store.isDelegationsRequested &&
      store.delegationGroups.length === 0
    ) {
      return;
    }

    // [Safety Guard] Prevent old deltas (e.g. synthetic or out-of-order) from clobbering newer state
    const tsOf = (v: any): number => {
      if (!v) return 0;
      const n = typeof v === 'string' ? Date.parse(v) : typeof v?.toMillis === 'function' ? v.toMillis() : Number(v);
      return Number.isFinite(n) ? n : 0;
    };

    const isDelete = action === 'delete' || data?.isDeleted === true;

    // Chỉ bỏ qua khi delta CÓ updatedAt hợp lệ và CŨ HƠN bản đang giữ.
    // Delta xóa, delta không có data, và delta vá không mang updatedAt luôn được áp dụng
    // (thứ tự đã do serverTime quyết định). Tuyệt đối không đọc data.updatedAt khi data có thể null.
    const isStaleVsLocal = (existing: any): boolean => {
      if (isDelete || !data) return false;
      const incoming = tsOf(data.updatedAt);
      if (!incoming) return false;
      return tsOf(existing?.updatedAt) > incoming;
    };

    switch (collectionName) {
      case 'taka_projects': {
        const idx = updatedProjects.findIndex((p) => p.id === docId);
        if (isDelete || data?.isDeleted) {
          if (idx !== -1) {
            projectsChanged = true;
            updatedProjects = updatedProjects.filter((p) => p.id !== docId);
          }
          break;
        }
        if (idx !== -1 && isStaleVsLocal(updatedProjects[idx])) return;
        
        projectsChanged = true;
        if (idx !== -1) {
          updatedProjects[idx] = applyFieldPatch(updatedProjects[idx], data) as Project;
        } else if (action === 'create') {
          // Ghost record guard: only add if action is 'create', never resurrect from an 'update' patch
          updatedProjects.push({ id: docId, ...data } as Project);
        }
        break;
      }

      case 'taka_tasks': {
        const idx = updatedTasks.findIndex((t) => t.id === docId);
        noteTouchedProject(idx !== -1 ? updatedTasks[idx] : null);
        noteTouchedProject(data);
        if (isDelete || data?.isDeleted) {
          markTasksDeleted([docId]);
          if (idx !== -1) {
            tasksChanged = true;
            updatedTasks = updatedTasks.filter((t) => t.id !== docId);
          }
          // Cắt đứt ngay lập tức task đã xóa khỏi project.taskSummaries trên máy nhận (chống hiện tượng bóng ma hồi sinh task trên iPad)
          updatedProjects = updatedProjects.map((p) => {
            if (Array.isArray(p.taskSummaries) && p.taskSummaries.some((s: any) => s.id === docId)) {
              projectsChanged = true;
              const filteredSummaries = p.taskSummaries.filter((s: any) => s.id !== docId);
              return {
                ...p,
                taskSummaries: filteredSummaries,
                taskCount: filteredSummaries.length,
              };
            }
            return p;
          });
          break;
        }
        if (idx !== -1 && isStaleVsLocal(updatedTasks[idx])) return;

        tasksChanged = true;
        if (idx !== -1) {
          updatedTasks[idx] = applyFieldPatch(updatedTasks[idx], data) as Task;
        } else if (action === 'create') {
          // Ghost record guard: only add if action is 'create', never resurrect from an 'update' patch
          updatedTasks.push({ id: docId, ...data } as Task);
        }
        break;
      }

      case 'taka_costs': {
        const idx = updatedCosts.findIndex((c) => c.id === docId);
        if (isDelete || data?.isDeleted) {
          if (idx !== -1) {
            costsChanged = true;
            updatedCosts = updatedCosts.filter((c) => c.id !== docId);
          }
          break;
        }
        if (idx !== -1 && isStaleVsLocal(updatedCosts[idx])) return;

        costsChanged = true;
        if (idx !== -1) {
          updatedCosts[idx] = applyFieldPatch(updatedCosts[idx], data) as Cost;
        } else if (action === 'create') {
          // Ghost record guard: only add if action is 'create', never resurrect from an 'update' patch
          updatedCosts.push({ id: docId, ...data } as Cost);
        }
        break;
      }

      case 'taka_delegations': {
        const idx = updatedDelegations.findIndex((g) => g.id === docId);
        if (isDelete || data?.isDeleted) {
          if (idx !== -1) {
            delegationsChanged = true;
            updatedDelegations = updatedDelegations.filter((g) => g.id !== docId);
          }
          break;
        }
        if (idx !== -1 && isStaleVsLocal(updatedDelegations[idx])) return;

        delegationsChanged = true;
        if (idx !== -1) {
          updatedDelegations[idx] = applyFieldPatch(updatedDelegations[idx], data) as DelegationGroup;
        } else if (action === 'create') {
          // Ghost record guard: only add if action is 'create', never resurrect from an 'update' patch
          updatedDelegations.push({ id: docId, ...data } as DelegationGroup);
        }
        break;
      }
    }
  };

  deltas.forEach((delta) => {
    if (deltaTime(delta) > maxTime) {
      maxTime = deltaTime(delta);
    }

    if (delta.docId === 'sys_restore_signal' || delta.data?.type === 'VERSION_RESTORE') {
      const cutoff = delta.data?.restoreCutoff || deltaTime(delta);
      if (cutoff > getRestoreCutoffTimestamp()) {
        markRestoreVersionCompleted(cutoff, true);
        localDb.projects.clear().catch(() => {});
        localDb.tasks.clear().catch(() => {});
        localDb.costs.clear().catch(() => {});
        localDb.delegations.clear().catch(() => {});
      }
      return;
    }

    // 1. Skip self-mutations on this same tab/session to prevent typing jitter (Rule 2)
    if (
      delta.sessionId &&
      delta.sessionId === CLIENT_SESSION_ID
    ) {
      return;
    }

    if (delta.action === 'batch_update' && Array.isArray(delta.batchItems)) {
      delta.batchItems.forEach((item) => {
        if (item.invalidate) {
          invalidations.push(item);
          return;
        }
        try {
          applyItem(item.docId, item.collectionName, item.action, item.data, item.year);
        } catch (err) {
          console.warn('[deltaSync] applyItem failed (batch item skipped):', item.docId, err);
        }
      });
    } else if (delta.docId && delta.collectionName) {
      try {
        applyItem(delta.docId, delta.collectionName, delta.action as any, delta.data, delta.year);
      } catch (err) {
        console.warn('[deltaSync] applyItem failed (delta skipped):', delta.docId, err);
      }
    }
  });

  // Cập nhật nguyên tử 1 lần duy nhất vào Zustand (0ms UI re-render)
  if (tasksChanged && !projectsChanged) {
    const delegationGroups = store.delegationGroups || [];
    const isTasksRequested = store.isTasksRequested;
    const hasTouchedInfo = touchedProjectIds.size + touchedProjectCodes.size > 0;
    updatedProjects = updatedProjects.map(p => {
      if (hasTouchedInfo) {
        // Dự án khớp theo id, hoặc theo mã chính / mã liên kết (linkCodes)
        const codeParts = `${p.code || ''},${(p as any).linkCodes || ''}`
          .toLowerCase()
          .split(/[,;/]+/)
          .map((s) => s.trim())
          .filter(Boolean);
        const isTouched =
          touchedProjectIds.has(String(p.id)) || codeParts.some((c) => touchedProjectCodes.has(c));
        if (!isTouched) return p;
      }
      const pTasks = getTasksForProject(p, updatedTasks);
      if (isTasksRequested || pTasks.length > 0) {
        const computedDates = calculateProjectComputedDates(p, pTasks, delegationGroups);
        return {
          ...p,
          ...computedDates,
          taskCount: computedDates.taskCount !== undefined ? computedDates.taskCount : pTasks.length,
          taskSummaries: computedDates.taskSummaries || [],
        };
      }
      return p;
    });
    projectsChanged = true;
  }

  if (projectsChanged) store.setProjects(updatedProjects);
  if (tasksChanged) store.setTasks(updatedTasks);
  if (costsChanged) store.setCosts(updatedCosts);
  if (delegationsChanged) store.setDelegationGroups(updatedDelegations);

  // Trụ cột 3: Kích hoạt Smart Change Flash cho các dòng vừa nhận thay đổi từ đồng nghiệp (Google Sheets standard)
  const flashEntries: Array<{ docId: string; collectionName: any; updatedBy?: string }> = [];
  deltas.forEach((d) => {
    if (d.action === 'batch_update' && Array.isArray(d.batchItems)) {
      d.batchItems.forEach((item) => {
        if (item.action === 'create' || item.action === 'update') {
          flashEntries.push({ docId: item.docId, collectionName: item.collectionName, updatedBy: d.updatedBy });
        }
      });
    } else if ((d.action === 'create' || d.action === 'update') && d.docId && d.collectionName) {
      flashEntries.push({ docId: d.docId, collectionName: d.collectionName, updatedBy: d.updatedBy });
    }
  });
  if (flashEntries.length > 0) {
    triggerBatchChangeFlash(flashEntries);
  }

  // Lưu các delta vào Dexie Local-First Database ngầm
  applyDeltasToLocalDatabase(deltas).catch(() => {});

  // Phát tín hiệu delta đã nạp sang các tab khác (giúp các tab phụ không cần gửi request lên Firestore)
  broadcastAppliedDeltas(deltas);

  // Cập nhật mốc đồng bộ mới nhất
  safeSetItemLocal(LAST_SYNC_KEY, String(maxTime));

  // Resolve signal-only (invalidate) items: re-read just those docs, then apply them like normal deltas
  if (invalidations.length > 0) {
    resolveInvalidations(invalidations)
      .then((synthetic) => {
        if (synthetic.length > 0) applyIncomingDeltas(synthetic);
      })
      .catch((err) => console.warn('[deltaSync] invalidate resolve failed:', err));
  }

  return deltas.length;
}

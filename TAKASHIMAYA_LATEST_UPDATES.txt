export const initAuth = (
  onAuthSuccess?: (user: User, token: string) => void,
  onAuthFailure?: () => void
) => {
  return onAuthStateChanged(auth, async (user: User | null) => {
    if (user) {
      if (cachedAccessToken) {
        if (onAuthSuccess) onAuthSuccess(user, cachedAccessToken);
      } else if (!isSigningIn) {
        cachedAccessToken = null;
        if (onAuthFailure) onAuthFailure();
      }
    } else {
      cachedAccessToken = null;
      if (onAuthFailure) onAuthFailure();
    }
  });
};

  }

function cleanDataForAudit(data: any) {
  if (!data) return null;
  const clone: any = {};
  for (const key in data) {
    if (Object.prototype.hasOwnProperty.call(data, key)) {
      const val = data[key];
      if (val === undefined) {
        clone[key] = null;
      } else if (val !== null && typeof val === 'object' && val.constructor && val.constructor.name !== 'Object' && val.constructor.name !== 'Array') {
        clone[key] = `[Special: ${val.constructor.name}]`;
      } else {
        clone[key] = val;
      }
    }
  }
  return clone;
}

function writeAuditLogAsync(action: 'CREATE' | 'UPDATE' | 'DELETE', docRef: any, data?: any) {
  try {
    if (!docRef || !docRef.path) return;
    const path = docRef.path;
    
    // Ignore audit logs and noisy system collections
    if (path.includes('audit_logs') || path.includes('device_telemetry') || path.includes('taka_config')) return;
    
    const user = auth.currentUser;
    let userId = user?.uid;
    let userEmail = user?.email;

    if (!userId || !userEmail) {
      try {
        const storedAuth = typeof window !== 'undefined' ? localStorage.getItem('taka_auth_v1') : null;
        if (storedAuth) {
          const parsed = JSON.parse(storedAuth);
          if (parsed && typeof parsed === 'object') {
            userId = userId || parsed.id || parsed.username;
            userEmail = userEmail || parsed.email || parsed.username;
          }
        }
      } catch {}
    }

    userId = userId || 'system';
    userEmail = userEmail || 'anonymous';
    
    const dbInstance = docRef.firestore;
    const logRef = doc(collection(dbInstance, 'audit_logs'));
    
    // Determine entity type from path
    let entityType = 'other';
    if (path.includes('taka_projects')) entityType = 'project';
    else if (path.includes('taka_tasks')) entityType = 'task';
    else if (path.includes('taka_costs')) entityType = 'cost';
    else if (path.includes('taka_delegation_groups')) entityType = 'delegation';
    else if (path.includes('taka_users')) entityType = 'user';
    else if (path.includes('taka_rules')) entityType = 'rule';

    // Capture previous state snapshot for Visual Diff Viewer
    let previousData: any = null;
    try {
      const storeGetter = typeof window !== 'undefined' ? (window as any).__TAKA_APP_STORE__?.getState : null;
      if (storeGetter) {
        const storeState = storeGetter();
        let existingDoc: any = null;
        if (entityType === 'project') existingDoc = storeState.projects?.find((p: any) => p.id === docRef.id);
        else if (entityType === 'task') existingDoc = storeState.tasks?.find((t: any) => t.id === docRef.id);
        else if (entityType === 'cost') existingDoc = storeState.costs?.find((c: any) => c.id === docRef.id);
        else if (entityType === 'delegation') existingDoc = storeState.delegationGroups?.find((d: any) => d.id === docRef.id);

        if (action === 'UPDATE' && data && typeof data === 'object' && existingDoc) {
          const prevSlice: Record<string, any> = {};
          Object.keys(data).forEach(k => {
            if (k.includes('.')) {
              const parts = k.split('.');
              let curr = existingDoc;
              for (let i = 0; i < parts.length && curr != null; i++) {
                curr = curr[parts[i]];
              }
              prevSlice[k] = curr !== undefined ? curr : null;
            } else {
              prevSlice[k] = existingDoc[k] !== undefined ? existingDoc[k] : null;
            }
          });
          previousData = cleanDataForAudit(prevSlice);
        } else if (action === 'DELETE' && existingDoc) {
          previousData = cleanDataForAudit(existingDoc);
        }
      }
    } catch {}

    const logData = {
      timestamp: Date.now(),
      userId,
      userEmail,
      action,
      entityType,
      docPath: path,
      docId: docRef.id,
      data: cleanDataForAudit(data),
      ...(previousData ? { previousData } : {})
    };
    
    // Fire and forget (Optimistic UI unaffected)
    firestoreSetDoc(logRef, logData).catch(err => {
      console.warn("Audit log failed:", err);
    });
  } catch(e) {
    console.warn("Error preparing audit log:", e);
  }
}

function writeBatchAuditLogAsync(
  dbInstance: any,
  ops: Array<{ action: 'CREATE' | 'UPDATE' | 'DELETE'; docRef: any; data?: any }>
) {
  try {
    if (!ops || ops.length === 0) return;
    if (ops.length === 1) {
      writeAuditLogAsync(ops[0].action, ops[0].docRef, ops[0].data);
      return;
    }

    const filteredOps = ops.filter((op) => {
      const p = op.docRef?.path || '';
      return !p.includes('audit_logs') && !p.includes('device_telemetry') && !p.includes('taka_config');
    });
    if (filteredOps.length === 0) return;

    const user = auth.currentUser;
    let userId = user?.uid;
    let userEmail = user?.email;

    if (!userId || !userEmail) {
      try {
        const storedAuth = typeof window !== 'undefined' ? localStorage.getItem('taka_auth_v1') : null;
        if (storedAuth) {
          const parsed = JSON.parse(storedAuth);
          if (parsed && typeof parsed === 'object') {
            userId = userId || parsed.id || parsed.username;
            userEmail = userEmail || parsed.email || parsed.username;
          }
        }
      } catch {}
    }

    userId = userId || 'system';
    userEmail = userEmail || 'anonymous';

    const logRef = doc(collection(dbInstance, 'audit_logs'));

    const entityTypes = new Set<string>();
    filteredOps.forEach((op) => {
      const p = op.docRef?.path || '';
      if (p.includes('taka_projects')) entityTypes.add('project');
      else if (p.includes('taka_tasks')) entityTypes.add('task');
      else if (p.includes('taka_costs')) entityTypes.add('cost');
      else if (p.includes('taka_delegation_groups')) entityTypes.add('delegation');
      else if (p.includes('taka_users')) entityTypes.add('user');
      else if (p.includes('taka_rules')) entityTypes.add('rule');
    });
    const entityType = entityTypes.size === 1 ? Array.from(entityTypes)[0] : 'batch';

    const actions = new Set(filteredOps.map((o) => o.action));
    const primaryAction = actions.size === 1 ? Array.from(actions)[0] : 'UPDATE';

    const summaryItems = filteredOps.slice(0, 50).map((op) => ({
      action: op.action,
      docId: op.docRef?.id,
      docPath: op.docRef?.path,
      data: cleanDataForAudit(op.data),
    }));

    const logData = {
      timestamp: Date.now(),
      userId,
      userEmail,
      action: primaryAction,
      entityType,
      docPath: `batch (${filteredOps.length} items: ${Array.from(entityTypes).join(', ')})`,
      docId: `batch_${filteredOps[0].docRef?.id || Date.now()}`,
      data: {
        batchSize: filteredOps.length,
        entityTypes: Array.from(entityTypes),
        items: summaryItems,
      },
    };

    firestoreSetDoc(logRef, logData).catch((err) => {
      console.warn('Batch audit log failed:', err);
    });
  } catch (e) {
    console.warn('Error preparing batch audit log:', e);
  }
}

function handleOfflineQueue(action: 'set' | 'update' | 'delete', docRef: any, data?: any, error?: any) {
  // Server từ chối vĩnh viễn thì không xếp hàng: phát lại về sau chỉ gây ghi đè/xóa ngoài ý muốn.
  if (error && isPermanentWriteError(error)) return;
  let collectionPath = docRef.path;
  // Firestore path is usually something like "artifacts/APP_ID/public/data/taka_projects/project1"
  // We want to extract the collection path, which is everything before the last slash.
  if (collectionPath) {
    const parts = collectionPath.split('/');
    parts.pop(); // remove docId
    collectionPath = parts.join('/');
  }
  
  addPendingUpdate({
    collectionPath: collectionPath,
    action,
    docId: docRef.id,
    data: data || {}
  } as any);
}

import { removeUndefinedDeep } from "./firestoreSanitizer";

export function setDoc(docRef: any, data: any, options?: any) {
  const cleanData = removeUndefinedDeep(data);
  validateFirestoreData(docRef.path, cleanData, options?.merge === true);
  writeAuditLogAsync("CREATE", docRef, cleanData);
  queueAutoDelta(docRef.path, docRef.id, 'write');
  if (cleanData?.isDeleted) {
    const logical = resolveLogicalCollection(docRef.path);
    if (logical && docRef.id) deleteLocalRecord(logical, docRef.id).catch(() => {});
  }
  return firestoreSetDoc(docRef, cleanData, options).catch(error => {
    console.warn("Network error during setDoc. Saving to local IndexedDB queue.", error);
    handleOfflineQueue('set', docRef, cleanData, error);
    throw error;
  });
}

export function updateDoc(docRef: any, data: any) {
  const cleanData = removeUndefinedDeep(data);
  validateFirestoreData(docRef.path, cleanData, true);
  writeAuditLogAsync("UPDATE", docRef, cleanData);
  queueAutoDelta(docRef.path, docRef.id, 'write');
  if (cleanData?.isDeleted) {
    const logical = resolveLogicalCollection(docRef.path);
    if (logical && docRef.id) deleteLocalRecord(logical, docRef.id).catch(() => {});
  }
  return firestoreUpdateDoc(docRef, cleanData).catch(error => {
    console.warn("Network error during updateDoc. Saving to local IndexedDB queue.", error);
    handleOfflineQueue('update', docRef, cleanData, error);
    throw error;
  });
}

export function deleteDoc(docRef: any) {
  writeAuditLogAsync("DELETE", docRef);
  queueAutoDelta(docRef.path, docRef.id, 'delete');
  const logical = resolveLogicalCollection(docRef.path);
  if (logical && docRef.id) deleteLocalRecord(logical, docRef.id).catch(() => {});
  return firestoreDeleteDoc(docRef).catch(error => {
    console.warn("Network error during deleteDoc. Saving to local IndexedDB queue.", error);
    handleOfflineQueue('delete', docRef, undefined, error);
    throw error;
  });
}

export function writeBatch(dbInstance: any): WriteBatch {
  const batch = firestoreWriteBatch(dbInstance);
  const originalSet = batch.set.bind(batch);
  const originalUpdate = batch.update.bind(batch);
  const originalDelete = batch.delete.bind(batch);
  const originalCommit = batch.commit.bind(batch);

  const operations: any[] = [];
  const auditOps: Array<{ action: 'CREATE' | 'UPDATE' | 'DELETE'; docRef: any; data?: any }> = [];

  (batch as any).set = (docRef: any, data: any, options?: any) => {
    const cleanData = removeUndefinedDeep(data);
    validateFirestoreData(docRef.path, cleanData, options?.merge === true);
    operations.push({ action: 'set', docRef, data: cleanData, options });
    auditOps.push({ action: 'CREATE', docRef, data: cleanData });
    return originalSet(docRef, cleanData, options);
  };
  
  (batch as any).update = (docRef: any, data: any, ...moreFieldsAndValues: any[]) => {
    const cleanData = removeUndefinedDeep(data);
    validateFirestoreData(docRef.path, cleanData, true);
    operations.push({ action: 'update', docRef, data: cleanData });
    auditOps.push({ action: 'UPDATE', docRef, data: cleanData });
    return (originalUpdate as any)(docRef, cleanData, ...moreFieldsAndValues);
  };

  (batch as any).delete = (docRef: any) => {
    operations.push({ action: 'delete', docRef });
    auditOps.push({ action: 'DELETE', docRef });
    return originalDelete(docRef);
  };

  (batch as any).commit = () => {
    // Queue after the ops are handed to the SDK so the delta write is ordered after the data writes
    operations.forEach(op => {
      queueAutoDelta(op.docRef.path, op.docRef.id, op.action === 'delete' ? 'delete' : 'write');
      if (op.action === 'delete' || op.data?.isDeleted) {
        const logical = resolveLogicalCollection(op.docRef.path);
        if (logical && op.docRef.id) deleteLocalRecord(logical, op.docRef.id).catch(() => {});
      }
    });
    // B3: Batch audit log consolidation into 1 single document write
    writeBatchAuditLogAsync(dbInstance, auditOps);
    return originalCommit().catch(error => {
      console.warn("Network error during batch commit. Saving to local IndexedDB queue.", error);
      operations.forEach(op => {
         handleOfflineQueue(op.action, op.docRef, op.data, error);
      });
      throw error;
    });
  };

  return batch;
}

/**
 * Ghi lại 1 delta change log khi có mutation phát sinh
 * Cho phép các máy trạm khác chỉ cần pull delta này thay vì tải lại toàn bộ collection
 */
export async function recordDelta(delta: {
  docId: string;
  collectionName: 'taka_projects' | 'taka_tasks' | 'taka_costs' | 'taka_delegations';
  action: 'create' | 'update' | 'delete';
  data: Record<string, any> | null;
  year?: string;
  updatedBy?: string;
  isPatch?: boolean;
}): Promise<void> {
  markExplicitDelta([{ docId: delta.docId, collectionName: delta.collectionName }]);
  const now = Date.now();
  const deltaId = `delta_${now}_${Math.random().toString(36).substring(2, 8)}`;
  const nowIso = new Date(now).toISOString();

  const deltaPayload: TakaDelta = {
    id: deltaId,
    docId: delta.docId,
    collectionName: delta.collectionName,
    action: delta.action,
    ...(delta.year ? { year: String(delta.year) } : {}),
    data: delta.data || {},
    ...(delta.isPatch !== undefined ? { isPatch: delta.isPatch } : {}),
    updatedBy: delta.updatedBy || useAppStore.getState().currentUser?.email || 'system',
    sessionId: CLIENT_SESSION_ID,
    timestamp: nowIso,
    numericTime: now,
    serverTime: serverTimestamp() as any,
    expireAt: Timestamp.fromMillis(now + 7 * 24 * 60 * 60 * 1000),
  };

  try {
    const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', DELTA_COLLECTION, deltaId);
    const cleanPayload = removeUndefinedDeep(deltaPayload);
    await setDoc(docRef, cleanPayload);

    // Kích hoạt cập nhật fingerprint năm tương ứng
    if (delta.year) {
      touchYearMeta(delta.year).catch(() => {});
    }

    // Phát tín hiệu tức thì sang các máy trạm khác trong cùng trình duyệt (0 Read)
    broadcastAppliedDeltas([deltaPayload]);
  } catch (err) {
    console.warn('[deltaSync] Failed to record delta to Firestore:', err);
  }
}

/**
 * Giai đoạn 2: Ghi gom hàng loạt mutation vào 1 Document Delta duy nhất (Giảm 90% Firestore Write)
 */
export async function recordBatchDeltas(
  items: TakaDeltaItem[],
  updatedBy?: string
): Promise<void> {
  if (!items || items.length === 0) return;
  markExplicitDelta(items.filter((i) => !i.invalidate));
  if (items.length === 1 && !items[0].invalidate) {
    const item = items[0];
    return recordDelta({
      docId: item.docId,
      collectionName: item.collectionName,
      action: item.action,
      data: item.data || {},
      year: item.year,
      isPatch: item.isPatch,
      updatedBy,
    });
  }

  const now = Date.now();
  const deltaId = `delta_batch_${now}_${Math.random().toString(36).substring(2, 8)}`;
  const nowIso = new Date(now).toISOString();

  const deltaPayload: TakaDelta = {
    id: deltaId,
    action: 'batch_update',
    batchItems: items,
    updatedBy: updatedBy || useAppStore.getState().currentUser?.email || 'system',
    sessionId: CLIENT_SESSION_ID,
    timestamp: nowIso,
    numericTime: now,
    serverTime: serverTimestamp() as any,
    expireAt: Timestamp.fromMillis(now + 7 * 24 * 60 * 60 * 1000),
  };

  try {
    const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', DELTA_COLLECTION, deltaId);
    const cleanPayload = removeUndefinedDeep(deltaPayload);
    await setDoc(docRef, cleanPayload);

    // Kích hoạt cập nhật fingerprint các năm bị ảnh hưởng
    const affectedYears = new Set(items.map((i) => i.year).filter(Boolean));
    affectedYears.forEach((yr) => touchYearMeta(yr!).catch(() => {}));

    // Phát tín hiệu tức thì sang các máy trạm khác (0 Read)
    broadcastAppliedDeltas([deltaPayload]);
  } catch (err) {
    console.warn('[deltaSync] Failed to record batch deltas to Firestore:', err);
  }
}

  const invalidations: TakaDeltaItem[] = [];

  // Các dự án có task thay đổi trong lô delta này (để chỉ tính lại đúng các dự án đó)
  const touchedProjectIds = new Set<string>();
  const touchedProjectCodes = new Set<string>();
  const noteTouchedProject = (t: any) => {
    if (!t) return;
    if (t.projectId) touchedProjectIds.add(String(t.projectId));
    if (t.projectCode) touchedProjectCodes.add(String(t.projectCode).trim().toLowerCase());
  };

  const applyItem = (
    docId: string,
    collectionNameRaw: string,
    action: 'create' | 'update' | 'delete',
    data: any,
    year?: string
  ) => {
    // Một số nơi ghi delta với tên đường dẫn thật `taka_delegation_groups`
    const collectionName =
      collectionNameRaw === 'taka_delegation_groups' ? 'taka_delegations' : collectionNameRaw;
    if (
      collectionName === 'taka_delegations' &&
      !store.isDelegationsRequested &&
      store.delegationGroups.length === 0
    ) {
      return;
    }

    // [Safety Guard] Prevent old deltas (e.g. synthetic or out-of-order) from clobbering newer state
    const tsOf = (v: any): number => {
      if (!v) return 0;
      const n = typeof v === 'string' ? Date.parse(v) : typeof v?.toMillis === 'function' ? v.toMillis() : Number(v);
      return Number.isFinite(n) ? n : 0;
    };

    const isDelete = action === 'delete' || data?.isDeleted === true;

    // Chỉ bỏ qua khi delta CÓ updatedAt hợp lệ và CŨ HƠN bản đang giữ.
    // Delta xóa, delta không có data, và delta vá không mang updatedAt luôn được áp dụng
    // (thứ tự đã do serverTime quyết định). Tuyệt đối không đọc data.updatedAt khi data có thể null.
    const isStaleVsLocal = (existing: any): boolean => {
      if (isDelete || !data) return false;
      const incoming = tsOf(data.updatedAt);
      if (!incoming) return false;
      return tsOf(existing?.updatedAt) > incoming;
    };

    switch (collectionName) {
      case 'taka_projects': {
        const idx = updatedProjects.findIndex((p) => p.id === docId);
        if (isDelete || data?.isDeleted) {
          if (idx !== -1) {
            projectsChanged = true;
            updatedProjects = updatedProjects.filter((p) => p.id !== docId);
          }
          break;
        }
        if (idx !== -1 && isStaleVsLocal(updatedProjects[idx])) return;
        
        projectsChanged = true;
        if (idx !== -1) {
          const prevProj = updatedProjects[idx];
          // If incoming project update has pruned taskSummaries, also remove those deleted task IDs from updatedTasks
          if (Array.isArray(data?.taskSummaries) && Array.isArray(prevProj.taskSummaries)) {
            const incomingSummaryIds = new Set(
              data.taskSummaries
                .map((s: any) => (s?.id ? String(s.id).trim().toLowerCase() : ''))
                .filter(Boolean)
            );
            const removedIdsLower = new Set<string>();
            prevProj.taskSummaries.forEach((oldS: any) => {
              const oldId = oldS?.id ? String(oldS.id).trim() : '';
              if (oldId && !incomingSummaryIds.has(oldId.toLowerCase())) {
                markTasksDeleted([oldId], docId);
                removedIdsLower.add(oldId.toLowerCase());
                localDb.tasks.delete(oldId).catch(() => {});
              }
            });
            if (removedIdsLower.size > 0) {
              const beforeLen = updatedTasks.length;
              updatedTasks = updatedTasks.filter((t) => !t?.id || !removedIdsLower.has(String(t.id).trim().toLowerCase()));
              if (updatedTasks.length !== beforeLen) {
                tasksChanged = true;
              }
            }
          }
          updatedProjects[idx] = applyFieldPatch(prevProj, data) as Project;
        } else if (action === 'create') {
          // Ghost record guard: only add if action is 'create', never resurrect from an 'update' patch
          updatedProjects.push({ id: docId, ...data } as Project);
        }
        break;
      }

      case 'taka_tasks': {
        const idx = updatedTasks.findIndex((t) => t.id === docId);
        const existingTask = idx !== -1 ? updatedTasks[idx] : null;
        noteTouchedProject(existingTask);
        noteTouchedProject(data);
        if (isDelete || data?.isDeleted) {
          const pIdHint = data?.projectId || existingTask?.projectId;
          markTasksDeleted([docId], pIdHint ? String(pIdHint) : undefined);
          localDb.tasks.delete(docId).catch(() => {});
          if (idx !== -1) {
            tasksChanged = true;
            updatedTasks = updatedTasks.filter((t) => t.id !== docId);
          }
          const docIdLower = String(docId).trim().toLowerCase();
          const delNameLower = String(data?.name || existingTask?.name || '').trim().toLowerCase();
          const delCodeLower = String(data?.taskCodePhu || existingTask?.taskCodePhu || '').trim().toLowerCase();

          // Cắt đứt ngay lập tức task đã xóa khỏi project.taskSummaries trên máy nhận (chống hiện tượng bóng ma hồi sinh task trên iPad)
          updatedProjects = updatedProjects.map((p) => {
            if (Array.isArray(p.taskSummaries) && p.taskSummaries.length > 0) {
              const filteredSummaries = p.taskSummaries.filter((s: any) => {
                const sId = s?.id ? String(s.id).trim().toLowerCase() : '';
                if (sId && sId === docIdLower) return false;
                if (!sId && delNameLower && String(s?.name || '').trim().toLowerCase() === delNameLower) {
                  const sCode = s?.taskCodePhu ? String(s.taskCodePhu).trim().toLowerCase() : '';
                  if (!delCodeLower || !sCode || sCode === delCodeLower) return false;
                }
                return true;
              });
              if (filteredSummaries.length !== p.taskSummaries.length) {
                projectsChanged = true;
                noteTouchedProject({ projectId: p.id, projectCode: p.code });
                return {
                  ...p,
                  taskSummaries: filteredSummaries,
                  taskCount: filteredSummaries.length,
                };
              }
            }
            return p;
          });
          break;
        }
        if (idx !== -1 && isStaleVsLocal(updatedTasks[idx])) return;

        tasksChanged = true;
        if (idx !== -1) {
          updatedTasks[idx] = applyFieldPatch(updatedTasks[idx], data) as Task;
        } else if (action === 'create') {
          // Ghost record guard: only add if action is 'create', never resurrect from an 'update' patch
          updatedTasks.push({ id: docId, ...data } as Task);
        }
        break;
      }

      case 'taka_costs': {
        const idx = updatedCosts.findIndex((c) => c.id === docId);
        if (isDelete || data?.isDeleted) {
          if (idx !== -1) {
            costsChanged = true;
            updatedCosts = updatedCosts.filter((c) => c.id !== docId);
          }
          break;
        }
        if (idx !== -1 && isStaleVsLocal(updatedCosts[idx])) return;

        costsChanged = true;
        if (idx !== -1) {
          updatedCosts[idx] = applyFieldPatch(updatedCosts[idx], data) as Cost;
        } else if (action === 'create') {
          // Ghost record guard: only add if action is 'create', never resurrect from an 'update' patch
          updatedCosts.push({ id: docId, ...data } as Cost);
        }
        break;
      }

      case 'taka_delegations': {
        const idx = updatedDelegations.findIndex((g) => g.id === docId);
        if (isDelete || data?.isDeleted) {
          if (idx !== -1) {
            delegationsChanged = true;
            updatedDelegations = updatedDelegations.filter((g) => g.id !== docId);
          }
          break;
        }
        if (idx !== -1 && isStaleVsLocal(updatedDelegations[idx])) return;

        delegationsChanged = true;
        if (idx !== -1) {
          updatedDelegations[idx] = applyFieldPatch(updatedDelegations[idx], data) as DelegationGroup;
        } else if (action === 'create') {
          // Ghost record guard: only add if action is 'create', never resurrect from an 'update' patch
          updatedDelegations.push({ id: docId, ...data } as DelegationGroup);
        }
        break;
      }
    }
  };

  deltas.forEach((delta) => {
    if (deltaTime(delta) > maxTime) {
      maxTime = deltaTime(delta);
    }

    if (delta.docId === 'sys_restore_signal' || delta.data?.type === 'VERSION_RESTORE') {
      const cutoff = delta.data?.restoreCutoff || deltaTime(delta);
      if (cutoff > getRestoreCutoffTimestamp()) {
        markRestoreVersionCompleted(cutoff, true);
        localDb.projects.clear().catch(() => {});
        localDb.tasks.clear().catch(() => {});
        localDb.costs.clear().catch(() => {});
        localDb.delegations.clear().catch(() => {});
      }
      return;
    }

    // 1. Skip self-mutations on this same tab/session to prevent typing jitter (Rule 2)
    if (
      delta.sessionId &&
      delta.sessionId === CLIENT_SESSION_ID
    ) {
      return;
    }

    if (delta.action === 'batch_update' && Array.isArray(delta.batchItems)) {
      delta.batchItems.forEach((item) => {
        if (item.invalidate) {
          invalidations.push(item);
          return;
        }
        try {
          applyItem(item.docId, item.collectionName, item.action, item.data, item.year);
        } catch (err) {
          console.warn('[deltaSync] applyItem failed (batch item skipped):', item.docId, err);
        }
      });
    } else if (delta.docId && delta.collectionName) {
      try {
        applyItem(delta.docId, delta.collectionName, delta.action as any, delta.data, delta.year);
      } catch (err) {
        console.warn('[deltaSync] applyItem failed (delta skipped):', delta.docId, err);
      }
    }
  });

  // Cập nhật nguyên tử 1 lần duy nhất vào Zustand (0ms UI re-render)
  const touchedProjectsToPersist: Project[] = [];
  if ((tasksChanged || projectsChanged) && (touchedProjectIds.size > 0 || touchedProjectCodes.size > 0)) {
    const delegationGroups = store.delegationGroups || [];
    updatedProjects = updatedProjects.map(p => {
      // Dự án khớp theo id, hoặc theo mã chính / mã liên kết (linkCodes)
      const codeParts = `${p.code || ''},${(p as any).linkCodes || ''}`
        .toLowerCase()
        .split(/[,;/]+/)
        .map((s) => s.trim())
        .filter(Boolean);
      const isTouched =
        touchedProjectIds.has(String(p.id)) || codeParts.some((c) => touchedProjectCodes.has(c));
      
      if (!isTouched) return p;

      const pTasks = getTasksForProject(p, updatedTasks);
      const computedDates = calculateProjectComputedDates(p, pTasks, delegationGroups);
      const nextProj: Project = {
        ...p,
        ...computedDates,
        taskCount: computedDates.taskCount !== undefined ? computedDates.taskCount : pTasks.length,
        taskSummaries: computedDates.taskSummaries || [],
      };
      touchedProjectsToPersist.push(nextProj);
      return nextProj;
    });
    projectsChanged = true;
  }

  if (projectsChanged) store.setProjects(updatedProjects);
  if (tasksChanged) store.setTasks(updatedTasks);
  if (costsChanged) store.setCosts(updatedCosts);
  if (delegationsChanged) store.setDelegationGroups(updatedDelegations);

  if (touchedProjectsToPersist.length > 0) {
    localDb.projects.bulkPut(touchedProjectsToPersist).catch(() => {});
  }

  // Trụ cột 3: Kích hoạt Smart Change Flash cho các dòng vừa nhận thay đổi từ đồng nghiệp (Google Sheets standard)
  const flashEntries: Array<{ docId: string; collectionName: any; updatedBy?: string }> = [];
  deltas.forEach((d) => {
    if (d.action === 'batch_update' && Array.isArray(d.batchItems)) {
      d.batchItems.forEach((item) => {
        if (item.action === 'create' || item.action === 'update') {
          flashEntries.push({ docId: item.docId, collectionName: item.collectionName, updatedBy: d.updatedBy });
        }
      });
    } else if ((d.action === 'create' || d.action === 'update') && d.docId && d.collectionName) {
      flashEntries.push({ docId: d.docId, collectionName: d.collectionName, updatedBy: d.updatedBy });
    }
  });
  if (flashEntries.length > 0) {
    triggerBatchChangeFlash(flashEntries);
  }

  // Lưu các delta vào Dexie Local-First Database ngầm
  applyDeltasToLocalDatabase(deltas).catch(() => {});

  // Phát tín hiệu delta đã nạp sang các tab khác (giúp các tab phụ không cần gửi request lên Firestore)
  broadcastAppliedDeltas(deltas);

  // Cập nhật mốc đồng bộ mới nhất
  safeSetItemLocal(LAST_SYNC_KEY, String(maxTime));

  // Resolve signal-only (invalidate) items: re-read just those docs, then apply them like normal deltas
  if (invalidations.length > 0) {
    resolveInvalidations(invalidations)
      .then((synthetic) => {
        if (synthetic.length > 0) applyIncomingDeltas(synthetic);
      })
      .catch((err) => console.warn('[deltaSync] invalidate resolve failed:', err));
  }

  return deltas.length;
}

// Cache in-flight requests to avoid duplicate fetches
const pendingProjectFetches = new Map<string, Promise<Task[]>>();

/**
 * Validates whether a task belongs to a specific project by verifying store and year.
 */
export function isTaskBelongsToProject(t: Partial<Task>, p: Partial<Project>): boolean {
  if (!t || !p) return false;
  if (t.isDeleted) return false;

  const pStore = p.store ? String(p.store).trim().toLowerCase() : '';
  const pYear = p.year !== undefined && p.year !== null ? String(p.year).trim() : '';
  const pCode = String(p.code || '').trim().toLowerCase();

  const tStore = t.projectStore ? String(t.projectStore).trim().toLowerCase() : '';
  const tYear = t.projectYear !== undefined && t.projectYear !== null ? String(t.projectYear).trim() : '';
  const tCode = t.projectCode ? String(t.projectCode).trim().toLowerCase() : '';

  if (pStore && tStore && pStore !== tStore) return false;
  if (pYear && tYear && pYear !== tYear) return false;

  if (t.projectId && p.id && String(t.projectId) === String(p.id)) return true;
  if (t.projectId && p.id && String(t.projectId) !== String(p.id)) return false;

  if (tCode && pCode && (tCode === pCode || matchProjectCode(tCode, pCode))) return true;

  return false;
}

/**
 * Ensures tasks for a specific projectCode are loaded into Zustand store.
 * Useful when resolving cross-project predecessors dynamically.
 */
export async function ensureProjectTasksLoaded(projectCode: string): Promise<Task[]> {
  if (!projectCode) return [];
  const cleanCode = projectCode.trim();
  const currentTasks = useAppStore.getState().tasks || [];
  
  // If we already have tasks for this project in memory, return them
  const existing = currentTasks.filter(t => 
    t.projectCode && t.projectCode.trim().toLowerCase() === cleanCode.toLowerCase()
  );
  if (existing.length > 0) return existing;

  // If already fetching, return existing promise
  if (pendingProjectFetches.has(cleanCode.toLowerCase())) {
    return pendingProjectFetches.get(cleanCode.toLowerCase())!;
  }

  const fetchPromise = (async () => {
    try {
      const q = query(
        collection(db, "artifacts", APP_ID, "public", "data", "taka_tasks"),
        where("projectCode", "==", cleanCode)
      );
      const snap = await getDocs(q);
      const loaded: Task[] = [];
      snap.forEach(docSnap => {
        const d = { id: docSnap.id, ...docSnap.data() } as Task;
        if (!d.isDeleted) loaded.push(d);
      });

      if (loaded.length > 0) {
        const latestTasks = useAppStore.getState().tasks || [];
        const currentIds = new Set(latestTasks.map(t => t.id));
        const toAdd = loaded.filter(t => !currentIds.has(t.id));
        if (toAdd.length > 0) {
          useAppStore.getState().setTasks([...latestTasks, ...toAdd]);
        }
      }
      return loaded;
    } catch (err) {
      console.warn("Failed to fetch tasks for external project:", cleanCode, err);
      return [];
    } finally {
      pendingProjectFetches.delete(cleanCode.toLowerCase());
    }
  })();

  pendingProjectFetches.set(cleanCode.toLowerCase(), fetchPromise);
  return fetchPromise;
}

/**
 * Tìm toàn bộ danh sách Projects bị ảnh hưởng bởi danh sách Tasks đưa vào
 */
export function findAffectedProjects(
  tasksToCheck: (Partial<Task>)[],
  allProjects?: Project[]
): Project[] {
  const projects = allProjects || useAppStore.getState().projects;
  if (!tasksToCheck.length || !projects.length) return [];

  const affectedMap = new Map<string, Project>();

  for (const t of tasksToCheck) {
    if (!t) continue;
    const tStore = t.projectStore ? String(t.projectStore).trim().toLowerCase() : "";
    const tYear = t.projectYear !== undefined && t.projectYear !== null ? String(t.projectYear).trim() : "";
    const tCode = t.projectCode ? String(t.projectCode).trim().toLowerCase() : "";
    const tProjId = t.projectId ? String(t.projectId) : "";

    for (const p of projects) {
      if (!p.id || !p.code) continue;

      const pStore = p.store ? String(p.store).trim().toLowerCase() : "";
      const pYear = p.year !== undefined && p.year !== null ? String(p.year).trim() : "";

      if (tProjId && String(p.id) === tProjId) {
        if (pStore && tStore && pStore !== tStore) continue;
        if (pYear && tYear && pYear !== tYear) continue;
        affectedMap.set(p.id, p);
        continue;
      }

      if (tProjId && String(p.id) !== tProjId) continue;

      if (!tCode) continue;
      const matchesCode =
        String(p.code).trim().toLowerCase() === tCode ||
        matchProjectCode(p.code, tCode);

      if (!matchesCode) continue;
      if (pStore && tStore && pStore !== tStore) continue;
      if (pYear && tYear && pYear !== tYear) continue;

      affectedMap.set(p.id, p);
    }
  }

  return Array.from(affectedMap.values());
}

/**
 * Đồng bộ tính toán lại mốc thời gian và tiến độ cho các dự án bị ảnh hưởng
 * Cập nhật ngay tức thì vào Zustand store (0ms) và đẩy ngầm Firestore (Dirty-checked).
 */
export async function syncAffectedProjects(
  affectedProjects: Project[],
  allTasks?: Task[]
): Promise<void> {
  if (!affectedProjects.length) return;

  const currentTasks = allTasks || useAppStore.getState().tasks;
  const delegationGroups = useAppStore.getState().delegationGroups;

  for (const project of affectedProjects) {
    if (!project.id) continue;
    const projectTasks = getTasksForProject(project, currentTasks);
    const computedDates = calculateProjectComputedDates(project, projectTasks, delegationGroups);
    await syncProjectDatesToFirestore(project.id, computedDates);
  }
}

/**
 * Shared State Handler: Cập nhật / Thêm mới Tasks đồng bộ giữa mọi Tab (DetailTab <-> GeneralTab)
 * 1. Optimistic update vào Zustand `tasks` (0ms UI lag).
 * 2. Tự động nhận diện Projects liên quan, tính lại Start, Finish/Handover, Opening, Tiến độ.
 * 3. Optimistic update vào Zustand `projects` và đồng bộ ngầm Firestore.
 */
export async function saveTaskStateUnified(
  taskUpdates: (Partial<Task> & { id: string })[],
  options?: { syncProjects?: boolean; allTasksOverride?: Task[] }
): Promise<void> {
  if (!taskUpdates || taskUpdates.length === 0) return;

  // 1. Cập nhật Optimistic tức thì vào State Tasks
  useAppStore.getState().optimisticUpdateTasks(taskUpdates);

  if (options?.syncProjects === false) return;

  // 2. Lấy danh sách Tasks và Projects mới nhất
  const updatedTasks = options?.allTasksOverride || useAppStore.getState().tasks;
  const affectedProjects = findAffectedProjects(taskUpdates);

  // 3. Đồng bộ lại toàn bộ Computed Dates cho các dự án bị ảnh hưởng
  await syncAffectedProjects(affectedProjects, updatedTasks);
}

/**
 * Shared State Handler: Xóa Tasks đồng bộ giữa DetailTab và GeneralTab
 */
export async function deleteTaskStateUnified(
  taskIdsOrTasks: (string | Task)[],
  options?: { syncProjects?: boolean }
): Promise<void> {
  if (!taskIdsOrTasks || taskIdsOrTasks.length === 0) return;

  const rawIds: string[] = [];
  const passedTaskObjs: Task[] = [];

  taskIdsOrTasks.forEach(item => {
    if (typeof item === 'string') {
      rawIds.push(item);
    } else if (item && typeof item === 'object') {
      if (item.id) rawIds.push(String(item.id));
      passedTaskObjs.push(item);
    }
  });

  const idSet = new Set(rawIds.map(String));
  const currentTasks = useAppStore.getState().tasks || [];
  const currentProjects = useAppStore.getState().projects || [];

  const foundInStore = currentTasks.filter(t => idSet.has(String(t.id)));
  const tasksToDelete = [...foundInStore];
  passedTaskObjs.forEach(pt => {
    if (!tasksToDelete.some(t => String(t.id) === String(pt.id))) {
      tasksToDelete.push(pt);
    }
  });

  const affectedProjects = findAffectedProjects(tasksToDelete, currentProjects);
  const affectedProjectIds = new Set<string>();
  affectedProjects.forEach(ap => {
    if (ap.id) affectedProjectIds.add(String(ap.id));
  });
  tasksToDelete.forEach(dt => {
    if (dt.projectId) affectedProjectIds.add(String(dt.projectId));
  });

  const deletedTaskIds = new Set<string>();
  tasksToDelete.forEach(t => {
    if (t.id) {
      const cleanId = String(t.id).trim().toLowerCase();
      deletedTaskIds.add(cleanId);
      markTasksDeleted([cleanId], t.projectId ? String(t.projectId) : undefined);
    }
  });
  rawIds.forEach(id => {
    const cleanId = String(id).trim().toLowerCase();
    deletedTaskIds.add(cleanId);
    markTasksDeleted([cleanId]);
  });

  const now = new Date().toISOString();
  const tasksAfterDeletion = currentTasks.filter(t => !idSet.has(String(t.id)));
  
  const unlinkedChildTasks: Task[] = [];
  const cleanedTasks = tasksAfterDeletion.map(t => {
    let updated = t;

    const rawParent = t.parentTaskId || (t as any).parentId;
    if (rawParent && deletedTaskIds.has(String(rawParent).trim().toLowerCase())) {
      updated = { ...updated, parentTaskId: undefined, updatedAt: now };
      delete (updated as any).parentId;
      unlinkedChildTasks.push(updated);
    }

    if (updated.predecessor && typeof updated.predecessor === 'string') {
      const predParts = updated.predecessor.split(',').map(p => p.trim()).filter(Boolean);
      const filteredParts = predParts.filter(part => {
        const codeOnly = part.replace(/(?:SS|FF|SF|FS)?\s*(?:[+-]\d+[dwmy]?)?$/i, '').trim().toLowerCase();
        const isDeletedPred = tasksToDelete.some(dt => {
          if (!isTaskBelongsToProject(t, { id: dt.projectId, code: dt.projectCode, store: dt.projectStore, year: dt.projectYear })) return false;
          return (dt.taskCodePhu && String(dt.taskCodePhu).trim().toLowerCase() === codeOnly) ||
                 (dt.id && String(dt.id).trim().toLowerCase() === codeOnly);
        });
        return !isDeletedPred;
      });
      const newPred = filteredParts.join(', ');
      if (newPred !== updated.predecessor) {
        updated = { ...updated, predecessor: newPred, updatedAt: now };
      }
    }

    return updated;
  });

  useAppStore.getState().setTasks(cleanedTasks);

  // 2. Clear matching summary items from p.taskSummaries and recalculate project computed dates
  const modifiedProjectsForDexie: Project[] = [];
  const delegationGroups = useAppStore.getState().delegationGroups || [];
  const updatedProjects = currentProjects.map(p => {
    const pIdStr = p.id ? String(p.id) : '';

    if (affectedProjectIds.size > 0 && !affectedProjectIds.has(pIdStr)) {
      return p;
    }

    let summaryModified = false;
    let newSummaries = p.taskSummaries ? [...p.taskSummaries] : [];

    if (p.taskSummaries && p.taskSummaries.length > 0) {
      newSummaries = p.taskSummaries.filter((ts) => {
        const tsId = ts?.id ? String(ts.id).trim().toLowerCase() : '';
        if (tsId && deletedTaskIds.has(tsId)) {
          summaryModified = true;
          return false;
        }
        const matchedDeleted = tasksToDelete.some(dt => {
          if (dt.projectId && p.id && String(dt.projectId) !== String(p.id)) return false;
          if (!dt.projectId && !isTaskBelongsToProject(dt, p)) return false;
          const dtId = dt.id ? String(dt.id).trim().toLowerCase() : '';
          if (dtId && tsId && dtId === tsId) return true;
          if (!tsId) {
            const tsName = String(ts?.name || '').trim().toLowerCase();
            const dtName = String(dt.name || '').trim().toLowerCase();
            const tsCode = String(ts?.taskCodePhu || '').trim().toLowerCase();
            const dtCode = String(dt.taskCodePhu || '').trim().toLowerCase();
            if (tsName && dtName && tsName === dtName && (!tsCode || !dtCode || tsCode === dtCode)) {
              return true;
            }
          }
          return false;
        });

        if (matchedDeleted) {
          summaryModified = true;
          return false;
        }
        return true;
      });
    }

    const isProjectAffected = summaryModified || affectedProjectIds.has(pIdStr);

    if (isProjectAffected) {
      const pForCalc = { ...p, taskSummaries: newSummaries, taskCount: newSummaries.length };
      const remainingProjTasks = getTasksForProject(pForCalc, cleanedTasks);
      const computed = calculateProjectComputedDates(pForCalc, remainingProjTasks, delegationGroups);
      const nextProj: Project = { 
        ...p, 
        ...computed,
        taskCount: computed.taskCount !== undefined ? computed.taskCount : remainingProjTasks.length,
        taskSummaries: computed.taskSummaries || newSummaries,
        updatedAt: now,
      };
      modifiedProjectsForDexie.push(nextProj);
      return nextProj;
    }
    return p;
  });

  useAppStore.getState().setProjects(updatedProjects);

  // 3. Xóa tức thì khỏi Dexie (IndexedDB) & lưu Projects mới xuống Dexie
  try {
    const { localDb } = await import('./localDb');
    await localDb.tasks.bulkDelete(Array.from(idSet));
    if (modifiedProjectsForDexie.length > 0) {
      await localDb.projects.bulkPut(modifiedProjectsForDexie);
    }
    if (unlinkedChildTasks.length > 0) {
      await Promise.all(unlinkedChildTasks.map(c => 
        (localDb.tasks.update as any)(c.id, { parentTaskId: null, parentId: null, updatedAt: now })
      ));
    }
  } catch (e) {
    console.warn('[taskStateService] Dexie bulkDelete tasks error:', e);
  }

  // 4. Xóa tệp đính kèm trên Firebase Storage
  try {
    const { extractStorageUrls, deleteStorageFiles } = await import('./storageUtils');
    const urls: string[] = [];
    tasksToDelete.forEach(t => {
      const extracted = extractStorageUrls(t);
      if (extracted.length > 0) urls.push(...extracted);
    });
    if (urls.length > 0) {
      deleteStorageFiles(urls).catch(() => {});
    }
  } catch (e) {}

  // 5. Đồng bộ xóa Task & cập nhật Project Summaries trong cùng 1 Batch + phát Delta Sync lên Firestore
  try {
    const { doc, deleteField } = await import('firebase/firestore');
    const { writeBatch } = await import('./firestoreWrites');
    const { db, APP_ID } = await import('./firebase');
    const { recordBatchDeltas } = await import('./deltaSyncService');
    const { markExplicitDelta } = await import('./autoDeltaQueue');
    const batch = writeBatch(db);

    const allIdsToDelete = new Set<string>();
    rawIds.forEach(id => { if (id) allIdsToDelete.add(String(id)); });
    tasksToDelete.forEach(t => { if (t?.id) allIdsToDelete.add(String(t.id)); });

    allIdsToDelete.forEach(id => {
      const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', id);
      batch.delete(docRef);
    });

    unlinkedChildTasks.forEach(c => {
      const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', c.id);
      batch.update(docRef, { parentTaskId: deleteField(), parentId: null, updatedAt: now });
    });

    // Cập nhật Project Summaries ngay trong cùng Batch này
    updatedProjects.forEach(up => {
      const orig = currentProjects.find(op => op.id === up.id);
      if (orig && orig !== up) {
        const pRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_projects', up.id);
        batch.update(pRef, {
          taskSummaries: up.taskSummaries || [],
          taskCount: up.taskCount || 0,
          computedTakStart: up.computedTakStart || '',
          computedTakCompl: up.computedTakCompl || '',
          computedOpening: up.computedOpening || '',
          computedPlanProgress: up.computedPlanProgress ?? 0,
          computedActProgress: up.computedActProgress ?? 0,
          updatedAt: up.updatedAt || now
        });
      }
    });

    const currentUser = useAppStore.getState().currentUser;
    const deltaItems: any[] = [
      ...Array.from(allIdsToDelete).map(id => {
        const dt = tasksToDelete.find(t => String(t.id) === String(id));
        return {
          docId: id,
          collectionName: 'taka_tasks',
          action: 'delete',
          isPatch: false,
          data: {
            id,
            isDeleted: true,
            ...(dt?.projectId ? { projectId: dt.projectId } : {}),
            ...(dt?.projectCode ? { projectCode: dt.projectCode } : {}),
            ...(dt?.projectStore ? { projectStore: dt.projectStore } : {}),
            ...(dt?.projectYear !== undefined && dt?.projectYear !== null ? { projectYear: dt.projectYear } : {}),
            ...(dt?.taskCodePhu ? { taskCodePhu: dt.taskCodePhu } : {}),
            ...(dt?.name ? { name: dt.name } : {}),
          },
          ...(dt?.projectYear ? { year: String(dt.projectYear) } : {}),
        };
      }),
      ...unlinkedChildTasks.map(c => ({
        docId: c.id,
        collectionName: 'taka_tasks',
        action: 'update',
        isPatch: true,
        data: { parentTaskId: null, parentId: null, updatedAt: now },
        ...(c.projectYear ? { year: String(c.projectYear) } : {}),
      })),
      ...updatedProjects.filter(up => {
        const orig = currentProjects.find(op => op.id === up.id);
        return orig && orig !== up;
      }).map(up => ({
        docId: up.id,
        collectionName: 'taka_projects',
        action: 'update',
        isPatch: true,
        data: {
          taskSummaries: up.taskSummaries || [],
          taskCount: up.taskCount || 0,
          computedTakStart: up.computedTakStart || '',
          computedTakCompl: up.computedTakCompl || '',
          computedOpening: up.computedOpening || '',
          computedPlanProgress: up.computedPlanProgress ?? 0,
          computedActProgress: up.computedActProgress ?? 0,
          updatedAt: up.updatedAt || now
        },
        year: String(up.year || '')
      }))
    ];

    markExplicitDelta(deltaItems);
    await batch.commit();
    await recordBatchDeltas(deltaItems, currentUser?.email || currentUser?.username);
  } catch (e) {
    console.warn('[taskStateService] Failed to persist task deletion to Firestore:', e);
  }
}

/**
 * Shared State Handler: Xóa Dự Án đồng bộ giữa mọi Tab (DetailTab, ProjectModal, QuickNav)
 * 1. Optimistic update: Xóa Project, Tasks, Costs, Delegations khỏi Zustand store (0ms UI lag).
 * 2. Purge khỏi Dexie (IndexedDB) ngay lập tức để tránh re-hydrate lại project & tasks đã xóa.
 * 3. Batch commit đánh dấu isDeleted = true lên Firestore (projects, tasks, costs) & delete delegations.
 * 4. Xóa tệp đính kèm trên Firebase Storage.
 * 5. Ghi Delta log (taka_deltas) để đồng bộ đa màn hình / đa tab.
 */
export async function deleteProjectStateUnified(
  projectIdOrCode: string,
  projectObj?: Project
): Promise<void> {
  if (!projectIdOrCode) return;

  const store = useAppStore.getState();
  const currentProjects = store.projects;
  const currentTasks = store.tasks;
  const currentCosts = store.costs || [];
  const currentDelegations = store.delegationGroups || [];

  const targetKey = String(projectIdOrCode).trim().toLowerCase();

  // Tìm dự án trong store
  const targetProject = projectObj || currentProjects.find(p => 
    (p.id && String(p.id).trim().toLowerCase() === targetKey) ||
    (p.code && String(p.code).trim().toLowerCase() === targetKey) ||
    (p.no !== undefined && String(p.no).trim().toLowerCase() === targetKey)
  );

  const pCode = targetProject?.code ? String(targetProject.code).trim().toLowerCase() : targetKey;
  const pId = targetProject?.id ? String(targetProject.id) : (targetProject?.no !== undefined ? String(targetProject.no) : projectIdOrCode);

  const pStore = targetProject?.store ? String(targetProject.store).trim().toLowerCase() : '';
  const pYear = targetProject?.year !== undefined && targetProject?.year !== null ? String(targetProject.year).trim() : '';

  // Xác định các tasks liên quan
  const relatedTasks = currentTasks.filter(t => {
    if (!targetProject) return false;
    return isTaskBelongsToProject(t, targetProject);
  });
  const relatedTaskIds = new Set(relatedTasks.map(t => String(t.id)));

  // Xác định các costs liên quan
  const relatedCosts = currentCosts.filter(c => {
    const cStore = (c as any).projectStore ? String((c as any).projectStore).trim().toLowerCase() : '';
    const cYear = (c as any).projectYear !== undefined && (c as any).projectYear !== null ? String((c as any).projectYear).trim() : '';
    const cCode = c.projectCode ? String(c.projectCode).trim().toLowerCase() : '';

    if (pStore && cStore && pStore !== cStore) return false;
    if (pYear && cYear && pYear !== cYear) return false;

    if ((c as any).projectId && String((c as any).projectId) === pId) return true;
    if (cCode && pCode && (cCode === pCode || matchProjectCode(cCode, pCode))) return true;
    return false;
  });
  const relatedCostIds = new Set(relatedCosts.map(c => String(c.id)));

  // Xác định các delegations liên quan
  const relatedDelegations = currentDelegations.filter(d => {
    const dStore = (d as any).projectStore ? String((d as any).projectStore).trim().toLowerCase() : '';
    const dYear = (d as any).projectYear !== undefined && (d as any).projectYear !== null ? String((d as any).projectYear).trim() : '';
    const dCode = d.projectCode ? String(d.projectCode).trim().toLowerCase() : '';

    if (pStore && dStore && pStore !== dStore) return false;
    if (pYear && dYear && pYear !== dYear) return false;

    if ((d as any).projectId && String((d as any).projectId) === pId) return true;
    if (dCode && pCode && (dCode === pCode || matchProjectCode(dCode, pCode))) return true;
    if (d.parentTaskId && relatedTaskIds.has(String(d.parentTaskId))) return true;
    return false;
  });
  const relatedDelegationIds = new Set(relatedDelegations.map(d => String(d.id)));

  const nowIso = new Date().toISOString();

  // 1. Optimistic delete tức thì trong Zustand store
  const updatedProjects = currentProjects.filter(p => {
    if (pId && String(p.id) === pId) return false;
    const pStrCode = String(p.code || '').trim().toLowerCase();
    const pStrStore = p.store ? String(p.store).trim().toLowerCase() : '';
    const pStrYear = p.year !== undefined && p.year !== null ? String(p.year).trim() : '';

    if (pStrCode === pCode) {
      if ((!pStore || !pStrStore || pStore === pStrStore) && (!pYear || !pStrYear || pYear === pStrYear)) {
        return false;
      }
    }
    return true;
  });
  const updatedTasks = currentTasks.filter(t => !relatedTaskIds.has(String(t.id)));
  const updatedCosts = currentCosts.filter(c => !relatedCostIds.has(String(c.id)));
  const updatedDelegations = currentDelegations.filter(d => !relatedDelegationIds.has(String(d.id)));

  store.setProjects(updatedProjects);
  store.setTasks(updatedTasks);
  store.setCosts(updatedCosts);
  store.setDelegationGroups(updatedDelegations);

  // 2. Clear khỏi Dexie (IndexedDB) lập tức
  try {
    const { localDb } = await import('./localDb');
    await localDb.projects.delete(pId);
    if (targetProject?.id) await localDb.projects.delete(targetProject.id);
    if (pCode) {
      await localDb.projects.where('code').equalsIgnoreCase(pCode).delete().catch(() => {});
    }
    if (relatedTaskIds.size > 0) {
      await localDb.tasks.bulkDelete(Array.from(relatedTaskIds));
    }
    if (relatedCostIds.size > 0) {
      await localDb.costs.bulkDelete(Array.from(relatedCostIds));
    }
    if (relatedDelegationIds.size > 0) {
      await localDb.delegations.bulkDelete(Array.from(relatedDelegationIds));
    }
  } catch (e) {
    console.warn('[taskStateService] Dexie project delete error:', e);
  }

  // 3. Storage Deep Clean (Rule 4)
  try {
    const { extractStorageUrls, deleteStorageFiles } = await import('./storageUtils');
    const urls: string[] = [];
    if (targetProject) urls.push(...extractStorageUrls(targetProject));
    relatedTasks.forEach(t => urls.push(...extractStorageUrls(t)));
    relatedCosts.forEach(c => urls.push(...extractStorageUrls(c)));
    if (urls.length > 0) {
      deleteStorageFiles(urls).catch(() => {});
    }
  } catch (e) {}

  // 4. Batch commit đánh dấu isDeleted = true trên Firestore
  try {
    const { doc } = await import('firebase/firestore');
    const { writeBatch } = await import('./firestoreWrites');
    const { db, APP_ID } = await import('./firebase');
    const { recordBatchDeltas } = await import('./deltaSyncService');

    const batch = writeBatch(db);

    // Đánh dấu xóa project
    const projectDocRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_projects', pId);
    batch.set(projectDocRef, { id: pId, isDeleted: true, updatedAt: nowIso }, { merge: true });

    // Đánh dấu xóa related tasks
    relatedTasks.forEach(t => {
      const taskRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', t.id);
      batch.set(taskRef, { id: t.id, isDeleted: true, updatedAt: nowIso }, { merge: true });
    });

    // Đánh dấu xóa related costs
    relatedCosts.forEach(c => {
      const costRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_costs', c.id);
      batch.set(costRef, { id: c.id, isDeleted: true, updatedAt: nowIso }, { merge: true });
    });

    // Xóa delegation groups
    relatedDelegations.forEach(d => {
      const delRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_delegation_groups', d.id);
      batch.delete(delRef);
    });

    await batch.commit();

    // 5. Ghi Delta Log
    const currentUser = store.currentUser;
    const deltaItems: any[] = [
      { docId: pId, collectionName: 'taka_projects', action: 'delete', isPatch: false, data: null },
      ...relatedTasks.map(t => ({
        docId: t.id, collectionName: 'taka_tasks', action: 'delete', isPatch: false, data: null
      })),
      ...relatedCosts.map(c => ({
        docId: c.id, collectionName: 'taka_costs', action: 'delete', isPatch: false, data: null
      })),
      ...relatedDelegations.map(d => ({
        docId: d.id, collectionName: 'taka_delegation_groups', action: 'delete', isPatch: false, data: null
      }))
    ];
    recordBatchDeltas(deltaItems, currentUser?.email || currentUser?.username).catch(() => {});
  } catch (err) {
    console.error('[taskStateService] deleteProjectStateUnified Firestore batch commit failed:', err);
  }
}

/**
 * Tự động phát hiện và xử lý các task mồ côi trong bộ nhớ (task con có parentTaskId chỉ đến task mẹ đã bị xóa)
 * Tự động chuyển task con mồ côi thành root task và ghi nhận sửa đổi xuống Dexie & Firestore.
 */
export function cleanOrphanTasks(tasks: Task[]): { cleaned: Task[]; fixedCount: number } {
  if (!tasks || tasks.length === 0) return { cleaned: tasks, fixedCount: 0 };

  const taskMap = new Map<string, Task>();
  const taskCodeMap = new Map<string, Task>();

  tasks.forEach(t => {
    if (t.id) taskMap.set(String(t.id).toLowerCase(), t);
    if (t.taskCodePhu) taskCodeMap.set(String(t.taskCodePhu).trim().toLowerCase(), t);
  });

  const now = new Date().toISOString();
  let fixedCount = 0;
  const orphanTaskUpdates: (Partial<Task> & { id: string })[] = [];

  const cleaned = tasks.map(t => {
    const rawParent = t.parentTaskId || (t as any).parentId;
    if (!rawParent) return t;

    const parentStr = String(rawParent).trim().toLowerCase();
    const parentTask = taskMap.get(parentStr) || taskCodeMap.get(parentStr);

    if (!parentTask || parentTask.isDeleted || String(parentTask.id) === String(t.id)) {
      // Parent task deleted or missing -> Promote orphan task to root task
      fixedCount++;
      const updated = { ...t, parentTaskId: undefined, updatedAt: now };
      delete (updated as any).parentId;
      orphanTaskUpdates.push({ id: t.id, parentTaskId: undefined, updatedAt: now });
      return updated;
    }

    return t;
  });

  // NOTE: Do NOT automatically delete parentTaskId in Firestore during partial memory loading!
  // When tasks are loaded in chunks or filtered by project, parent tasks may not be in memory yet.
  // Deleting parentTaskId in Firestore would destroy the task hierarchy across other devices.

  return { cleaned, fixedCount };
}

/**
 * Shared State Handler: Sắp xếp / Đổi toàn bộ danh sách Tasks (Reorder / Batch Sort)
 */
export async function reorderTasksStateUnified(
  allUpdatedTasks: Task[],
  options?: { affectedProjectIds?: string[]; syncProjects?: boolean }
): Promise<void> {
  if (!allUpdatedTasks) return;

  // 1. Cập nhật toàn bộ Tasks vào Zustand
  useAppStore.getState().setTasks(allUpdatedTasks);

  if (options?.syncProjects === false) return;

  const allProjects = useAppStore.getState().projects;
  let targetProjects: Project[] = [];

  if (options?.affectedProjectIds && options.affectedProjectIds.length > 0) {
    const idSet = new Set(options.affectedProjectIds);
    targetProjects = allProjects.filter(p => p.id && idSet.has(p.id));
  } else {
    targetProjects = findAffectedProjects(allUpdatedTasks, allProjects);
  }

  await syncAffectedProjects(targetProjects, allUpdatedTasks);
}

export function matchesAdvancedSearch(
  searchTerm: string,
  mainFields: (string | number | boolean | undefined | null)[],
  subFields: (string | number | boolean | undefined | null)[] = [],
  evaluateSubItems?: (term: string, andTerms: string[]) => boolean,
): boolean {
  if (!searchTerm) return true;
  const orGroups = searchTerm
    .split(",")
    .map((s) => s.trim().toLowerCase())
    .filter(Boolean);
  if (orGroups.length === 0) return true;

  return orGroups.some((orGroup) => {
    const andTerms = orGroup
      .split("+")
      .map((s) => s.trim())
      .filter(Boolean);
    if (andTerms.length === 0) return false;

    return andTerms.every((term) => {
      let isExact = false;
      let actualTerm = term;
      if (term.startsWith('"') && term.endsWith('"') && term.length >= 2) {
        isExact = true;
        actualTerm = term.slice(1, -1);
      }

      const inMain = mainFields.some((v) => {
        const val = String(v || "").toLowerCase();
        return isExact ? val === actualTerm : val.includes(actualTerm);
      });
      if (inMain) return true;
      if (subFields.length > 0) {
        const inSub = subFields.some((v) => {
          const val = String(v || "").toLowerCase();
          return isExact ? val === actualTerm : val.includes(actualTerm);
        });
        if (inSub) return true;
      }
      if (evaluateSubItems) {
        return evaluateSubItems(actualTerm, andTerms);
      }
      return false;
    });
  });
}

export function formatToDatetimeLocal(d: Date): string {
  if (!d || isNaN(d.getTime())) return "";
  const pad = (n: number) => String(n).padStart(2, "0");
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
}

export function getDefaultDatetimeLocal(): string {
  const d = new Date();
  d.setHours(21, 0, 0, 0);
  return formatToDatetimeLocal(d);
}

export function getSuggestedCompletionTime(baseDate?: string | Date): string {
  const d = baseDate ? new Date(baseDate) : new Date();
  const currentHour = d.getHours();
  if (currentHour < 12) {
    d.setHours(9, 0, 0, 0);
  } else {
    d.setHours(21, 0, 0, 0);
  }
  return formatToDatetimeLocal(d);
}

export function parseDateTime(dateStr: string | undefined): Date | null {
  if (!dateStr) return null;
  if (dateStr.length === 10) return new Date(dateStr + "T09:00");
  return new Date(dateStr);
}

export function calculateDays(
  start: string | undefined,
  finish: string | undefined,
): string {
  if (!start || !finish) return "";
  const s = parseDateTime(start);
  const f = parseDateTime(finish);
  if (!s || !f) return "";
  const diffTime = f.getTime() - s.getTime();
  const diffDays = diffTime / (1000 * 60 * 60 * 24);
  return (diffDays >= 0 ? Number(diffDays.toFixed(2)) : 0) + " days";
}

export const getUserAllowedStores = (
  currentUser: AppUser | null | undefined,
): string[] => {
  if (!currentUser || currentUser.role === "admin") return ["HCM", "HN"];

  // CP & Manager roles default to both stores unless explicitly disabled (set to false)
  if (currentUser.role === "CP" || currentUser.role === "Manager") {
    const list: string[] = [];
    if (currentUser.storePermissions?.hcm !== false) list.push("HCM");
    if (currentUser.storePermissions?.hn !== false) list.push("HN");
    return list.length > 0 ? list : ["HCM", "HN"];
  }

  const list: string[] = [];
  if (currentUser.storePermissions?.hcm) list.push("HCM");
  if (currentUser.storePermissions?.hn) list.push("HN");

  // Default to all stores if storePermissions is not explicitly configured
  if (list.length === 0 && (!currentUser.storePermissions || (currentUser.storePermissions.hcm === undefined && currentUser.storePermissions.hn === undefined))) {
    return ["HCM", "HN"];
  }

  return list.length > 0 ? list : ["HCM", "HN"];
};

export const getEffectiveStoreFilter = (
  globalStoreFilter: string,
  currentUser: AppUser | null | undefined,
): string => {
  if (!currentUser || currentUser.role === "admin") return globalStoreFilter || "all";
  const allowed = getUserAllowedStores(currentUser);
  if (allowed.length === 1) {
    return allowed[0];
  }
  if (globalStoreFilter && globalStoreFilter !== "all" && !allowed.includes(globalStoreFilter)) {
    return allowed[0] || "all";
  }
  return globalStoreFilter || "all";
};

export const getDefaultStoreFilter = (
  currentUser: AppUser | null | undefined,
): string => {
  if (!currentUser) return "all";

  const allowed = getUserAllowedStores(currentUser);
  if (allowed.length === 1) return allowed[0];

  let savedHabit: string | null = null;
  try {
    savedHabit = safeGetItemLocal(`taka_store_filter_${currentUser.id}`);
  } catch (e) {
    console.warn("Could not read localStorage", e);
  }

  if (currentUser.role === "admin" || currentUser.locationPermissions?.['all'] || currentUser.partyPermissions?.['all']) return savedHabit || "all";

  if (savedHabit) {
    if (savedHabit.toUpperCase() === "HCM" && currentUser.storePermissions?.hcm)
      return "HCM";
    if (savedHabit.toUpperCase() === "HN" && currentUser.storePermissions?.hn)
      return "HN";
    if (
      savedHabit === "all" &&
      currentUser.storePermissions?.hcm &&
      currentUser.storePermissions?.hn
    )
      return "all";
  }

  if (currentUser.storePermissions?.hcm && currentUser.storePermissions?.hn)
    return "all";
  if (currentUser.storePermissions?.hcm) return "HCM";
  if (currentUser.storePermissions?.hn) return "HN";
  return "all";
};

export const resolveActualStatus = (
  plStatus: string,
  subTasks: any[],
  task?: { runningStatus?: string; approval?: string; status?: string; [key: string]: any }
) => {
  if (plStatus === "Canceled") {
    return {
      label: "Cancel",
      color: "bg-slate-400 text-white border-slate-500",
    };
  }

  if (!subTasks || subTasks.length === 0) {
    if (task) {
      const running = (task.runningStatus || '').toLowerCase();
      const approval = (task.approval || '').toLowerCase();
      if (running.includes('complete') || approval === 'approved' || running.includes('finish')) {
        return {
          label: "Completed",
          color: "bg-emerald-50 dark:bg-[#28282a] text-emerald-600 dark:text-[#81c995] border-emerald-200 dark:border-emerald-800",
        };
      }
      if (running.includes('delay')) {
        return {
          label: "Delayed",
          color: "bg-amber-100 dark:bg-amber-900/30 text-amber-800 dark:text-amber-300 border-amber-300 dark:border-amber-700",
        };
      }
      if (running.includes('process') || running.includes('progress') || running.includes('accept')) {
        return {
          label: "In Progress",
          color: "bg-blue-100 dark:bg-blue-900/30 text-blue-800 dark:text-blue-300 border-blue-300 dark:border-blue-700",
        };
      }
      if (running.includes('cancel')) {
        return {
          label: "Cancel",
          color: "bg-slate-400 text-white border-slate-500",
        };
      }
    }

    const pLower = (plStatus || '').toLowerCase();
    if (pLower === 'completed') {
      return {
        label: "Completed",
        color: "bg-emerald-50 dark:bg-[#28282a] text-emerald-600 dark:text-[#81c995] border-emerald-200 dark:border-emerald-800",
      };
    }
    if (pLower === 'on process' || pLower === 'handover') {
      return {
        label: "In Progress",
        color: "bg-blue-100 dark:bg-blue-900/30 text-blue-800 dark:text-blue-300 border-blue-300 dark:border-blue-700",
      };
    }
    return {
      label: "Upcoming",
      color: "bg-slate-50 dark:bg-[#28282a] text-slate-600 dark:text-[#9aa0a6] border-slate-200 dark:border-[#3c4043]",
    };
  }

  const statuses = subTasks.map((st) =>
    (st.delegationStatus || st.status || "").toLowerCase(),
  );

  let T_group = "Assigned";
  if (statuses.some((s) => ["rejected", "needs rework", "reqreview"].includes(s))) {
    T_group = "Rejected";
  } else if (
    statuses.every((s) => ["completed", "approved", "finish", "finished"].includes(s) || s === "")
  ) {
    if (statuses.some((s) => s !== "")) {
      T_group = "Completed";
    }
  } else if (
    statuses.some((s) =>
      [
        "pending completed",
        "waiting for approval",
        "hold",
        "requested",
        "reqapproval",
      ].includes(s),
    )
  ) {
    T_group = "Pending Completed";
  } else if (
    statuses.some((s) =>
      [
        "accepted",
        "on process",
        "in progress",
        "inprogress",
        "doing",
        "accepttask",
      ].includes(s),
    )
  ) {
    T_group = "In Progress";
  }

  const pLower = (plStatus || "").toLowerCase();
  const P_group =
    pLower === "completed"
      ? "Completed"
      : pLower === "on process" || pLower === "handover"
        ? "On Process"
        : "Upcoming";

  if (T_group === "Completed") {
    return {
      label: "Completed",
      color:
        "bg-emerald-50 dark:bg-[#28282a] text-emerald-600 dark:text-[#81c995] border-emerald-200 dark:border-emerald-800",
    };
  }

  if (T_group === "Rejected") {
    return {
      label: "Correction Needed",
      color: "bg-red-500 text-white border-red-600",
    };
  }

  if (T_group === "Pending Completed") {
    if (P_group === "Completed") {
      return {
        label: "Review Late",
        color: "bg-orange-500 text-white border-orange-600",
      };
    }
    return {
      label: "Pending Approval",
      color: "bg-yellow-100 dark:bg-yellow-900/30 text-yellow-800 dark:text-yellow-300 border-yellow-300 dark:border-yellow-700",
    };
  }

  if (T_group === "In Progress") {
    if (P_group === "Completed") {
      return {
        label: "Delayed",
        color: "bg-amber-100 dark:bg-amber-900/30 text-amber-800 dark:text-amber-300 border-amber-300 dark:border-amber-700",
      };
    }
    return {
      label: "In Progress",
      color: "bg-blue-100 dark:bg-blue-900/30 text-blue-800 dark:text-blue-300 border-blue-300 dark:border-blue-700",
    };
  }

  // T_group === "Assigned"
  if (P_group === "Completed") {
    return {
      label: "Critical Delay",
      color: "bg-red-800 text-red-100 border-red-900",
    };
  }
  if (P_group === "On Process") {
    return {
      label: "In Progress",
      color: "bg-blue-100 dark:bg-blue-900/30 text-blue-800 dark:text-blue-300 border-blue-300 dark:border-blue-700",
    };
  }

  return {
    label: "Planned",
    color: "bg-slate-200 dark:bg-slate-800 text-slate-700 dark:text-slate-300 border-slate-300 dark:border-slate-700",
  };
};

export function getStatus(
  start: string | undefined,
  handover: string | undefined,
  opening: string | undefined,
  code: string = "",
) {
  if (code && String(code).toLowerCase().includes("cancel")) return "Canceled";
  if (!start) return "Upcoming";
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const dStart = parseDateTime(start);
  if (!dStart) return "Upcoming";
  dStart.setHours(0, 0, 0, 0);

  const dHandover = handover ? parseDateTime(handover) : null;
  if (dHandover) dHandover.setHours(0, 0, 0, 0);

  const dOpening = opening ? parseDateTime(opening) : null;
  if (dOpening) dOpening.setHours(0, 0, 0, 0);

  const dEnd =
    dOpening && dHandover
      ? dOpening > dHandover
        ? dOpening
        : dHandover
      : dOpening || dHandover;

  if (today < dStart) return "Upcoming";
  if (dEnd && today > dEnd) return "Completed";

  if (dHandover && dOpening && today >= dHandover && today <= dOpening)
    return "Handover";

  return "On process";
}

export function computePlanProgressFromSummaries(
  summaries: Array<{ start?: string; finish?: string; [key: string]: any }>
): number {
  if (!summaries || summaries.length === 0) return 0;
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const todayTime = today.getTime();

  let totalScore = 0;
  let validCount = 0;

  for (const s of summaries) {
    if (!s.start && !s.finish) continue;
    validCount++;

    const start = parseDateTime(s.start);
    if (start) start.setHours(0, 0, 0, 0);
    const finish = parseDateTime(s.finish);
    if (finish) finish.setHours(0, 0, 0, 0);

    if (finish && todayTime > finish.getTime()) {
      totalScore += 1;
    } else if (start && finish && todayTime >= start.getTime() && todayTime <= finish.getTime()) {
      const totalDuration = finish.getTime() - start.getTime();
      if (totalDuration > 0) {
        const passed = todayTime - start.getTime();
        totalScore += passed / totalDuration;
      } else {
        totalScore += 1;
      }
    }
  }

  if (validCount === 0) return 0;
  return Math.round((totalScore / validCount) * 100);
}

export function computeActProgressFromSummaries(
  summaries: Array<{ start?: string; finish?: string; runningStatus?: string; approval?: string; [key: string]: any }>
): number {
  if (!summaries || summaries.length === 0) return 0;
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const todayTime = today.getTime();

  const validScores: number[] = [];

  for (const s of summaries) {
    if (!s.start && !s.finish && !s.runningStatus && !s.actStatus && !s.status) continue;

    const statusStr = String(s.runningStatus || s.actStatus || s.status || s.delegationStatus || '').toLowerCase();
    const approvalStr = String(s.approval || s.approvalStatus || '').toLowerCase();

    if (statusStr.includes('complete') || statusStr.includes('finished') || approvalStr === 'approved') {
      validScores.push(1.0);
      continue;
    }
    if (statusStr.includes('pending') || statusStr.includes('reqapproval') || statusStr.includes('waiting') || approvalStr === 'reqapproval' || approvalStr === 'requested') {
      validScores.push(0.8);
      continue;
    }
    if (statusStr.includes('accept') || statusStr.includes('process') || statusStr.includes('progress') || statusStr.includes('delay') || approvalStr === 'accepttask') {
      validScores.push(0.5);
      continue;
    }
    if (statusStr.includes('rework') || statusStr.includes('reject') || approvalStr === 'rejected' || approvalStr === 'reqreview') {
      validScores.push(0.3);
      continue;
    }

    const start = parseDateTime(s.start);
    if (start) start.setHours(0, 0, 0, 0);
    const finish = parseDateTime(s.finish);
    if (finish) finish.setHours(0, 0, 0, 0);

    if (finish && todayTime > finish.getTime()) {
      validScores.push(1.0);
    } else if (start && finish && todayTime >= start.getTime() && todayTime <= finish.getTime()) {
      const totalDuration = finish.getTime() - start.getTime();
      if (totalDuration > 0) {
        const passed = todayTime - start.getTime();
        validScores.push(Math.max(0.3, Math.min(0.9, passed / totalDuration)));
      } else {
        validScores.push(1.0);
      }
    } else {
      validScores.push(0.0);
    }
  }

  if (validScores.length === 0) return 0;

  let maxCompletedIdx = -1;
  for (let i = 0; i < validScores.length; i++) {
    if (validScores[i] >= 1.0) {
      maxCompletedIdx = i;
    }
  }

  let totalScore = 0;
  for (let i = 0; i < validScores.length; i++) {
    if (maxCompletedIdx !== -1 && i <= maxCompletedIdx) {
      totalScore += 1.0;
    } else {
      totalScore += validScores[i];
    }
  }

  return Math.round((totalScore / validScores.length) * 100);
}

export function getProjectDatesAndStatus(p: Project, tasks?: Task[]) {
  let computedTakStart = p.computedTakStart || p.start;
  let computedTakCompl: string | undefined = p.computedTakCompl || p.handover || (p as any).takCompl;
  let computedOpening = p.computedOpening || (p.openingMode === 'manual' ? p.opening : '');

  let derivedFromTasks = false;
  let projectTasks: Task[] = [];
  if (tasks && tasks.length > 0) {
    const pCode = String(p.code || "").trim().toLowerCase();
    const pYear = p.year !== undefined && p.year !== null ? String(p.year).trim() : "";
    const pStore = p.store ? String(p.store).trim().toLowerCase() : "";

    let hasOpeningTask = false;
    tasks.forEach(t => {
      if (!t.projectCode) return;
      const tCode = String(t.projectCode).trim().toLowerCase();
      if (tCode !== pCode) return;
      if (pYear && t.projectYear !== undefined && t.projectYear !== null && String(t.projectYear).trim() !== pYear) return;
      if (pStore && t.projectStore && String(t.projectStore).trim().toLowerCase() !== pStore) return;

      derivedFromTasks = true;
      projectTasks.push(t);
      if (t.start && (!computedTakStart || t.start < computedTakStart)) {
        computedTakStart = t.start;
      }
      if (t.finish && (!computedTakCompl || t.finish > computedTakCompl)) {
        computedTakCompl = t.finish;
      }
      const name = t.name ? String(t.name).toLowerCase() : "";
      if ((name.includes("opening") || name.includes("open")) && t.finish) {
        if (!hasOpeningTask || t.finish > computedOpening) {
          computedOpening = t.finish;
          hasOpeningTask = true;
        }
      }
    });
    if (!hasOpeningTask && p.openingMode !== 'manual') {
      computedOpening = '';
    }
  }

  if (!derivedFromTasks && p.taskSummaries && Array.isArray(p.taskSummaries) && p.taskSummaries.length > 0) {
    let hasOpeningTask = false;
    p.taskSummaries.forEach(ts => {
      if (ts.start && (!computedTakStart || ts.start < computedTakStart)) {
        computedTakStart = ts.start;
      }
      if (ts.finish && (!computedTakCompl || ts.finish > computedTakCompl)) {
        computedTakCompl = ts.finish;
      }
      const name = ts.name ? String(ts.name).toLowerCase() : "";
      if ((name.includes("opening") || name.includes("open")) && ts.finish) {
        if (!hasOpeningTask || ts.finish > computedOpening) {
          computedOpening = ts.finish;
          hasOpeningTask = true;
        }
      }
    });
    if (!hasOpeningTask && p.openingMode !== 'manual') {
      computedOpening = '';
    }
  }

  const status = p.computedStatus || getStatus(computedTakStart, computedTakCompl, computedOpening || p.opening, p.code);

  let planProgressPercent = p.computedPlanProgress;
  let actProgressPercent = p.computedActProgress;

  if (projectTasks.length > 0) {
    planProgressPercent = computePlanProgressPercent(projectTasks);
    actProgressPercent = computeActProgressPercent(projectTasks, []);
  } else if (p.taskSummaries && Array.isArray(p.taskSummaries) && p.taskSummaries.length > 0) {
    if (planProgressPercent === undefined || planProgressPercent === null || planProgressPercent === 0) {
      planProgressPercent = computePlanProgressFromSummaries(p.taskSummaries);
    }
    if (actProgressPercent === undefined || actProgressPercent === null || actProgressPercent === 0) {
      actProgressPercent = computeActProgressFromSummaries(p.taskSummaries);
    }
  }

  return {
    computedTakStart,
    computedTakCompl,
    computedOpening,
    status,
    planProgressPercent: planProgressPercent ?? 0,
    actProgressPercent: actProgressPercent ?? 0
  };
}

export function matchProjectStatusFilter(
  status: string | undefined,
  filterValue: string,
  startStr?: string
): boolean {
  if (!filterValue || filterValue === 'all') return true;

  const currentStatus = status || 'Upcoming';

  if (filterValue === 'On process & Upcoming' || filterValue === 'On process + Upcoming') {
    return currentStatus === 'On process' || currentStatus === 'Upcoming' || currentStatus === 'Handover';
  }

  if (filterValue === 'On process + Upcoming 3 days') {
    if (currentStatus === 'On process' || currentStatus === 'Handover') {
      return true;
    }
    if (currentStatus === 'Upcoming') {
      if (!startStr) return false;
      const today = new Date();
      today.setHours(0, 0, 0, 0);
      const dStart = parseDateTime(startStr);
      if (!dStart) return false;
      dStart.setHours(0, 0, 0, 0);
      const diffDays = Math.ceil((dStart.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
      return diffDays <= 3 && diffDays >= 0;
    }
    return false;
  }

  return currentStatus.toLowerCase() === filterValue.toLowerCase();
}

const INVALID_UNIT_TOKENS = new Set([
  'b1', 'b2', 'b3', 'l1', 'l2', 'l3', 'l4', 'l5',
  '1f', '2f', '3f', '4f', '5f', 'gf', 'ug', 'mz',
  'f1', 'f2', 'f3', 'f4', 'f5',
  'hcm', 'hn', 'taka', 'takashimaya', 'pm', 'project',
  'reserve', 'event', 'common', 'area', 'mall', 'store', 'floor', 'all', 'na', 'null', 'undefined'
]);

/**
 * Strips common floor/zone/PM prefixes like 'PM-L1-HCM-2026-', 'L1-', 'F1-', '1F-', 'B1-', 'GF-', 'UG-', 'Unit '
 */
function normalizeUnitCodeBase(code: string): string {
  if (!code) return '';
  let str = code.toLowerCase().trim();
  // Remove leading 'unit ' or 'shop ' or 'lot ' or 'booth '
  str = str.replace(/^(unit|shop|lot|booth|kiosk|store)\s*[-_:]?\s*/i, '');
  // Remove flexible PM prefix like 'pm-l1-hcm-2026-', 'pm-b1-hcm-', 'pm-hcm-l1-', 'pm-l2-hn-', 'pm-hn-l2-', 'pm-l2-', 'pm-2027-l2-'
  str = str.replace(/^pm-(?:(?:b\d|l\d|\d+f|gf|ug|hcm|hn|20\d\d|y\d\d)[-_:]\s*)+/i, '');
  // Remove store prefix like 'hn-', 'hcm-' if followed by floor or unit (e.g. 'hn-l2-201' -> 'l2-201')
  str = str.replace(/^(?:hcm|hn)[-_:]\s*/i, '');
  // Remove floor prefixes like l1-, l2-, f1-, 1f-, 2f-, b1-, b2-, gf-, ug- (strictly 1-2 digits to avoid stripping unit numbers like 219-)
  str = str.replace(/^(?:[blf]\d{1,2}|\d{1,2}f|gf|ug|mz|basement|ground)\s*[-_:]\s*/i, '');
  return str.trim();
}

/**
 * Extracts all project codes from both project.code and project.linkCodes
 */
export function extractAllProjectCodes(p: { code?: string | null; linkCodes?: string | null }): string[] {
  const codeSet = new Set<string>();

  const addIfValid = (rawCode: string) => {
    const trimmed = (rawCode || '').trim().toLowerCase();
    if (!trimmed || INVALID_UNIT_TOKENS.has(trimmed)) return;
    codeSet.add(trimmed);

    // Also extract base code without floor prefix (e.g. for 'l2-219-3-2' add '219-3-2')
    const baseCode = normalizeUnitCodeBase(trimmed);
    if (baseCode && !INVALID_UNIT_TOKENS.has(baseCode)) {
      codeSet.add(baseCode);
    }

    // If PM standard code e.g. PM-L1-HCM-2026-DC or PM-L1-HCM-DC, also extract DC and L1-DC
    const pmMatch = trimmed.match(/^pm-(?:b\d|l\d|\d+f|gf|ug|hcm|hn)-(?:hcm|hn|b\d|l\d|\d+f|gf|ug)(?:-(?:20\d\d|y\d\d))?-([a-z0-9_-]+)$/i);
    if (pmMatch && pmMatch[1]) {
      const unitPart = pmMatch[1].toLowerCase();
      if (!INVALID_UNIT_TOKENS.has(unitPart)) {
        codeSet.add(unitPart);
        const locMatch = trimmed.match(/(b\d|l\d|\d+f)/i);
        if (locMatch) {
          let fl = locMatch[1].toLowerCase();
          if (fl.endsWith('f')) fl = `l${fl[0]}`;
          codeSet.add(`${fl}-${unitPart}`);
        }
      }
    }
  };

  // When project code link is explicitly provided, ONLY use the linkCodes to define linked units!
  // This cleanly isolates linked units and prevents leaking into p.code (which could be a parent unit or different code).
  const hasCustomLinkCodes = Boolean(p.linkCodes && String(p.linkCodes).trim());

  if (hasCustomLinkCodes) {
    String(p.linkCodes)
      .split(/[\r\n\t,;+/&|]+|\s{2,}/)
      .map(c => c.trim().toLowerCase())
      .filter(Boolean)
      .forEach(c => {
        addIfValid(c);
      });
  } else if (p.code && String(p.code).trim()) {
    addIfValid(String(p.code));
  }

  return Array.from(codeSet);
}

/**
 * Extracts normalized Year, Store, Location (Floor) from project properties and code/linkCodes.
 * Follows enterprise data rules for Takashimaya project codes:
 * - Year: p.year, or 4-digit/2-digit year (e.g. y2026, 2026, Y26)
 * - Store: p.store, or HCM / HN in code
 * - Location: p.floor / p.location, or floor code (B1, B2, L1, L2, L3, L4, L5, 1F..5F)
 */
export function extractProjectMetadata(p: any): { year: string; store: string; location: string } {
  if (!p) return { year: '', store: '', location: '' };

  // 1. Year
  let year = String(p.year || (p as any).projectYear || '').trim();
  if (!year || year === 'undefined' || year === 'null') {
    const rawSearch = `${p.code || ''} ${p.linkCodes || ''} ${p.name || ''}`;
    // Match 2024-2035 with optional y/Y prefix, e.g. y2026, Y2026, 2026
    const yrMatch = rawSearch.match(/(?:^|[^0-9a-zA-Z])(?:y|Y)?(202\d|203\d)(?:[^0-9a-zA-Z]|$)/i) 
      || rawSearch.match(/y(202\d|203\d)\b/i)
      || rawSearch.match(/\b(202\d|203\d)\b/);
    if (yrMatch) {
      year = yrMatch[1];
    } else {
      const yrShortMatch = rawSearch.match(/(?:^|[^0-9a-zA-Z])(?:y|Y)(2[4-9]|3[0-5])(?:[^0-9a-zA-Z]|$)/i);
      if (yrShortMatch) {
        year = `20${yrShortMatch[1]}`;
      } else if (p.start || p.opening || p.handover || p.computedTakStart || p.computedTakCompl || p.takStart || p.takCompl) {
        const dStr = p.start || p.opening || p.handover || p.computedTakStart || p.computedTakCompl || p.takStart || p.takCompl;
        const dMatch = String(dStr).match(/\b(202\d|203\d)\b/);
        if (dMatch) year = dMatch[1];
      }
    }
  }

  // 2. Store
  let store = String(p.store || (p as any).projectStore || '').trim().toUpperCase();
  if (!store || store === 'UNDEFINED' || store === 'NULL') {
    const rawSearch = `${p.code || ''} ${p.linkCodes || ''} ${p.name || ''}`;
    if (/(?:^|[^a-zA-Z])HCM(?:[^a-zA-Z]|$)/i.test(rawSearch)) {
      store = 'HCM';
    } else if (/(?:^|[^a-zA-Z])HN(?:[^a-zA-Z]|$)/i.test(rawSearch)) {
      store = 'HN';
    }
  }

  // 3. Location (Floor)
  let rawLoc = String((p as any).floor || p.location || (p as any).projectLocation || '').trim().toUpperCase();
  let location = '';
  if (rawLoc && rawLoc !== 'UNDEFINED' && rawLoc !== 'NULL') {
    const m = rawLoc.match(/(?:^|[^a-zA-Z0-9])(B\d|L\d|\d+F|\d+L|GF|MF|RF|BASEMENT\s*\d?|LEVEL\s*\d+[a-zA-Z]?|FLOOR\s*\d+[a-zA-Z]?)(?:[^a-zA-Z0-9]|$)/i);
    if (m) {
      let token = m[1].toUpperCase().replace(/\s+/g, '');
      if (token.startsWith('BASEMENT')) token = token.replace('BASEMENT', 'B') || 'B1';
      if (token.startsWith('LEVEL')) {
        const num = token.match(/LEVEL(\d+)/i);
        token = num ? `L${num[1]}` : 'L1';
      }
      if (token.startsWith('FLOOR')) {
        const num = token.match(/FLOOR(\d+)/i);
        token = num ? `L${num[1]}` : 'L1';
      }
      if (/^\d+F$/.test(token)) token = 'L' + token.replace('F', '');
      if (/^\d+L$/.test(token)) token = 'L' + token.replace('L', '');
      if (/^L\d+[A-Z]$/i.test(token)) token = token.slice(0, -1);
      location = token;
    } else {
      location = rawLoc;
    }
  }

  if (!location || location === 'UNDEFINED' || location === 'NULL') {
    const rawSearch = `${p.code || ''} ${p.linkCodes || ''} ${p.name || ''}`;
    const m = rawSearch.match(/(?:^|[^a-zA-Z0-9])(B\d|L\d|\d+F|\d+L|GF|MF|RF|BASEMENT\s*\d?|LEVEL\s*\d+[a-zA-Z]?|FLOOR\s*\d+[a-zA-Z]?)(?:[^a-zA-Z0-9]|$)/i);
    if (m) {
      let token = m[1].toUpperCase().replace(/\s+/g, '');
      if (token.startsWith('BASEMENT')) token = token.replace('BASEMENT', 'B') || 'B1';
      if (token.startsWith('LEVEL')) {
        const num = token.match(/LEVEL(\d+)/i);
        token = num ? `L${num[1]}` : 'L1';
      }
      if (token.startsWith('FLOOR')) {
        const num = token.match(/FLOOR(\d+)/i);
        token = num ? `L${num[1]}` : 'L1';
      }
      if (/^\d+F$/.test(token)) token = 'L' + token.replace('F', '');
      if (/^\d+L$/.test(token)) token = 'L' + token.replace('L', '');
      if (/^L\d+[A-Z]$/i.test(token)) token = token.slice(0, -1);
      location = token;
    }
  }

  if (location && location.endsWith('F')) {
    location = `L${location[0]}`;
  }
  if (/^L\d+[A-Z]$/i.test(location)) {
    location = location.slice(0, -1);
  }

  return { year, store, location };
}

/**
 * Checks if a project matches the requested scope (Store, Location, Year)
 */
export function projectMatchesScope(
  meta: { year: string; store: string; location: string },
  scope: { store?: string; location?: string; year?: string }
): boolean {
  // Store check
  if (scope.store && scope.store !== 'all') {
    const targetStore = scope.store.toUpperCase().trim();
    if (meta.store && meta.store !== targetStore) return false;
  }

  // Location / Floor check
  if (scope.location && scope.location !== 'all') {
    let targetLoc = scope.location.toUpperCase().trim();
    if (targetLoc.endsWith('F')) targetLoc = `L${targetLoc[0]}`;
    if (/^L\d+[A-Z]$/i.test(targetLoc)) targetLoc = targetLoc.slice(0, -1);
    if (meta.location) {
      let pLoc = meta.location.toUpperCase().trim();
      if (pLoc.endsWith('F')) pLoc = `L${pLoc[0]}`;
      if (/^L\d+[A-Z]$/i.test(pLoc)) pLoc = pLoc.slice(0, -1);
      const isStorewide = pLoc === 'ALL' || pLoc === 'STORE' || pLoc === 'STOREWIDE' || pLoc === 'WHOLE STORE' || pLoc === 'ALL FLOORS';
      if (!isStorewide && pLoc !== targetLoc) return false;
    }
  }

  // Year check (Strict Isolation)
  if (scope.year && scope.year !== 'all') {
    const targetYear = scope.year.trim();
    if (meta.year && meta.year !== targetYear) return false;
  }

  return true;
}

/**
 * Checks if a project's codes match a given unit name (exact or normalized floor-prefix match).
 * Strictly isolates parent-lot and sub-lots so that e.g. L2-219-3-2 never matches L2-219 or vice-versa.
 */
export function projectMatchesUnitCode(pCodes: string[], unitName: string): boolean {
  const u = (unitName || '').toLowerCase().trim();
  if (!u || INVALID_UNIT_TOKENS.has(u)) return false;
  const uStripped = normalizeUnitCodeBase(u);

  for (const rawCode of pCodes) {
    const c = (rawCode || '').toLowerCase().trim();
    if (!c || INVALID_UNIT_TOKENS.has(c)) continue;

    // 1. Exact match
    if (u === c) return true;

    // 2. Base code match after stripping floor prefixes (e.g. 'L2-219-3-2' matching '219-3-2' or 'L1-01' matching '01')
    const cStripped = normalizeUnitCodeBase(c);
    if (uStripped && cStripped && !INVALID_UNIT_TOKENS.has(uStripped) && !INVALID_UNIT_TOKENS.has(cStripped)) {
      if (uStripped === cStripped) return true;
    }
  }
  return false;
}

/**
 * Removes a specific unit code from a linkCodes string.
 * Example: removeCodeFromLinkCodesString("101, 102, 103", "102") -> "101, 103"
 */
export function removeCodeFromLinkCodesString(linkCodesStr: string | null | undefined, codeToRemove: string): string {
  if (!linkCodesStr) return '';
  const target = codeToRemove.trim().toLowerCase();
  if (!target) return linkCodesStr;
  const parts = String(linkCodesStr)
    .split(/[\r\n\t,;+/&|]+|\s{2,}/)
    .map(c => c.trim())
    .filter(Boolean);
  const filtered = parts.filter(c => c.toLowerCase() !== target);
  return filtered.join(', ');
}


export function getTaskStatus(
  start: string | undefined,
  finish: string | undefined,
): string {
  if (!start && !finish) return "Upcoming";
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const dStart = parseDateTime(start);
  if (dStart) dStart.setHours(0, 0, 0, 0);

  const dFinish = parseDateTime(finish);
  if (dFinish) dFinish.setHours(0, 0, 0, 0);

  if (dStart && today < dStart) return "Upcoming";
  if (dFinish && today > dFinish) return "Completed";

  return "On process";
}

export function ensureUrlProtocol(url: string | undefined): string {
  if (!url) return "#";
  const trimmedUrl = url.trim();
  if (trimmedUrl.match(/^[a-zA-Z]+:\/\//) || trimmedUrl.startsWith("mailto:")) {
    return trimmedUrl;
  }
  return `http://${trimmedUrl}`;
}

export function formatDisplayDateTime(dateStr: string | undefined): string {
  if (!dateStr) return "";
  const d = parseDateTime(dateStr);
  if (!d || isNaN(d.getTime())) return dateStr || "";
  return d.toLocaleDateString("en-US", {
    month: "short",
    day: "numeric",
    year: "numeric",
    hour: "numeric",
    minute: "2-digit",
    hour12: true,
  });
}

export function matchProjectCode(codeA?: string, codeB?: string): boolean {
  if (!codeA || !codeB) return false;
  const a = String(codeA).trim().toLowerCase();
  const b = String(codeB).trim().toLowerCase();
  if (a === b) return true;
  const partsA = a.split(/[,;/]+/).map(s => s.trim()).filter(Boolean);
  const partsB = b.split(/[,;/]+/).map(s => s.trim()).filter(Boolean);
  return partsA.some(pa => partsB.includes(pa));
}

export function getTasksForProject(project: Partial<Project>, allTasks: Task[]): Task[] {
  if (!project.code || !allTasks || allTasks.length === 0) return [];
  const pYear = project.year !== undefined && project.year !== null ? String(project.year).trim() : '';
  const pStore = project.store ? String(project.store).trim().toLowerCase() : '';
  const rawPCode = String(project.code).trim().toLowerCase();
  const pCodeParts = rawPCode.split(/[,;/]+/).map(s => s.trim()).filter(Boolean);

  return allTasks.filter(t => {
    if (!t.projectCode) return false;
    const tCode = String(t.projectCode).trim().toLowerCase();
    const matchesCode = tCode === rawPCode || pCodeParts.includes(tCode) || matchProjectCode(t.projectCode, project.code);
    if (!matchesCode) return false;

    const tStore = t.projectStore ? String(t.projectStore).trim().toLowerCase() : '';
    if (pStore && tStore && pStore !== tStore) return false;

    const tYear = t.projectYear !== undefined && t.projectYear !== null ? String(t.projectYear).trim() : '';
    if (pYear && tYear && pYear !== tYear) return false;

    return true;
  });
}

export function computeProjectActStatus(
  p: Project,
  delegationGroups: DelegationGroup[] = [],
  matchedTasks: Task[] = []
): string {
  if ((p as any).actStatus) return String((p as any).actStatus);
  if ((p as any).actualStatus) return String((p as any).actualStatus);

  if (delegationGroups && delegationGroups.length > 0) {
    const pCode = String(p.code || '').trim().toLowerCase();
    const matchedTaskIds = new Set(matchedTasks.map(t => t.id).filter(Boolean));
    const pGroups = delegationGroups.filter(g => {
      if (g.projectCode && String(g.projectCode).trim().toLowerCase() === pCode) return true;
      if (g.parentTaskId && matchedTaskIds.has(g.parentTaskId)) return true;
      return false;
    });

    if (pGroups.length > 0) {
      const statuses = pGroups.map(g => {
        if (g.actStatus) return g.actStatus;
        return resolveActualStatus(p.computedStatus || 'On process', g.subTasks || []).label;
      });
      if (statuses.some(s => s === 'Rejected' || s === 'Correction Needed')) return 'Rejected';
      if (statuses.every(s => s === 'Completed')) return 'Completed';
      if (statuses.some(s => s === 'Pending Completed' || s === 'Pending Approval' || s === 'Review Late')) return 'Pending Completed';
      if (statuses.some(s => s === 'In Progress' || s === 'On Process' || s === 'Delayed' || s === 'Accepted')) return 'In Progress';
      return statuses[0] || 'Planned';
    }
  }

  if (p.computedActProgress === 100) return 'Completed';
  const plStatus = p.computedStatus || getStatus(p.computedTakStart || p.start, p.computedTakCompl || p.handover, p.computedOpening || p.opening, p.code);
  if (plStatus === 'Completed') return 'Completed';
  if (plStatus === 'Canceled') return 'Cancel';
  if (p.computedActProgress && p.computedActProgress > 0) return 'In Progress';
  if (plStatus === 'On process' || plStatus === 'Handover') return 'In Progress';
  return 'Planned';
}

// Module-level cache for zero-cost multi-component calls to enrichProjects
let _lastProjectsRef: Project[] | null = null;
let _lastTasksRef: Task[] | null = null;
let _lastDelegationRef: DelegationGroup[] | null = null;
let _lastEnrichedResult: Project[] | null = null;

export function enrichProjects(
  projects: Project[], 
  tasks: Task[], 
  delegationGroups: DelegationGroup[] = []
): Project[] {
  if (
    _lastProjectsRef === projects &&
    _lastTasksRef === tasks &&
    _lastDelegationRef === delegationGroups &&
    _lastEnrichedResult
  ) {
    return _lastEnrichedResult;
  }

  const tasksByProjectId = new Map<string, Task[]>();
  const tasksByFullKey = new Map<string, Task[]>();
  const tasksByCodeOnly = new Map<string, Task[]>();

  tasks.forEach(t => {
    if (t.isDeleted) return;
    if (t.projectId) {
      const pid = String(t.projectId);
      let pList = tasksByProjectId.get(pid);
      if (!pList) {
        pList = [];
        tasksByProjectId.set(pid, pList);
      }
      pList.push(t);
    }
    if (!t.projectCode) return;
    const tCode = String(t.projectCode).trim().toLowerCase();
    const tYear = t.projectYear !== undefined && t.projectYear !== null ? String(t.projectYear).trim() : "";
    const tStore = t.projectStore ? String(t.projectStore).trim().toLowerCase() : "";

    const tCodeParts = tCode.split(/[,;/]+/).map(s => s.trim()).filter(Boolean);
    const codesToIndex = tCodeParts.length > 0 ? tCodeParts : [tCode];

    codesToIndex.forEach(code => {
      // 1. Full key
      const fullKey = `${code}-${tYear}-${tStore}`;
      let fullList = tasksByFullKey.get(fullKey);
      if (!fullList) {
        fullList = [];
        tasksByFullKey.set(fullKey, fullList);
      }
      fullList.push(t);

      // 2. Code only fallback
      let codeList = tasksByCodeOnly.get(code);
      if (!codeList) {
        codeList = [];
        tasksByCodeOnly.set(code, codeList);
      }
      codeList.push(t);
    });
  });

  const enrichedProjects = projects.map((p) => {
    const pYear = p.year !== undefined && p.year !== null ? String(p.year).trim() : "";
    const pStore = p.store ? String(p.store).trim().toLowerCase() : "";
    const pCode = String(p.code || "").trim().toLowerCase();
    const rawCodeCombined = (p.linkCodes && p.linkCodes.trim()) ? `${pCode}, ${p.linkCodes.toLowerCase()}` : pCode;
    const pCodeParts = rawCodeCombined.split(/[,;/]+/).map(s => s.trim()).filter(Boolean);

    const matchedTaskMap = new Map<string, Task>();

    // Check direct ID match first
    if (p.id) {
      const idTasks = tasksByProjectId.get(String(p.id));
      if (idTasks) {
        for (let i = 0; i < idTasks.length; i++) {
          const t = idTasks[i];
          const tStore = t.projectStore ? String(t.projectStore).trim().toLowerCase() : '';
          const tYear = t.projectYear !== undefined && t.projectYear !== null ? String(t.projectYear).trim() : '';
          if (pStore && tStore && pStore !== tStore) continue;
          if (pYear && tYear && pYear !== tYear) continue;
          matchedTaskMap.set(t.id, t);
        }
      }
    }

    // Check direct key and part keys
    const directKey = `${pCode}-${pYear}-${pStore}`;
    const directTasks = tasksByFullKey.get(directKey);
    if (directTasks) {
      for (let i = 0; i < directTasks.length; i++) {
        const t = directTasks[i];
        if (t.projectId && p.id && String(t.projectId) !== String(p.id)) continue;
        matchedTaskMap.set(t.id, t);
      }
    }

    for (let i = 0; i < pCodeParts.length; i++) {
      const partKey = `${pCodeParts[i]}-${pYear}-${pStore}`;
      const subTasks = tasksByFullKey.get(partKey);
      if (subTasks) {
        for (let j = 0; j < subTasks.length; j++) {
          const t = subTasks[j];
          if (t.projectId && p.id && String(t.projectId) !== String(p.id)) continue;
          matchedTaskMap.set(t.id, t);
        }
      }
    }

    // Fast fallback to code-only if no exact store/year match
    if (matchedTaskMap.size === 0 && tasks.length > 0) {
      for (let i = 0; i < pCodeParts.length; i++) {
        const fallbackTasks = tasksByCodeOnly.get(pCodeParts[i]);
        if (fallbackTasks) {
          for (let j = 0; j < fallbackTasks.length; j++) {
            const t = fallbackTasks[j];
            if (t.projectId && p.id && String(t.projectId) !== String(p.id)) continue;
            const tStore = t.projectStore ? String(t.projectStore).trim().toLowerCase() : '';
            if (pStore && tStore && pStore !== tStore) continue;
            const tYear = t.projectYear !== undefined && t.projectYear !== null ? String(t.projectYear).trim() : '';
            if (pYear && tYear && pYear !== tYear) continue;
            matchedTaskMap.set(t.id, t);
          }
        }
      }
    }

    const projectTasks = Array.from(matchedTaskMap.values());

    const sortedProjTasks = [...projectTasks].sort((a, b) => {
      const codeA = a.taskCodePhu || '';
      const codeB = b.taskCodePhu || '';
      
      const numA = codeA.match(/#\s*(\d+)/)?.[1] || codeA.match(/[-_.:/\s]+(\d+)\s*$/)?.[1] || (/^\d+$/.test(codeA.trim()) ? codeA.trim() : null);
      const numB = codeB.match(/#\s*(\d+)/)?.[1] || codeB.match(/[-_.:/\s]+(\d+)\s*$/)?.[1] || (/^\d+$/.test(codeB.trim()) ? codeB.trim() : null);

      if (numA !== null && numB !== null) {
        const intA = parseInt(numA, 10);
        const intB = parseInt(numB, 10);
        if (!isNaN(intA) && !isNaN(intB) && intA !== intB) return intA - intB;
      } else if (numA !== null && numB === null) {
        return -1;
      } else if (numA === null && numB !== null) {
        return 1;
      }

      const valA = a.orderIndex !== undefined ? a.orderIndex : 0;
      const valB = b.orderIndex !== undefined ? b.orderIndex : 0;
      if (valA !== valB) return valA - valB;
      return String(a.id || '').localeCompare(String(b.id || ''));
    });

    const taskSummaries = sortedProjTasks.length > 0
      ? sortedProjTasks.filter(t => t.start && t.finish && t.name).map((t, idx) => ({
          id: t.id,
          name: t.name!,
          start: t.start!,
          finish: t.finish!,
          orderIndex: t.orderIndex !== undefined ? t.orderIndex : idx,
          duration: t.duration,
          parentTaskId: t.parentTaskId,
          taskCodePhu: t.taskCodePhu,
          runningStatus: t.runningStatus,
          approval: t.approval,
          party: t.party
        }))
      : (p.taskSummaries || []);

    let computedTakStart = p.computedTakStart || p.start;
    let computedTakCompl: string | undefined = p.computedTakCompl || p.handover || (p as any).takCompl || undefined;
    let computedOpening = p.computedOpening || (p.openingMode === 'manual' ? p.opening : '');

    if (projectTasks.length > 0) {
      let hasOpeningTask = false;
      for (let i = 0; i < projectTasks.length; i++) {
        const t = projectTasks[i];
        if (t.start && (!computedTakStart || t.start < computedTakStart)) {
          computedTakStart = t.start;
        }
        if (t.finish && (!computedTakCompl || t.finish > computedTakCompl)) {
          computedTakCompl = t.finish;
        }
        const name = t.name ? String(t.name).toLowerCase() : "";
        if ((name.includes("opening") || name.includes("open")) && t.finish) {
          if (!hasOpeningTask || t.finish > computedOpening) {
            computedOpening = t.finish;
            hasOpeningTask = true;
          }
        }
      }
      if (!hasOpeningTask && p.openingMode !== 'manual') {
        computedOpening = '';
      }
    } else if (taskSummaries.length > 0) {
      let hasOpeningTask = false;
      for (let i = 0; i < taskSummaries.length; i++) {
        const ts = taskSummaries[i];
        if (ts.start && !computedTakStart) {
          computedTakStart = ts.start;
        } else if (ts.start && computedTakStart && ts.start < computedTakStart) {
          computedTakStart = ts.start;
        }
        if (ts.finish && !computedTakCompl) {
          computedTakCompl = ts.finish;
        } else if (ts.finish && computedTakCompl && ts.finish > computedTakCompl) {
          computedTakCompl = ts.finish;
        }
        const name = ts.name ? String(ts.name).toLowerCase() : "";
        if ((name.includes("opening") || name.includes("open")) && ts.finish) {
          if (!hasOpeningTask || ts.finish > computedOpening) {
            computedOpening = ts.finish;
            hasOpeningTask = true;
          }
        }
      }
      if (!hasOpeningTask && p.openingMode !== 'manual') {
        computedOpening = '';
      }
    }

    const planProgressPercent = projectTasks.length > 0 
      ? computePlanProgressPercent(projectTasks)
      : (p.computedPlanProgress && p.computedPlanProgress > 0 
          ? p.computedPlanProgress 
          : (taskSummaries.length > 0 ? computePlanProgressFromSummaries(taskSummaries) : (p.computedPlanProgress ?? 0)));

    const actProgressPercent = projectTasks.length > 0
      ? computeActProgressPercent(projectTasks, delegationGroups)
      : (p.computedActProgress && p.computedActProgress > 0 
          ? p.computedActProgress 
          : (taskSummaries.length > 0 ? computeActProgressFromSummaries(taskSummaries) : (p.computedActProgress ?? 0)));

    const taskCount = projectTasks.length > 0
      ? Math.max(projectTasks.length, taskSummaries.length)
      : (Array.isArray(p.taskSummaries) ? p.taskSummaries.length : (p.taskCount ?? 0));

    const status = getStatus(computedTakStart, computedTakCompl, computedOpening || p.opening, p.code);
    const actStatus = computeProjectActStatus(
      { ...p, computedStatus: status, computedActProgress: actProgressPercent },
      delegationGroups,
      projectTasks
    );

    return {
      ...p,
      computedTakStart,
      computedTakCompl,
      computedOpening,
      computedPlanProgress: planProgressPercent,
      computedActProgress: actProgressPercent,
      computedStatus: status,
      computedActStatus: actStatus,
      taskSummaries,
      taskCount,
    };
  });

  const statusPriority: Record<string, number> = {
    'On process': 1,
    'Handover': 2,
    'Upcoming': 3,
    'Completed': 4,
    'Canceled': 5
  };

  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const todayTime = today.getTime();

  return enrichedProjects.sort((a, b) => {
    const statusA = a.computedStatus || 'Upcoming';
    const statusB = b.computedStatus || 'Upcoming';

    const priorityA = statusPriority[statusA] || 99;
    const priorityB = statusPriority[statusB] || 99;

    if (priorityA !== priorityB) {
      return priorityA - priorityB;
    }

    if (statusA === 'On process') {
      const finishA = parseDateTime(a.computedTakCompl || a.computedOpening || a.opening || '9999-12-31');
      const finishB = parseDateTime(b.computedTakCompl || b.computedOpening || b.opening || '9999-12-31');
      if (finishA && finishB) {
        const diffA = Math.abs(finishA.getTime() - todayTime);
        const diffB = Math.abs(finishB.getTime() - todayTime);
        if (diffA !== diffB) {
          return diffA - diffB;
        }
      }
    }

    const startA = a.computedTakStart || '9999-12-31';
    const startB = b.computedTakStart || '9999-12-31';

    if (startA < startB) return -1;
    if (startA > startB) return 1;

    return String(a.code || '').localeCompare(String(b.code || '')) || String(a.year || '').localeCompare(String(b.year || ''));
  });

  _lastProjectsRef = projects;
  _lastTasksRef = tasks;
  _lastDelegationRef = delegationGroups;
  _lastEnrichedResult = enrichedProjects;

  return enrichedProjects;
}

export function calculateActiveNotifications(
  projects: Project[],
  tasks: Task[],
  rules: NotificationRule[],
  delegationGroups: DelegationGroup[] = [],
  currentUserId?: string,
  currentUserUsername?: string,
  currentUserObj?: any,
): ActiveNotification[] {
  const now = new Date();
  const fieldLabels: Record<string, string> = {
    start: "Start",
    handover: "Pj Compl",
    opening: "Opening",
    finish: "Finish",
    flow_status: "Flow Status Update",
    mention: "@mention",
    delegation_party: "@delegation party",
  };

  const isMatchUserParty = (partyStr?: string) => {
    if (!partyStr) return false;
    const p = String(partyStr).trim().toLowerCase();
    if (currentUserUsername && p === currentUserUsername.trim().toLowerCase()) return true;
    if (currentUserObj?.name && p === currentUserObj.name.trim().toLowerCase()) return true;
    if (currentUserObj?.partyPermissions && currentUserObj.partyPermissions[partyStr.trim()]) return true;
    return false;
  };

  const notifications: ActiveNotification[] = [];

  const currentActualYear = now.getFullYear();

  // O(1) Pre-indexing for projects by code_year_store, code_store, and code (AGENTS.md Rule 7)
  const projectByKeyMap = new Map<string, Project>();
  const projectByCodeStoreMap = new Map<string, Project>();
  const projectByCodeMap = new Map<string, Project>();

  projects.forEach((p) => {
    const pCode = String(p.code || "").trim().toLowerCase();
    if (!pCode) return;
    const pStore = String(p.store || "HCM").trim().toUpperCase();
    const pYear = String(p.year || "").trim();

    if (!projectByCodeMap.has(pCode)) {
      projectByCodeMap.set(pCode, p);
    }
    const csKey = `${pCode}_${pStore}`;
    if (!projectByCodeStoreMap.has(csKey)) {
      projectByCodeStoreMap.set(csKey, p);
    }
    if (pYear) {
      projectByKeyMap.set(`${pCode}_${pYear}_${pStore}`, p);
    }
  });

  const findParentProject = (code?: string, year?: string | number, store?: string): Project | undefined => {
    const c = String(code || "").trim().toLowerCase();
    if (!c) return undefined;
    const s = String(store || "HCM").trim().toUpperCase();
    const y = String(year || "").trim();

    if (y && projectByKeyMap.has(`${c}_${y}_${s}`)) {
      return projectByKeyMap.get(`${c}_${y}_${s}`);
    }
    if (y && !store) {
      if (projectByKeyMap.has(`${c}_${y}_HCM`)) return projectByKeyMap.get(`${c}_${y}_HCM`);
      if (projectByKeyMap.has(`${c}_${y}_HN`)) return projectByKeyMap.get(`${c}_${y}_HN`);
    }
    if (!y && store) {
      if (projectByCodeStoreMap.has(`${c}_${s}`)) {
        return projectByCodeStoreMap.get(`${c}_${s}`);
      }
    }
    if (!y && !store) {
      return projectByCodeMap.get(c);
    }
    return undefined;
  };

  // Consolidate tasks: combining explicit `tasks` array with embedded `project.taskSummaries` (task summarize data)
  const taskMap = new Map<string, Task>();

  // 1. First extract embedded task summarize items from all projects
  projects.forEach((p) => {
    if (p.taskSummaries && Array.isArray(p.taskSummaries)) {
      p.taskSummaries.forEach((ts, idx) => {
        const pCode = p.code || "";
        const pStore = String(p.store || "HCM").toUpperCase().trim();
        const pYear = String(p.year || "").trim();
        const tsId = ts.id || `ts_${p.id || pCode}_${pStore}_${pYear}_${idx}`;

        taskMap.set(tsId, {
          id: tsId,
          name: ts.name || `Task ${idx + 1}`,
          start: ts.start || "",
          finish: ts.finish || "",
          projectCode: pCode,
          projectStore: pStore,
          projectYear: pYear,
          party: ts.party || p.party || "",
          runningStatus: ts.runningStatus || "On process",
          duration: ts.duration || "",
          parentTaskId: ts.parentTaskId,
          taskCodePhu: ts.taskCodePhu,
          comments: ts.comments || [],
          ...ts
        } as Task);
      });
    }
  });

  // 2. Overwrite or complement with detailed tasks from explicit `tasks` array
  tasks.forEach((t) => {
    if (t.id) {
      taskMap.set(t.id, t);
    }
  });

  const effectiveTasks = Array.from(taskMap.values());

  // Generate notifications for Cross-Year Rollover (only active/pending tasks, skip completed & canceled)
  effectiveTasks.forEach((task) => {
    const isCompleted = task.runningStatus === 'Completed';
    if (!isCompleted && task.projectYear && Number(task.projectYear) < currentActualYear) {
      const project = findParentProject(task.projectCode, task.projectYear, task.projectStore);
      if (project?.computedStatus === 'Canceled' || project?.mdStatus === 'Canceled') return;

      const subtitleStr = project
        ? `${task.projectCode} - ${project.name} - ${task.name}`
        : `${task.projectCode || ""} ${task.name}`.trim();

      const pStore = String(task.projectStore || project?.store || "HCM").toUpperCase().trim();
      const pYear = String(task.projectYear || project?.year || "").trim();

      const isMyTask = isMatchUserParty(task.party);
      notifications.push({
        id: `rollover_task_${task.id}`,
        title: "Cross-Year Task Rollover",
        subtitle: subtitleStr,
        message: `Task from ${task.projectYear} is still pending. Would you like to roll it over to ${currentActualYear}?`,
        dateStr: now.toISOString(),
        diffMs: 0,
        generatedAt: now.getTime(),
        targetType: "task",
        targetId: task.id,
        projectCode: task.projectCode,
        projectStore: pStore,
        projectYear: pYear,
        action: 'cross_year_rollover',
        actionData: { taskId: task.id, newYear: currentActualYear },
        priority: 'action_required',
        isAssignedToMe: isMyTask,
        party: task.party || "",
      });
    }
  });

  // Generate notifications for @mentions
  if (currentUserId) {
    effectiveTasks.forEach((task) => {
      const project = findParentProject(task.projectCode, task.projectYear, task.projectStore);
      if (task.comments && task.comments.length > 0) {
        task.comments.forEach((comment) => {
          if (
            comment.mentions &&
            comment.mentions.includes(currentUserId)
          ) {
            const commentTime = new Date(comment.createdAt).getTime();
            const diffMs = commentTime - now.getTime(); // it will be negative
            // Only show mentions from the last 30 days
            if (diffMs >= -30 * 24 * 60 * 60 * 1000) {
              const subtitleStr = project
                ? `${task.projectCode} - ${project.name} - ${task.name}`
                : `${task.projectCode || ""} ${task.name}`.trim();
              const pStore = String(task.projectStore || project?.store || "HCM").toUpperCase().trim();
              const pYear = String(task.projectYear || project?.year || "").trim();
              notifications.push({
                id: `mention_${comment.id}`,
                title: "You were mentioned in a task",
                subtitle: subtitleStr,
                message: `"${comment.text}"`,
                dateStr: comment.createdAt,
                diffMs,
                generatedAt: commentTime,
                targetType: "task",
                targetId: task.id,
                projectCode: task.projectCode,
                projectStore: pStore,
                projectYear: pYear,
                priority: 'action_required',
                isAssignedToMe: true,
                party: task.party || "",
              });
            }
          }
        });
      }
    });

    projects.forEach((project) => {
      if (project.comments && project.comments.length > 0) {
        project.comments.forEach((comment) => {
          if (
            comment.mentions &&
            comment.mentions.includes(currentUserId)
          ) {
            const commentTime = new Date(comment.createdAt).getTime();
            const diffMs = commentTime - now.getTime(); // it will be negative
            // Only show mentions from the last 30 days
            if (diffMs >= -30 * 24 * 60 * 60 * 1000) {
              const subtitleStr = `${project.code} - ${project.name}`;
              notifications.push({
                id: `mention_proj_${comment.id}`,
                title: "You were mentioned in a project",
                subtitle: subtitleStr,
                message: `"${comment.text}"`,
                dateStr: comment.createdAt,
                diffMs,
                generatedAt: commentTime,
                targetType: "project",
                targetId: project.id || project.no,
                projectCode: project.code,
                projectStore: String(project.store || "HCM").toUpperCase().trim(),
                projectYear: String(project.year || "").trim(),
                priority: 'action_required',
                isAssignedToMe: true,
              });
            }
          }
        });
      }
    });
  }

  rules.forEach((rule) => {
    if (rule.target === "task_delegation") {
      delegationGroups.forEach((group) => {
        if (!group.updatedAt) return;
        const targetDate = new Date(group.updatedAt);
        if (isNaN(targetDate.getTime())) return;

        let shouldTrigger = false;
        let flowStatus = "";
        let party = "";
        let taskId = group.parentTaskId || group.id;

        let subTaskName = "";
        if (rule.field === "flow_status") {
          // Check subTasks
          if (group.subTasks && group.subTasks.length > 0) {
            let recentSt = group.subTasks[0];
            group.subTasks.forEach((st) => {
              if (
                st.updatedAt &&
                recentSt.updatedAt &&
                new Date(st.updatedAt).getTime() >
                  new Date(recentSt.updatedAt).getTime()
              ) {
                recentSt = st;
              } else if (st.updatedAt && !recentSt.updatedAt) {
                recentSt = st;
              }
            });
            party = recentSt.party || "N/A";
            subTaskName = recentSt.name || "";
            flowStatus =
              recentSt.delegationStatus || recentSt.status || "Assigned";
            shouldTrigger = true;
          }
        } else if (rule.field === "mention" && currentUserUsername) {
          if (group.subTasks && group.subTasks.length > 0) {
            let recentSt = group.subTasks[0];
            group.subTasks.forEach((st) => {
              if (
                st.updatedAt &&
                recentSt.updatedAt &&
                new Date(st.updatedAt).getTime() >
                  new Date(recentSt.updatedAt).getTime()
              ) {
                recentSt = st;
              }
            });

            if (
              recentSt.comments &&
              recentSt.comments.includes(`@${currentUserUsername}`)
            ) {
              party = recentSt.party || "N/A";
              subTaskName = recentSt.name || "";
              flowStatus =
                recentSt.delegationStatus || recentSt.status || "Assigned";
              shouldTrigger = true;
            }
          }
        } else if (rule.field === "delegation_party") {
          if (group.subTasks && group.subTasks.length > 0) {
            let recentSt = group.subTasks[0];
            group.subTasks.forEach((st) => {
              if (
                st.updatedAt &&
                recentSt.updatedAt &&
                new Date(st.updatedAt).getTime() >
                  new Date(recentSt.updatedAt).getTime()
              ) {
                recentSt = st;
              }
            });

            if (
              recentSt.comments &&
              (recentSt.comments.includes("@delegation party") ||
                recentSt.comments.includes("@Delegation party") ||
                (recentSt.party &&
                  recentSt.comments.includes(`@${recentSt.party}`)))
            ) {
              party = recentSt.party || "N/A";
              subTaskName = recentSt.name || "";
              flowStatus =
                recentSt.delegationStatus || recentSt.status || "Assigned";
              shouldTrigger = true;
            }
          }
        }

        if (shouldTrigger) {
          const diffMs = targetDate.getTime() - now.getTime();
          const thresholdMs =
            ((rule.daysBefore || 0) * 24 + (rule.hoursBefore || 0)) *
            60 *
            60 *
            1000;

          if (diffMs <= thresholdMs && diffMs >= -30 * 24 * 60 * 60 * 1000) {
            let projectName = group.name || "Custom Task";
            const isLinked = !!group.parentTaskId || !!group.projectCode;

            let parentTask: Task | null = null;
            let parentProject: Project | undefined = undefined;
            let ps = group.projectStore;
            let py = group.projectYear;

            if (isLinked) {
              parentTask = group.parentTaskId
                ? tasks.find((t) => t.id === group.parentTaskId) || null
                : null;

              if (!ps && parentTask?.projectStore) ps = parentTask.projectStore;
              if (!py && parentTask?.projectYear) py = parentTask.projectYear;

              parentProject = projects.find(
                (p) =>
                  ((group.parentTaskId && p.code === group.parentTaskId) ||
                    (group.projectCode && p.code === group.projectCode) ||
                    (parentTask && p.code === parentTask.projectCode)) &&
                  (ps ? String(p.store) === String(ps) : true) &&
                  (py ? String(p.year) === String(py) : true),
              );

              if (parentTask) {
                projectName = `${parentTask.name}${parentProject?.name ? ` - ${parentProject.name}` : ""}`;
              } else if (parentProject) {
                projectName = parentProject.name || group.name;
              }
            }

            let displayTaskName = subTaskName || group.name || "task";
            const isMyDelegation = isMatchUserParty(party) || (rule.field === 'mention') || (rule.field === 'delegation_party');
            const isRejected = String(flowStatus).toLowerCase() === 'rejected';
            const delPriority: NotificationPriority = isRejected ? 'critical' : 'action_required';

            notifications.push({
              id: `${rule.id}_${ps || 'store'}_${group.id}_${targetDate.getTime()}`,
              title: projectName,
              subtitle: `Party: ${party} | Status: ${flowStatus}`,
              message: `Delegation task: ${displayTaskName}, status: ${flowStatus.toLowerCase()}.`,
              dateStr: formatDisplayDateTime(group.updatedAt),
              diffMs: diffMs,
              generatedAt: targetDate.getTime() - thresholdMs,
              targetType: "task_delegation",
              targetId: group.id,
              projectCode: group.projectCode || parentTask?.projectCode || parentProject?.code || "",
              projectStore: ps || parentProject?.store || "",
              projectYear: py || parentProject?.year || "",
              priority: delPriority,
              isAssignedToMe: isMyDelegation,
              party: party,
            });
          }
        }
      });
      return; // Skip normal project/task processing for task_delegation
    }

    const targetList = rule.target === "project" ? projects : effectiveTasks;

    targetList.forEach((item) => {
      // For projects, we use computed fields if the rule is for 'start' or 'handover'
      let dateVal = (item as any)[rule.field];
      if (rule.target === "project") {
        const p = item as Project;
        if (rule.field === "start") dateVal = p.computedTakStart || p.start;
        if (rule.field === "handover") dateVal = p.computedTakCompl; // Chỉ lấy từ task con
        if (rule.field === "opening") dateVal = p.computedOpening || p.opening;
      }

      if (dateVal) {
        const targetDate = parseDateTime(dateVal);
        if (targetDate && !isNaN(targetDate.getTime())) {
          const diffMs = targetDate.getTime() - now.getTime();
          const thresholdMs =
            ((rule.daysBefore || 0) * 24 + (rule.hoursBefore || 0)) *
            60 *
            60 *
            1000;

          if (diffMs <= thresholdMs && diffMs >= -30 * 24 * 60 * 60 * 1000) {
            let title = "";
            let subtitle = "";
            let notificationBody = "";

            const isOverdue = diffMs < 0;
            const diffHrsTotal = Math.abs(
              Math.floor(diffMs / (1000 * 60 * 60)),
            );
            const d = Math.floor(diffHrsTotal / 24);
            const h = diffHrsTotal % 24;

            let timeMsg = "";
            if (d === 0 && h === 0)
              timeMsg = isOverdue
                ? "Overdue (under 1 hour)"
                : "Due soon (under 1 hour)";
            else
              timeMsg = isOverdue
                ? `overdue by ${d > 0 ? d + " days " : ""}${h > 0 ? h + " hours" : ""}`
                : `in ${d > 0 ? d + " days " : ""}${h > 0 ? h + " hours" : ""}`;

            const fName = fieldLabels[rule.field] || rule.field;

            let notifId = `${rule.id}_${(item as any).id || (item as any).no}`;
            let itemProjectCode = (item as any).code || (item as any).projectCode || "";
            let itemProjectStore = (item as any).store || (item as any).projectStore || "";
            let itemProjectYear = (item as any).year || (item as any).projectYear || "";

            if (rule.target === "project") {
              const p = item as Project;
              title = `${p.code} - ${p.name || "N/A"}`;
              subtitle = p.name || "N/A";
              notificationBody = `${fName} deadline ${timeMsg}`;
              itemProjectCode = p.code || "";
              itemProjectStore = String(p.store || "HCM").toUpperCase().trim();
              itemProjectYear = String(p.year || "").trim();
              notifId = `${rule.id}_${p.id || p.no}`;
            } else {
              const t = item as Task;
              let projName = "N/A";
              const p = findParentProject(t.projectCode, t.projectYear, t.projectStore);

              if (p && p.name) projName = p.name;

              title = `${t.projectCode} - ${projName} - ${t.name || "N/A"}`;
              subtitle = `Project ${t.projectCode} | Party: ${t.party || "N/A"}`;
              const pParty = t.party ? ` by ${t.party}` : "";
              notificationBody = `${fName} deadline ${timeMsg}${pParty}`;

              itemProjectCode = t.projectCode || "";
              itemProjectStore = String(t.projectStore || p?.store || "HCM").toUpperCase().trim();
              itemProjectYear = String(t.projectYear || p?.year || "").trim();
              notifId = `${rule.id}_${t.id}`;
            }

            const isDueSoon = !isOverdue && diffMs <= 48 * 60 * 60 * 1000;
            const itemPriority: NotificationPriority = isOverdue ? 'critical' : isDueSoon ? 'urgent' : 'info';
            const isMyItem = rule.target === 'task' ? isMatchUserParty((item as Task).party) : false;
            const taskParty = rule.target === 'task' ? ((item as Task).party || '') : '';

            notifications.push({
              id: notifId,
              title: title,
              subtitle: subtitle,
              message: notificationBody,
              dateStr: formatDisplayDateTime(dateVal),
              diffMs: diffMs,
              generatedAt: targetDate.getTime() - thresholdMs,
              targetType: rule.target as any,
              targetId:
                rule.target === "project"
                  ? ((item as Project).id || (item as Project).no)
                  : (item as Task).id,
              projectCode: itemProjectCode,
              projectStore: itemProjectStore,
              projectYear: itemProjectYear,
              priority: itemPriority,
              isAssignedToMe: isMyItem,
              party: taskParty,
            });
          }
        }
      }
    });
  });

  // Smart Grouping / Batching: Group task deadline notifications by project to avoid notification fatigue
  const batchedNotifications: ActiveNotification[] = [];
  const taskNotifsByProject = new Map<string, ActiveNotification[]>();
  const nonBatchedNotifs: ActiveNotification[] = [];

  notifications.forEach((notif) => {
    if (notif.targetType === "task" && !notif.id.startsWith("mention_")) {
      const pCode = notif.projectCode || (notif.title.split(" - ")[0] || "").trim();
      const pStore = notif.projectStore || "";
      const pYear = notif.projectYear || "";
      const groupKey = `${pCode}_${pStore}_${pYear}`;
      if (!taskNotifsByProject.has(groupKey)) {
        taskNotifsByProject.set(groupKey, []);
      }
      taskNotifsByProject.get(groupKey)!.push(notif);
    } else {
      nonBatchedNotifs.push(notif);
    }
  });

  taskNotifsByProject.forEach((items, groupKey) => {
    if (items.length > 1) {
      const first = items[0];
      const pCode = first.projectCode || groupKey.split("_")[0];
      const pStore = first.projectStore || groupKey.split("_")[1] || "";
      const pYear = first.projectYear || groupKey.split("_")[2] || "";
      const project = projects.find(
        (p) =>
          p.code === pCode &&
          (pStore ? String(p.store || "").toUpperCase() === String(pStore).toUpperCase() : true) &&
          (pYear ? String(p.year || "") === String(pYear) : true),
      );
      const projName = project?.name || (first.title.split(" - ")[1] || "").trim() || "Project";

      const urgentDiff = Math.min(...items.map((i) => i.diffMs));
      const latestGen = Math.max(...items.map((i) => i.generatedAt));
      const earliestDateStr = items[0].dateStr;

      const summaryList = items.map((i) => {
        const parts = i.title.split(" - ");
        const taskPart = parts.length >= 3 ? parts.slice(2).join(" - ") : i.title;
        return `• <strong>${taskPart}</strong>: ${i.message}`;
      });

      const messagePreview =
        `Batched <strong>${items.length} tasks</strong> requiring attention:<br/>` +
        summaryList.slice(0, 3).join("<br/>") +
        (items.length > 3 ? `<br/><em>...and ${items.length - 3} more tasks</em>` : "");

      const hasCritical = items.some((i) => i.priority === 'critical');
      const hasUrgent = items.some((i) => i.priority === 'urgent');
      const batchPriority: NotificationPriority = hasCritical ? 'critical' : hasUrgent ? 'urgent' : 'info';
      const hasAssignedToMe = items.some((i) => i.isAssignedToMe);

      batchedNotifications.push({
        id: `batch_task_${groupKey}_${items.length}`,
        title: `${pCode} - ${projName}`,
        subtitle: `Store: ${pStore || "ALL"} • Year: ${pYear || "ALL"} | ${items.length} tasks`,
        message: messagePreview,
        dateStr: earliestDateStr,
        diffMs: urgentDiff,
        generatedAt: latestGen,
        targetType: "project",
        targetId: project ? (project.id || project.no) : pCode,
        count: items.length,
        projectCode: pCode,
        projectStore: pStore,
        projectYear: pYear,
        priority: batchPriority,
        isAssignedToMe: hasAssignedToMe,
      });
    } else {
      batchedNotifications.push(items[0]);
    }
  });

  const thirtyDaysMs = 30 * 24 * 60 * 60 * 1000;
  return [...nonBatchedNotifs, ...batchedNotifications]
    .filter((n) => now.getTime() - n.generatedAt <= thirtyDaysMs)
    .sort((a, b) => b.generatedAt - a.generatedAt);
}

export function parseCSVFull(text: string) {
  let result: string[][] = [];
  let row: string[] = [];
  let curVal = "";
  let inQuotes = false;

  if (text.charCodeAt(0) === 0xfeff) text = text.slice(1);

  let firstLine = text.split("\n")[0] || "";
  let sep = ",";

  const commaCount = (firstLine.match(/,/g) || []).length;
  const semicolonCount = (firstLine.match(/;/g) || []).length;
  const tabCount = (firstLine.match(/\t/g) || []).length;

  if (tabCount > commaCount && tabCount > semicolonCount) {
    sep = "\t";
  } else if (semicolonCount > commaCount) {
    sep = ";";
  }

  for (let i = 0; i < text.length; i++) {
    let char = text[i];
    if (inQuotes) {
      if (char === '"') {
        if (i + 1 < text.length && text[i + 1] === '"') {
          curVal += '"';
          i++;
        } else {
          inQuotes = false;
        }
      } else {
        curVal += char;
      }
    } else {
      if (char === '"') {
        inQuotes = true;
      } else if (char === sep) {
        row.push(curVal.trim());
        curVal = "";
      } else if (char === "\n" || (char === "\r" && text[i + 1] === "\n")) {
        if (char === "\r") i++;
        row.push(curVal.trim());
        if (row.some((cell) => cell !== "")) result.push(row);
        row = [];
        curVal = "";
      } else {
        curVal += char;
      }
    }
  }
  if (curVal !== "" || row.length > 0) {
    row.push(curVal.trim());
    if (row.some((cell) => cell !== "")) result.push(row);
  }
  return { rows: result, separator: sep };
}

export function computePlanProgressPercent(groupTasks: Task[]): number {
  if (groupTasks.length === 0) return 0;

  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const todayTime = today.getTime();

  let totalScore = 0;

  for (const t of groupTasks) {
    if (!t.start && !t.finish) continue;
    
    const start = parseDateTime(t.start);
    if (start) start.setHours(0, 0, 0, 0);
    const finish = parseDateTime(t.finish);
    if (finish) finish.setHours(0, 0, 0, 0);

    if (finish && todayTime > finish.getTime()) {
      totalScore += 1;
    } else if (start && finish && todayTime >= start.getTime() && todayTime <= finish.getTime()) {
      const totalDuration = finish.getTime() - start.getTime();
      if (totalDuration > 0) {
        const passed = todayTime - start.getTime();
        totalScore += passed / totalDuration;
      } else {
        totalScore += 1; // Start and finish on the same day
      }
    }
  }
  
  return Math.round((totalScore / groupTasks.length) * 100);
}

/**
 * Trả về tỷ lệ hoàn thành (0 - 1.0) của một subTask ủy quyền dựa trên trạng thái
 * - Completed / Approved: 1.0 (100%)
 * - Pending completed / ReqApproval / Waiting for approval / Requested / Hold: 0.8 (80%)
 * - Accepted / On process / In progress / Doing / AcceptTask: 0.5 (50%)
 * - Needs Rework / Rejected / ReqReview: 0.3 (30%)
 * - Assigned / Upcoming / None: 0.0 (0%)
 */
export function getSubTaskProgressScore(st: {
  delegationStatus?: string;
  status?: string;
  approvalStatus?: string;
  [key: string]: any;
}): number {
  const status = (st.delegationStatus || st.status || "").toLowerCase();
  const approval = (st.approvalStatus || "").toLowerCase();

  if (
    ["completed", "approved", "finish", "finished"].includes(status) ||
    approval === "approved"
  ) {
    return 1.0;
  }
  if (
    [
      "pending completed",
      "waiting for approval",
      "reqapproval",
      "requested",
      "hold",
    ].includes(status) ||
    approval === "reqapproval" ||
    approval === "requested"
  ) {
    return 0.8;
  }
  if (
    [
      "accepted",
      "on process",
      "in progress",
      "inprogress",
      "doing",
      "accepttask",
    ].includes(status) ||
    approval === "accepttask"
  ) {
    return 0.5;
  }
  if (
    ["needs rework", "rejected", "reqreview"].includes(status) ||
    approval === "rejected" ||
    approval === "reqreview"
  ) {
    return 0.3;
  }
  return 0.0;
}

/**
 * Trả về tỷ lệ phần trăm (0 - 100%) của một nhãn trạng thái thực tế (actStatus)
 */
export function getActStatusPercent(status: string | undefined | null): number {
  if (!status) return 0;
  const s = String(status).trim().toLowerCase();

  if (["completed", "approved", "finish", "finished"].includes(s)) {
    return 100;
  }
  if (
    [
      "pending approval",
      "pending completed",
      "review late",
      "waiting for approval",
      "reqapproval",
      "requested",
      "hold",
    ].includes(s)
  ) {
    return 80;
  }
  if (
    [
      "accepted",
      "in progress",
      "on process",
      "inprogress",
      "doing",
      "accepttask",
      "delayed",
    ].includes(s)
  ) {
    return 50;
  }
  if (
    ["correction needed", "needs rework", "rejected", "reqreview"].includes(s)
  ) {
    return 30;
  }
  return 0;
}

/**
 * Tính toán tiến độ % thực tế của một nhóm ủy quyền dựa trên trung bình các subtask
 */
export function computeDelegationGroupProgressPercent(group: DelegationGroup): number {
  if (!group || !group.subTasks || group.subTasks.length === 0) return 0;
  
  const rawScores = group.subTasks.map((st) => getSubTaskProgressScore(st));
  let maxCompletedIdx = -1;
  for (let i = 0; i < rawScores.length; i++) {
    if (rawScores[i] >= 1.0) {
      maxCompletedIdx = i;
    }
  }

  let totalScore = 0;
  for (let i = 0; i < rawScores.length; i++) {
    if (maxCompletedIdx !== -1 && i <= maxCompletedIdx) {
      totalScore += 1.0;
    } else {
      totalScore += rawScores[i];
    }
  }

  return Math.round((totalScore / group.subTasks.length) * 100);
}

export function computeActProgressPercent(
  groupTasks: Task[],
  delegationGroups: DelegationGroup[] | Map<string, DelegationGroup>,
): number {
  if (groupTasks.length === 0) return 0;

  const getGroup = (tid: string | undefined) => {
    if (!tid) return undefined;
    if (delegationGroups instanceof Map) return delegationGroups.get(tid);
    return delegationGroups.find((g) => g.parentTaskId === tid);
  };

  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const todayTime = today.getTime();

  // Find max completed timestamp across explicitly completed tasks (direct or subtasks)
  const completedTaskStartTimestamps: number[] = [];

  for (const t of groupTasks) {
    const delegationGroup = getGroup(t.id);
    const subTasks = delegationGroup?.subTasks || [];
    let isCompleted = false;

    if (subTasks.length > 0) {
      isCompleted = subTasks.every((st) => {
        const s = (st.delegationStatus || st.status || "").toLowerCase();
        const a = (st.approvalStatus || "").toLowerCase();
        return (
          ["completed", "approved", "finish", "finished"].includes(s) || a === "approved"
        );
      });
    } else {
      const running = (t.runningStatus || "").toLowerCase();
      isCompleted = running.includes("complete") || t.approval === "Approved";
    }

    if (isCompleted) {
      const startTs = parseDateTime(t.start)?.getTime() || 0;
      if (startTs > 0) completedTaskStartTimestamps.push(startTs);
    }
  }

  const maxCompletedStartTs =
    completedTaskStartTimestamps.length > 0
      ? Math.max(...completedTaskStartTimestamps)
      : 0;

  const rawScores: number[] = [];

  for (const t of groupTasks) {
    const delegationGroup = getGroup(t.id);
    const subTasks = delegationGroup?.subTasks || [];

    // Case 1: Task has delegation subtasks
    if (subTasks.length > 0) {
      const groupPct = computeDelegationGroupProgressPercent(delegationGroup);
      rawScores.push(groupPct / 100);
      continue;
    }

    // Case 2: Pure task without delegation subtasks
    const running = (t.runningStatus || "").toLowerCase();
    const approval = (t.approval || "").toLowerCase();

    if (running.includes("complete") || approval === "approved") {
      rawScores.push(1.0);
      continue;
    }

    // Waterfall predecessor completion: if an end-stream task completed, earlier task is completed
    const tFinishTs = parseDateTime(t.finish)?.getTime() || 0;
    if (
      tFinishTs > 0 &&
      maxCompletedStartTs > 0 &&
      tFinishTs <= maxCompletedStartTs
    ) {
      rawScores.push(1.0);
      continue;
    }

    if (running.includes("process") || running.includes("progress")) {
      const start = parseDateTime(t.start);
      const finish = parseDateTime(t.finish);
      if (
        start &&
        finish &&
        todayTime >= start.getTime() &&
        todayTime <= finish.getTime()
      ) {
        const duration = finish.getTime() - start.getTime();
        const passed = todayTime - start.getTime();
        rawScores.push(
          duration > 0 ? Math.max(0.3, Math.min(0.9, passed / duration)) : 0.5
        );
      } else {
        rawScores.push(0.5);
      }
      continue;
    }

    if (running.includes("delay")) {
      rawScores.push(0.5);
      continue;
    }

    if (approval === "pending" || approval === "requested") {
      rawScores.push(0.8);
      continue;
    }

    // Fallback to planned timeline
    const plStatus = getTaskStatus(t.start, t.finish);
    if (plStatus === "Completed") {
      rawScores.push(1.0);
    } else if (plStatus === "On process") {
      const start = parseDateTime(t.start);
      const finish = parseDateTime(t.finish);
      if (
        start &&
        finish &&
        todayTime >= start.getTime() &&
        todayTime <= finish.getTime()
      ) {
        const duration = finish.getTime() - start.getTime();
        const passed = todayTime - start.getTime();
        rawScores.push(
          duration > 0 ? Math.max(0.3, Math.min(0.9, passed / duration)) : 0.5
        );
      } else {
        rawScores.push(0.5);
      }
    } else {
      rawScores.push(0.0);
    }
  }

  // Find max index of a task with 100% score
  let maxCompletedTaskIdx = -1;
  for (let i = 0; i < rawScores.length; i++) {
    if (rawScores[i] >= 1.0) {
      maxCompletedTaskIdx = i;
    }
  }

  let totalScore = 0;
  for (let i = 0; i < rawScores.length; i++) {
    if (maxCompletedTaskIdx !== -1 && i <= maxCompletedTaskIdx) {
      totalScore += 1.0;
    } else {
      totalScore += rawScores[i];
    }
  }

  return Math.round((totalScore / groupTasks.length) * 100);
}

export function scrollToElement(el: HTMLElement) {
  if (!el) return;
  const isIOS =
    /iPad|iPhone|iPod/.test(navigator.userAgent) ||
    (typeof navigator !== "undefined" &&
      navigator.userAgent.includes("Mac") &&
      "ontouchend" in document);

  if (isIOS) {
    try {
      // Manual scroll calculation for iOS to bypass smooth/overflow bugs
      const rect = el.getBoundingClientRect();
      const st = window.pageYOffset || document.documentElement.scrollTop;

      let scrollParent = el.parentElement;
      let handledInnerScroll = false;

      while (
        scrollParent &&
        scrollParent !== document.body &&
        scrollParent !== document.documentElement
      ) {
        const style = window.getComputedStyle(scrollParent);
        if (
          (style.overflowY === "auto" ||
            style.overflowY === "scroll" ||
            style.overflow === "auto" ||
            style.overflow === "scroll") &&
          scrollParent.scrollHeight > scrollParent.clientHeight + 5
        ) {
          const parentRect = scrollParent.getBoundingClientRect();
          const childTargetY =
            rect.top -
            parentRect.top +
            scrollParent.scrollTop -
            scrollParent.clientHeight / 2 +
            rect.height / 2;
          scrollParent.scrollTop = childTargetY;
          handledInnerScroll = true;
          break;
        }
        scrollParent = scrollParent.parentElement;
      }

      if (!handledInnerScroll) {
        // Fallback to window scroll
        const targetY =
          rect.top + st - window.innerHeight / 2 + rect.height / 2;
        window.scrollTo(0, targetY);
      }
      return;
    } catch (e) {
      el.scrollIntoView(true);
      return;
    }
  }

  try {
    el.scrollIntoView({ behavior: "smooth", block: "center" });
  } catch (e) {
    el.scrollIntoView(true);
  }
}

export function getUserDelegationRoleForTask(task: import('../types').Task, currentUser: import('../types').AppUser, delegationGroups: import('../types').DelegationGroup[]): 'assignee' | 'creator' | null {
  if (!currentUser) return null;
  const linkedGroup = delegationGroups.find((g) => g.parentTaskId === task.id);
  if (!linkedGroup) return null;

  const userParties = currentUser.partyPermissions 
    ? Object.keys(currentUser.partyPermissions).filter(k => currentUser.partyPermissions![k]).map(k => k.toLowerCase().trim()) 
    : [];
  
  let isAssignee = false;
  if (userParties.includes('all')) {
    isAssignee = true;
  } else {
    const currentUsername = String(currentUser.username || '').toLowerCase().trim();
    const currentUserId = currentUser.id;

    for (const st of linkedGroup.subTasks || []) {
      const pPartyList = String(st.party || '').toLowerCase().split(',').map(s => s.trim()).filter(Boolean);
      const isPIC = pPartyList.some(p => userParties.includes(p)) || 
                    pPartyList.includes(currentUsername) || 
                    (st.involvedUsers && st.involvedUsers.includes(currentUserId));
      if (isPIC) {
        isAssignee = true;
        break;
      }
    }
  }

  if (isAssignee) return 'assignee';
  if (linkedGroup.createdBy === currentUser.id) return 'creator';
  
  return null;
}
export function formatCurrencyValue(val: number | undefined, currency: 'VND' | 'JPY' | 'USD', jpyRate: number, usdRate: number): string {
  if (val === null || val === undefined) return "-";
  let num = Number(val);
  if (currency === 'JPY') {
    num = (num / 10) / jpyRate;
  } else if (currency === 'USD') {
    num = (num * 1000) / usdRate;
  }
  return Math.round(num).toLocaleString("en-US");
}

              (hasLocalYearTasks && isClosedOrPastYear) ||
              (DELTA_FIRST.tasks && hasMarker(markers, 'tasks_year', year));
            if (deltaManaged) {
              tasksPool.current.set(yearKey, () => {});
              setIsTasksLoading(false);
              needTasksDeltaSync = true;
              usePerformanceStore
                .getState()
                .trackGetDoc(
                  `taka_tasks_${yearKey}`,
                  true,
                  currentTasks.length,
                  "Dexie Local-First Tasks (0 Read full collection)"
                );
            } else {
              const colRef = collection(db, "artifacts", APP_ID, "public", "data", "taka_tasks");
              const qTasks = buildStoreAndYearQuery(
                colRef,
                "projectStore",
                "projectYear",
                year,
                userRole,
                allowedStores
              );

              usePerformanceStore.getState().registerListener(`taka_tasks_${yearKey}`);
              const unsub = onSnapshot(
                qTasks,
                (snapshot) => {
                  usePerformanceStore.getState().trackSnapshot(`taka_tasks_${yearKey}`, snapshot);
                  setIsTasksLoading(false);
                  batchTasksUpdate(snapshot.docChanges(), snapshot.metadata.fromCache);
                  setCloudStatus("online");

                  if (DELTA_FIRST.tasks && !snapshot.metadata.fromCache) {
                    markLoaded('tasks_year', year).catch(() => {});
                    const u = tasksPool.current.get(yearKey);
                    tasksPool.current.set(yearKey, () => {});
                    u?.();
                    usePerformanceStore.getState().unregisterListener(`taka_tasks_${yearKey}`);
                  }
                },
                (e) => {
                  setIsTasksLoading(false);
                  console.error("Tasks consolidated listener error:", e);
                  setCloudStatus("offline");
                }
              );
              tasksPool.current.set(yearKey, unsub);
            }
          }
        } else {
          const visibleSet = new Set(
            (visibleActiveProjectIdsRef.current || []).map((id) => String(id))
          );
          const sortedProjs = [...projs].sort((a, b) => {
            const aVis = visibleSet.has(a.id) || visibleSet.has(a.code) ? 1 : 0;
            const bVis = visibleSet.has(b.id) || visibleSet.has(b.code) ? 1 : 0;
            return bVis - aVis;
          });

          sortedProjs.forEach((p) => {
            const projKey = `proj_${p.id}`;
            const yearKey = `year_${year}`;

            if (tasksPool.current.has(yearKey)) {
              neededKeys.add(yearKey);
              return;
            }

            neededKeys.add(projKey);

            if (tasksTTLTimeouts.current.has(projKey)) {
              clearTimeout(tasksTTLTimeouts.current.get(projKey)!);
              tasksTTLTimeouts.current.delete(projKey);
            }

            if (!tasksPool.current.has(projKey)) {
              const currentTasks = useAppStore.getState().tasks || [];
              // So khớp theo đúng dự án (projectId, hoặc đủ store + year + code), không chỉ theo mã
              const belongsHere = (t: Task): boolean => {
                if (t.projectId) return String(t.projectId) === String(p.id);
                return (
                  String(t.projectCode || "") === String(p.code) &&
                  String(t.projectStore || "").trim().toLowerCase() === String(p.store || "").trim().toLowerCase() &&
                  String(t.projectYear ?? "").trim() === String(p.year ?? "").trim()
                );
              };
              const hasLocalProjTasks =
                currentTasks.some(belongsHere) ||
                knownHydratedProjectCodes.current.has(String(p.id));
              const isClosedOrPastYear = Number(year) < new Date().getFullYear();

              const deltaManaged =
                (hasLocalProjTasks && isClosedOrPastYear) ||
                (DELTA_FIRST.tasks && (hasMarker(markers, 'tasks_proj', p.code) || hasMarker(markers, 'tasks_year', year)));
              if (deltaManaged) {
                tasksPool.current.set(projKey, () => {});
                setIsTasksLoading(false);
                needTasksDeltaSync = true;
                usePerformanceStore
                  .getState()
                  .trackGetDoc(
                    `taka_tasks_${projKey}`,
                    true,
                    1,
                    "Dexie Local-First Project Tasks (0 Read)"
                  );
              } else {
                const qTasks = query(
                  collection(db, "artifacts", APP_ID, "public", "data", "taka_tasks"),
                  where("projectCode", "==", p.code)
                );

                usePerformanceStore.getState().registerListener(`taka_tasks_${projKey}`);
                const unsub = onSnapshot(
                  qTasks,
                  (snapshot) => {
                    knownHydratedProjectCodes.current.add(String(p.id));
                    usePerformanceStore.getState().trackSnapshot(`taka_tasks_${projKey}`, snapshot);
                    setIsTasksLoading(false);

                    // Truy vấn theo projectCode trả về cả task của store/năm khác có cùng mã: chỉ nhận task đúng dự án.
                    // Task cũ thiếu projectStore/projectYear vẫn được giữ (khớp lỏng như getTasksForProject).
                    const compatibleChanges = snapshot.docChanges().filter((ch) => {
                      const dt: any = ch.doc.data();
                      if (ch.type === "removed") {
                        if (dt && dt.projectId && String(dt.projectId) !== String(p.id)) return false;
                        if (dt && dt.projectStore && p.store &&
                            String(dt.projectStore).trim().toLowerCase() !== String(p.store).trim().toLowerCase()) return false;
                        return true;
                      }
                      if (dt.projectId && String(dt.projectId) !== String(p.id)) return false;
                      if (dt.projectStore && p.store &&
                          String(dt.projectStore).trim().toLowerCase() !== String(p.store).trim().toLowerCase()) return false;
                      const dy = dt.projectYear;
                      if (dy !== undefined && dy !== null && dy !== "" && p.year &&
                          String(dy).trim() !== String(p.year).trim()) return false;
                      return true;
                    });
                    batchTasksUpdate(compatibleChanges, snapshot.metadata.fromCache);
                    setCloudStatus("online");

                    if (DELTA_FIRST.tasks && !snapshot.metadata.fromCache) {
                      markLoaded('tasks_proj', p.code).catch(() => {});
                      const u = tasksPool.current.get(projKey);
                      tasksPool.current.set(projKey, () => {});
                      u?.();
                      usePerformanceStore.getState().unregisterListener(`taka_tasks_${projKey}`);
                    }
                  },
                  (e) => {
                    setIsTasksLoading(false);
                    console.error("Tasks listener error:", e);
                    setCloudStatus("offline");
                  }
                );
                tasksPool.current.set(projKey, unsub);
              }
            }
          });
        }
      });

      if (needTasksDeltaSync) {
        pullAndApplyDeltas()
          .then(() => setCloudStatus("online"))
          .catch(() => {});
      }

      Array.from(tasksPool.current.keys()).forEach((key) => {
        if (!neededKeys.has(key)) {
          if (!tasksTTLTimeouts.current.has(key)) {
            const timeoutId = setTimeout(() => {
              const unsub = tasksPool.current.get(key);
              if (unsub) unsub();
              tasksPool.current.delete(key);
              tasksTTLTimeouts.current.delete(key);
              usePerformanceStore.getState().unregisterListener(`taka_tasks_${key}`);
            }, TTL_DURATION_MS);
            tasksTTLTimeouts.current.set(key, timeoutId);
          }
        }
      });
    };

    runTasksSync().catch((err) => {
      console.warn("[useTasksHydration] runTasksSync error:", err);
    });

    return () => {
      isCancelled = true;
    };
  }, [
    querySignature,
    userRole,
    allowedStores,
    isTasksRequested,
    expandedProjectIds,
    searchActiveProjectIds,
    globalYearFilter,
    batchTasksUpdate,
    setIsTasksLoading,
    setCloudStatus,
  ]);
}

// Chuyển updatedAt (ISO string / Firestore Timestamp / number) về ms
const tsOf = (v: any): number => {
  if (!v) return 0;
  const n =
    typeof v === "string" ? Date.parse(v)
    : typeof v?.toMillis === "function" ? v.toMillis()
    : Number(v);
  return Number.isFinite(n) ? n : 0;
};
// Snapshot lấy từ cache cục bộ có thể cũ hơn dữ liệu đã nhận qua delta -> không cho ghi đè.
// Chỉ áp dụng cho snapshot fromCache; snapshot từ server luôn được tin (tránh lệch đồng hồ)
const isStaleCacheDoc = (existing: any, incoming: any): boolean => {
  if (!existing) return false;
  const e = tsOf(existing.updatedAt);
  const i = tsOf(incoming?.updatedAt);
  return e > 0 && i > 0 && i < e;
};

// Các (người dùng, bộ lọc store, năm) thuộc nhóm năm phụ đã được nạp trong phiên này: không truy vấn server lặp lại
const secondaryYearsFetched = new Set<string>();

export function useProjectsHydration({
  querySignature,
  userRole,
  allowedStores,
}: UseProjectsHydrationProps) {
  const globalYearFilter = useAppStore((state) => state.globalYearFilter);
  const globalStoreFilter = useAppStore((state) => state.globalStoreFilter);
  const setCloudStatus = useAppStore((state) => state.setCloudStatus);

  const projectsPool = useRef<Map<string, () => void>>(new Map());
  const prevSigProjects = useRef<string>("");
  const remoteYearsMetaRef = useRef<YearsMetaMap>({});

  // Atomic batching buffers for Projects and Tasks updates to prevent JS main thread freezing
  const pendingProjectsMap = useRef<Map<string, Project> | null>(null);
  const pendingProjectsChanged = useRef<Map<string, Project>>(new Map());
  const pendingProjectsRemoved = useRef<Set<string>>(new Set());
  const pendingDeletedTaskIds = useRef<Set<string>>(new Set());
  const projectsBatchTimer = useRef<NodeJS.Timeout | null>(null);

  const batchProjectsUpdate = useCallback((docChanges: any[], fromCache: boolean = false) => {
    if (!pendingProjectsMap.current) {
      const currentProjects = useAppStore.getState().projects || [];
      const map = new Map<string, Project>();
      currentProjects.forEach(p => map.set(p.id, p));
      pendingProjectsMap.current = map;
    }

    docChanges.forEach((change) => {
      const id = change.doc.id;
      const rawDocData = change.doc.data();
      const data = { id, ...rawDocData } as Project;
      const isTrueDelete = Boolean(data.isDeleted || change.type === "removed");

      if (isTrueDelete) {
        pendingProjectsMap.current!.delete(id);
        pendingProjectsChanged.current.delete(id);
        pendingProjectsRemoved.current.add(id);
      } else {
        if (fromCache && isStaleCacheDoc(pendingProjectsMap.current!.get(id), data)) return;
        const existing = pendingProjectsMap.current!.get(id);
        const resolvedData = existing ? {
          ...data,
          taskCount: data.taskCount !== undefined ? data.taskCount : (existing.taskCount ?? 0),
          taskSummaries: data.taskSummaries !== undefined ? data.taskSummaries : (existing.taskSummaries || []),
        } : data;

        // If this is a live server update and taskSummaries has shrunk (a task was deleted on another device),
        // immediately detect the removed task IDs and purge them from Zustand tasks & Dexie!
        if (!fromCache && Array.isArray(data.taskSummaries)) {
          const incomingIds = new Set(
            data.taskSummaries
              .map((s: any) => (s?.id ? String(s.id).trim().toLowerCase() : ""))
              .filter(Boolean)
          );
          if (existing && Array.isArray(existing.taskSummaries)) {
            existing.taskSummaries.forEach((oldS: any) => {
              const oldId = oldS?.id ? String(oldS.id).trim() : "";
              if (oldId && !incomingIds.has(oldId.toLowerCase())) {
                markTasksDeleted([oldId], id);
                pendingDeletedTaskIds.current.add(oldId);
              }
            });
          }
          // Also check in-memory tasks belonging to this project if all incoming summaries carry IDs (or summaries is empty)
          const allIncomingHaveIds = data.taskSummaries.every((s: any) => Boolean(s?.id));
          if (allIncomingHaveIds) {
            const currentTasks = useAppStore.getState().tasks || [];
            const nowMs = Date.now();
            currentTasks.forEach((t) => {
              if (!t || !t.id || t.isDeleted) return;
              if (!isTaskBelongsToProject(t, resolvedData)) return;
              const tIdLower = String(t.id).trim().toLowerCase();
              if (!incomingIds.has(tIdLower)) {
                // Guard against tasks created/updated locally within the last 15 seconds (optimistic UI in-flight)
                const taskUpdatedMs = tsOf(t.updatedAt || (t as any).createdAt);
                if (taskUpdatedMs > 0 && nowMs - taskUpdatedMs < 15000) return;
                markTasksDeleted([t.id], id);
                pendingDeletedTaskIds.current.add(t.id);
              }
            });
          }
        }

        pendingProjectsMap.current!.set(id, resolvedData);
        pendingProjectsChanged.current.set(id, resolvedData);
        pendingProjectsRemoved.current.delete(id);
      }
    });

    if (projectsBatchTimer.current) {
      clearTimeout(projectsBatchTimer.current);
    }

    projectsBatchTimer.current = setTimeout(() => {
      if (pendingProjectsMap.current) {
        const updatedList = Array.from(pendingProjectsMap.current.values());
        updatedList.sort((a, b) => (a.no || 0) - (b.no || 0));
        useAppStore.getState().setProjects(updatedList);

        if (pendingDeletedTaskIds.current.size > 0) {
          const delIdsLower = new Set(
            Array.from(pendingDeletedTaskIds.current).map((tid) => String(tid).trim().toLowerCase())
          );
          const delRawIds = Array.from(pendingDeletedTaskIds.current);
          pendingDeletedTaskIds.current.clear();

          const currentTasks = useAppStore.getState().tasks || [];
          const filteredTasks = currentTasks.filter(
            (t) => !t?.id || !delIdsLower.has(String(t.id).trim().toLowerCase())
          );
          if (filteredTasks.length !== currentTasks.length) {
            useAppStore.getState().setTasks(filteredTasks);
          }
          delRawIds.forEach((tid) => {
            deleteLocalRecord('taka_tasks', tid).catch(() => {});
          });
        }

        // Chỉ ghi Dexie phần thay đổi; xóa khỏi Dexie phần đã bị xóa (chống "sống lại")
        const changed = Array.from(pendingProjectsChanged.current.values());
        const removed = Array.from(pendingProjectsRemoved.current);
        pendingProjectsChanged.current.clear();
        pendingProjectsRemoved.current.clear();
        if (changed.length > 0) offloadPersistSnapshot('taka_projects', changed);
        removed.forEach((id) => { deleteLocalRecord('taka_projects', id).catch(() => {}); });
        const setNotifs = useAppStore.getState().setAllProjectsForNotifs;
        if (setNotifs) setNotifs(updatedList);
        pendingProjectsMap.current = null;
      }
    }, 40);
  }, []);

  // Stage 3 & Delta Pipeline Startup
  useEffect(() => {
    bootstrapLocalStore().catch(() => {});

    fetchYearsMeta().then((meta) => {
      remoteYearsMetaRef.current = meta;
      setMemoryYearsMeta(meta);
    });

    const stopDeltaSync = startDeltaSyncService();
    return () => {
      stopDeltaSync();
    };
  }, []);

  // Sleep/Wake Recovery Listener
  useEffect(() => {
    const handleSleepWakeRecovery = () => {
      // KHÔNG hủy listener khi thức giấc.
      // Hủy xong mà không effect nào tạo lại => mất realtime Projects, và bộ đếm activeListeners
      // không giảm => deltaInvalidateService tưởng doc đã có listener phủ rồi bỏ qua việc đọc lại.
      // Socket chết do iOS treo đã có initNetworkRecovery (syncResilience) xử lý.
      hydrateProjectsFromLocal(
        useAppStore.getState().globalYearFilter,
        useAppStore.getState().globalStoreFilter
      ).catch(() => {});
    };

    window.addEventListener("taka_sleep_wake_recovery", handleSleepWakeRecovery);
    return () => {
      window.removeEventListener("taka_sleep_wake_recovery", handleSleepWakeRecovery);
    };
  }, []);

  // Unmount cleanup
  useEffect(() => {
    return () => {
      projectsPool.current.forEach((u) => u());
      projectsPool.current.clear();
      if (projectsBatchTimer.current) clearTimeout(projectsBatchTimer.current);
    };
  }, []);

  // Main Unified Global Delta Sync for Projects
  useEffect(() => {
    if (!querySignature) return;

    if (prevSigProjects.current !== querySignature) {
      projectsPool.current.forEach((u) => u());
      projectsPool.current.clear();
      prevSigProjects.current = querySignature;
    }

    const isAllYears = globalYearFilter === "all" || globalYearFilter === "ALL";
    const liveTargetYears = isAllYears ? PRIMARY_ACTIVE_YEARS : [globalYearFilter];

    liveTargetYears.forEach(async (year) => {
      if (!projectsPool.current.has(year)) {
        // Step 1: Instant Local-First Dexie Hydration (0ms, 0 Read)
        const localCount = await hydrateProjectsFromLocal(year, globalStoreFilter);

        const isClosedOrPastYear = Number(year) < new Date().getFullYear();
        const hasLocalData =
          localCount > 0 ||
          (useAppStore.getState().projects || []).some((p) => String(p.year) === String(year));

        const deltaManaged =
          (hasLocalData && isClosedOrPastYear) ||
          (DELTA_FIRST.projects && hasMarker(await getLoadedMarkers(), 'projects_year', year));
        if (deltaManaged) {
          projectsPool.current.set(year, () => {});
          pullAndApplyDeltas(year)
            .then(() => setCloudStatus("online"))
            .catch(() => {});
          usePerformanceStore
            .getState()
            .trackGetDoc(
              `taka_projects_${year}`,
              true,
              localCount,
              "Dexie Local-First Projects Hydration (0 Read full collection)"
            );
          return;
        }

        if (false && isClosedOrPastYear && isYearCacheValid(year, remoteYearsMetaRef.current)) {
          const cached = loadYearDataFromLocal(year);
          if (cached && cached.projects && cached.projects.length > 0) {
            batchProjectsUpdate(
              cached.projects.map((p) => ({
                type: "added",
                doc: { id: p.id, data: () => p },
              }))
            );
            usePerformanceStore
              .getState()
              .trackGetDoc(
                `taka_projects_${year}`,
                true,
                cached.projects.length,
                "Metadata Fingerprint Cache Hit"
              );
            projectsPool.current.set(year, () => {});
            return;
          }
        }

        const colRef = collection(db, "artifacts", APP_ID, "public", "data", "taka_projects");
        const qProjects = buildStoreAndYearQuery(colRef, "store", "year", year, userRole, allowedStores);

        usePerformanceStore.getState().registerListener(`taka_projects_${year}`);
        const unsub = onSnapshot(
          qProjects,
          (snapshot) => {
            usePerformanceStore.getState().trackSnapshot(`taka_projects_${year}`, snapshot);
            batchProjectsUpdate(snapshot.docChanges(), snapshot.metadata.fromCache);
            setCloudStatus("online");

            if (DELTA_FIRST.projects && !snapshot.metadata.fromCache) {
              markLoaded('projects_year', year).catch(() => {});
              const u = projectsPool.current.get(year);
              projectsPool.current.set(year, () => {});
              u?.();
              usePerformanceStore.getState().unregisterListener(`taka_projects_${year}`);
            }
          },
          (e) => {
            console.error("Projects listener error:", e);
            setCloudStatus("offline");
          }
        );
        projectsPool.current.set(year, unsub);
      }
    });

    if (isAllYears) {
      SECONDARY_ACTIVE_YEARS.forEach(async (year) => {
        if (projectsPool.current.has(year)) return;
        const secKey = `${querySignature}_${globalStoreFilter}_${year}`;
        if (secondaryYearsFetched.has(secKey)) return;
        secondaryYearsFetched.add(secKey);
        try {
          const colRef = collection(db, "artifacts", APP_ID, "public", "data", "taka_projects");
          const qProjects = buildStoreAndYearQuery(colRef, "store", "year", year, userRole, allowedStores);
          let snapshot;
          try {
            snapshot = await getDocsFromCache(qProjects);
            if (snapshot.empty) {
              snapshot = await getDocsFromServer(qProjects);
            }
          } catch {
            snapshot = await getDocsFromServer(qProjects);
          }

          if (snapshot && !snapshot.empty) {
            const currentProjects = useAppStore.getState().projects || [];
            const projectMap = new Map<string, Project>();
            currentProjects.forEach((p) => projectMap.set(p.id, p));

            snapshot.forEach((docSnap) => {
              const data = { id: docSnap.id, ...docSnap.data() } as Project;
              if (!data.isDeleted) {
                projectMap.set(docSnap.id, data);
              }
            });

            const updatedList = Array.from(projectMap.values());
            updatedList.sort((a, b) => (a.no || 0) - (b.no || 0));
            useAppStore.getState().setProjects(updatedList);
          }
        } catch {
          // Lỗi mạng hoặc cache: cho phép thử lại ở lần chạy sau
          secondaryYearsFetched.delete(secKey);
        }
      });
    }
  }, [querySignature, userRole, allowedStores, globalYearFilter, globalStoreFilter, batchProjectsUpdate, setCloudStatus]);
}

      const map = new Map<string, Cost>();
      currentCosts.forEach((c) => map.set(c.id, c));
      pendingCostsMap.current = map;
    }

    docChanges.forEach((change) => {
      const id = change.doc.id;
      const rawDocData = change.doc.data();
      const data = { id, ...rawDocData } as Cost;
      const isTrueDelete = Boolean(data.isDeleted || change.type === "removed");

      if (isTrueDelete) {
        pendingCostsMap.current!.delete(id);
        pendingCostsChanged.current.delete(id);
        pendingCostsRemoved.current.add(id);
      } else {
        if (fromCache && isStaleCacheDoc(pendingCostsMap.current!.get(id), data)) return;
        pendingCostsMap.current!.set(id, data);
        pendingCostsChanged.current.set(id, data);
        pendingCostsRemoved.current.delete(id);
      }
    });

      clearTimeout(costsBatchTimer.current);
    }

        pendingCostsChanged.current.clear();
        pendingCostsRemoved.current.clear();

      costsTTLTimeouts.current.clear();

    };
  }, []);

  useEffect(() => {
    if (!querySignature || !isCostsRequested) {
      if (!isCostsRequested) {
        setIsCostsLoading(false);
        Array.from(costsPool.current.keys()).forEach((key) => {

            }, TTL_DURATION_MS);
            costsTTLTimeouts.current.set(key, timeoutId);
          }
        });
      }
      return;
    }

    costsTTLTimeouts.current.forEach((t) => clearTimeout(t));
    costsTTLTimeouts.current.clear();

    if (prevSigCosts.current !== querySignature) {
      costsPool.current.forEach((u) => u());
      costsPool.current.clear();
      prevSigCosts.current = querySignature;
    }

    const liveTargetYears = isAllYears ? PRIMARY_ACTIVE_YEARS : [globalYearFilter];

        const hasLocalCosts =
          localCount > 0 || currentCosts.some((c) => String(c.projectYear || "") === String(year));

        const deltaManaged =
          (hasLocalCosts && isClosedOrPastYear) ||
          (DELTA_FIRST.costs && hasMarker(await getLoadedMarkers(), 'costs_year', year));
        if (deltaManaged) {
          costsPool.current.set(year, () => {});
          setIsCostsLoading(false);
          pullAndApplyDeltas(year)
            .then(() => setCloudStatus("online"))
            .catch(() => {});
          usePerformanceStore
            .getState()
            .trackGetDoc(
              `taka_costs_${year}`,
              true,
              localCount,
              "Dexie Local-First Costs (0 Read full collection)"
            );
          return;
        } else {
          const colRef = collection(db, "artifacts", APP_ID, "public", "data", "taka_costs");
          const qCosts = buildStoreAndYearQuery(
            colRef,
            "projectStore",
            "projectYear",
            year,
            userRole,
            allowedStores
          );

          usePerformanceStore.getState().registerListener(`taka_costs_${year}`);
          const unsub = onSnapshot(
            qCosts,
            (snapshot) => {
              usePerformanceStore.getState().trackSnapshot(`taka_costs_${year}`, snapshot);
              setIsCostsLoading(false);
              const fromCache = snapshot.metadata.fromCache;
              batchCostsUpdate(snapshot.docChanges(), fromCache);

              if (DELTA_FIRST.costs && !fromCache) {
                markLoaded('costs_year', year).catch(() => {});
                const u = costsPool.current.get(year);
                costsPool.current.set(year, () => {});
                u?.();
                usePerformanceStore.getState().unregisterListener(`taka_costs_${year}`);
              }
            },
            (e) => {
              setIsCostsLoading(false);
              console.error("Costs listener error:", e);
            }
          );
          costsPool.current.set(year, unsub);
        }
      } else {
        setIsCostsLoading(false);
      }
    });

    if (isAllYears) {
      SECONDARY_ACTIVE_YEARS.forEach(async (year) => {
        if (costsPool.current.has(year)) return;
        try {
          const colRef = collection(db, "artifacts", APP_ID, "public", "data", "taka_costs");
          const qCosts = buildStoreAndYearQuery(
            colRef,
            "projectStore",
            "projectYear",
            year,
            userRole,

          );
          let snapshot;
          try {
            snapshot = await getDocsFromCache(qCosts);
            if (snapshot.empty) {
              snapshot = await getDocsFromServer(qCosts);
            }
          } catch {
            snapshot = await getDocsFromServer(qCosts);

            liveCosts.forEach((c) => costMap.set(c.id, c));

            snapshot.forEach((docSnap) => {
              const data = { id: docSnap.id, ...docSnap.data() } as Cost;
              if (!data.isDeleted) {
                costMap.set(docSnap.id, data);
              }
            });

            const updatedList = Array.from(costMap.values());
            useAppStore.getState().setCosts(updatedList);
          }
        } catch {}
      });
    }
  }, [
    querySignature,
    userRole,
    allowedStores,
    isCostsRequested,

    setIsCostsLoading,
    setCloudStatus,
  ]);
}

// Chuyển updatedAt (ISO string / Firestore Timestamp / number) về ms
const tsOf = (v: any): number => {
  if (!v) return 0;
  const n =
    typeof v === "string" ? Date.parse(v)
    : typeof v?.toMillis === "function" ? v.toMillis()
    : Number(v);
  return Number.isFinite(n) ? n : 0;
};
// Snapshot lấy từ cache cục bộ có thể cũ hơn dữ liệu đã nhận qua delta -> không cho ghi đè.
// Chỉ áp dụng cho snapshot fromCache; snapshot từ server luôn được tin (tránh lệch đồng hồ)
const isStaleCacheDoc = (existing: any, incoming: any): boolean => {
  if (!existing) return false;
  const e = tsOf(existing.updatedAt);
  const i = tsOf(incoming?.updatedAt);
  return e > 0 && i > 0 && i < e;
};

import React, { Dispatch, SetStateAction } from 'react';
import { Project, Task, DelegationGroup, AppUser } from '../../types';
import { useAppStore } from '../../store/useAppStore';
import { doc, deleteField } from 'firebase/firestore';
import { writeBatch } from '../../lib/firestoreWrites';
import { db, APP_ID } from '../../lib/firebase';
import { generateId } from '../../lib/utils';
import { saveTaskStateUnified, deleteTaskStateUnified } from '../../lib/taskStateService';

interface UseDetailBulkActionsParams {
  selectedTaskIds: Set<string>;
  setSelectedTaskIds: React.Dispatch<React.SetStateAction<Set<string>>>;
  lastCheckedTaskIdRef: React.MutableRefObject<string | null>;
  tasks: Task[];
  projects: Project[];
  delegationGroups: DelegationGroup[];
  currentUser: AppUser;
}

export function useDetailBulkActions({
  selectedTaskIds,
  setSelectedTaskIds,
  lastCheckedTaskIdRef,
  tasks,
  projects,
  delegationGroups,
  currentUser,
}: UseDetailBulkActionsParams) {
  const handleBulkAction = async (action: 'delegation_on' | 'delegation_off' | 'flexible_on' | 'flexible_off' | 'baseline' | 'unlink' | 'delete') => {
    if (selectedTaskIds.size === 0) return;

    const now = new Date().toISOString();
    const prevTasks = useAppStore.getState().tasks;

    // Special handler: Unlink modular tasks
    if (action === 'unlink') {
      const targetIds = Array.from(selectedTaskIds);
      const targetSet = new Set(targetIds);

      const parentCodeSet = new Set<string>();
      prevTasks.forEach(t => {
        if (targetSet.has(t.id) && t.taskCodePhu) {
          parentCodeSet.add(String(t.taskCodePhu).trim().toLowerCase());
        }
      });

      const allIdsToUnlink = new Set<string>();
      targetIds.forEach(id => allIdsToUnlink.add(id));
      prevTasks.forEach(t => {
        const rawParent = t.parentTaskId || (t as any).parentId;
        if (rawParent) {
          const rawStr = String(rawParent).trim().toLowerCase();
          if (targetSet.has(String(rawParent)) || parentCodeSet.has(rawStr)) {
            allIdsToUnlink.add(t.id);
          }
        }
      });

      const unlinkedArray = Array.from(allIdsToUnlink);

      // 1. 0ms Optimistic UI update in Zustand
      const updatedTasks = prevTasks.map(t =>
        allIdsToUnlink.has(t.id)
          ? { ...t, parentTaskId: undefined, parentId: undefined, updatedAt: now }
          : t
      );
      useAppStore.getState().setTasks(updatedTasks);
      setSelectedTaskIds(new Set());
      lastCheckedTaskIdRef.current = null;

      // 2. Update Dexie (IndexedDB)
      try {
        const { localDb } = await import('../../lib/localDb');
        await Promise.all(
          unlinkedArray.map(id => (localDb.tasks.update as any)(id, { parentTaskId: null, parentId: null, updatedAt: now }))
        );
      } catch (e) {
        console.warn('[useDetailBulkActions] Dexie update failed:', e);
      }

      // 3. Write batch to Firestore
      const batch = writeBatch(db);
      unlinkedArray.forEach(id => {
        const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', id);
        batch.update(docRef, { parentTaskId: deleteField(), parentId: null, updatedAt: now });
      });

      try {
        await batch.commit();
        const { recordBatchDeltas } = await import('../../lib/deltaSyncService');
        const deltaItems: any[] = unlinkedArray.map(id => {
          const t = prevTasks.find(item => item.id === id);
          return {
            docId: id,
            collectionName: 'taka_tasks',
            action: 'update',
            isPatch: true,
            data: { parentTaskId: null, parentId: null, updatedAt: now },
            year: t?.projectYear ? String(t.projectYear) : undefined,
          };
        });
        if (deltaItems.length > 0) {
          recordBatchDeltas(deltaItems, currentUser?.email || currentUser?.username).catch(() => {});
        }
        useAppStore.getState().setToast({
          title: "Success",
          message: `Unlinked ${unlinkedArray.length} modular task(s).`
        });
      } catch (err: any) {
        console.error(err);
        useAppStore.getState().setTasks(prevTasks);
        useAppStore.getState().setToast({
          title: "Error",
          message: "Failed to unlink tasks on server. Changes reverted."
        });
      }
      return;
    }

    try {
      if (action === 'delete') {
        const count = selectedTaskIds.size;
        const selectedObjs = tasks.filter(t => selectedTaskIds.has(t.id));
        setSelectedTaskIds(new Set());
        lastCheckedTaskIdRef.current = null;
        await deleteTaskStateUnified(selectedObjs.length > 0 ? selectedObjs : Array.from(selectedTaskIds));
        useAppStore.getState().setToast({ title: "Success", message: `Successfully deleted ${count} tasks.` });
        return;
      }

      const batch = writeBatch(db);
      let updateCount = 0;

      Array.from(selectedTaskIds).forEach(taskId => {
        const docRef = doc(db, "artifacts", APP_ID, "public", "data", "taka_tasks", taskId);
        const payload: any = { updatedAt: now };
        if (action === 'delegation_on') {
          payload.delegation = true;
          const task = tasks.find(t => t.id === taskId);
          if (task) {
            const deterministicGroupId = `del_${task.id}`;
            const existingGroup = delegationGroups.find(g => g.parentTaskId === task.id || g.id === deterministicGroupId);
            if (!existingGroup) {
               const dateStr = now.substring(0, 10);
               const activeUser = useAppStore.getState().currentUser || currentUser;
               const activeUserId = activeUser?.id && activeUser.id !== 'guest' ? activeUser.id : (activeUser?.username || 'system');
               const rawParty = (task.party && task.party.trim()) ? task.party.trim() : (activeUser?.username || '');
               const parsedParty = rawParty.split(/[,;/+]+/).map(s => s.trim()).filter(Boolean)[0] || rawParty;
               const newGroup = {
                 id: deterministicGroupId,
                 parentTaskId: task.id,
                 name: task.name || 'Delegated Task',
                 projectCode: task.projectCode,
                 projectStore: task.projectStore,
                 projectYear: (task.projectYear !== undefined && task.projectYear !== null && !isNaN(Number(task.projectYear))) ? Number(task.projectYear) : (Number(projects.find(p => p.code === task.projectCode)?.year) || new Date().getFullYear()),
                 subTasks: [{
                     id: generateId(),
                     name: '',
                     comments: '',
                     status: 'Assigned',
                     delegationStatus: 'Assigned',
                     priority: 'Mid',
                     party: parsedParty,
                     startAct: dateStr,
                     finishAct: dateStr,
                     updatedAt: now
                  }],
                 updatedAt: now,
                 createdBy: activeUserId
               };
               const delRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_delegation_groups', deterministicGroupId);
               batch.set(delRef, newGroup, { merge: true });
            }
          }
        }
        if (action === 'delegation_off') payload.delegation = false;
        if (action === 'flexible_on') payload.realtimeUpdate = true;
        if (action === 'flexible_off') payload.realtimeUpdate = false;
        
        if (action === 'baseline') {
          const task = tasks.find(t => t.id === taskId);
          if (task) {
            payload.baselineStart = task.start;
            payload.baselineFinish = task.finish;
          }
        }
        
        batch.update(docRef, payload);
        updateCount++;
      });

      if (updateCount > 0) {
        const updatedTasksList: (Partial<Task> & { id: string })[] = [];
        Array.from(selectedTaskIds).forEach(taskId => {
          const task = tasks.find(t => t.id === taskId);
          if (task) {
            const u: any = { id: taskId, updatedAt: now };
            if (action === 'delegation_on') u.delegation = true;
            if (action === 'delegation_off') u.delegation = false;
            if (action === 'flexible_on') u.realtimeUpdate = true;
            if (action === 'flexible_off') u.realtimeUpdate = false;
            if (action === 'baseline') {
              u.baselineStart = task.start;
              u.baselineFinish = task.finish;
            }
            updatedTasksList.push(u);
          }
        });
        if (updatedTasksList.length > 0) {
          saveTaskStateUnified(updatedTasksList);
        }

        try {
          await batch.commit();
        } catch (error: any) {
          console.error("Batch commit failed", error);
          useAppStore.getState().setToast({ title: "Save Failed", message: "Network error or permission denied. Data was not saved.", type: "error" });
          return;
        }

        useAppStore.getState().setToast({ title: "Success", message: `Successfully updated ${updateCount} tasks.` });
        setSelectedTaskIds(new Set());
        lastCheckedTaskIdRef.current = null;
      }
    } catch (e: any) {
      console.error(e);
      useAppStore.getState().setToast({ title: "Error", message: e.message });
    }
  };

  return {
    handleBulkAction,
  };
}

    // 0ms Optimistic UI update
    const updatedTasks = prevTasks.map((t) => {
      if (!allIdsToUnlink.has(String(t.id))) return t;
      const copy: any = { ...t, parentTaskId: undefined, parentId: undefined, updatedAt: now };
      delete copy.parentTask;
      delete copy.parentTaskCode;
      delete copy.parentCode;
      return copy as Task;
    });
    useAppStore.getState().setTasks(updatedTasks);
    setIsTaskModularModalOpen(false);
    setSelectedTaskIds([]);

    const batch = writeBatch(db);
    unlinkedArray.forEach((id) => {
      const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', String(id));
      batch.set(docRef, { parentTaskId: deleteField(), parentId: null, updatedAt: now }, { merge: true });
    });

    // Update Dexie (IndexedDB)
    try {
      const { localDb } = await import('../../lib/localDb');
      await Promise.all(
        unlinkedArray.map(id => (localDb.tasks.update as any)(String(id), { parentTaskId: null, parentId: null, updatedAt: now }))
      );
    } catch (e) {
      console.warn('[useGanttBatchOps] Dexie update on unlink failed:', e);
    }

    try {
      await batch.commit();
      const deltaItems: TakaDeltaItem[] = unlinkedArray.map(id => {
        const t = prevTasks.find(item => String(item.id) === String(id));
        return {
          docId: String(id),
          collectionName: 'taka_tasks' as const,
          action: 'update' as const,
          isPatch: true,
          data: { parentTaskId: null, parentId: null, updatedAt: now },
          year: t?.projectYear ? String(t.projectYear) : undefined,
        };
      });
      if (deltaItems.length > 0) {
        recordBatchDeltas(deltaItems, currentUser?.email || currentUser?.username).catch(() => {});
      }
      useAppStore.getState().setToast({
        title: "Success",
        message: `Unlinked ${unlinkedArray.length} modular task(s).`
      });
    } catch (err) {
      console.error('[handleUnlinkModularTasks] Batch commit error:', err);
      useAppStore.getState().setTasks(prevTasks);
      useAppStore.getState().setToast({
        title: "Error",
        message: "Failed to unlink tasks on server. Changes reverted."
      });
    }
  };

  // Batch text color update
  const handleSetTextColor = async (color: string | null) => {
    const batch = writeBatch(db);
    const now = new Date().toISOString();

    selectedTaskIds.forEach(id => {
      const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', id);
      batch.update(docRef, { textColor: color, updatedAt: now });
    });

    try {
      await batch.commit();
      const deltaItems: TakaDeltaItem[] = selectedTaskIds.map(id => {
        const t = tasks.find(item => item.id === id);
        return {
          docId: id,
          collectionName: 'taka_tasks',
          action: 'update',
          isPatch: true,
          data: { textColor: color, updatedAt: now },
          year: t?.projectYear ? String(t.projectYear) : undefined,
        };
      });
      recordBatchDeltas(deltaItems, currentUser?.email || currentUser?.username).catch(() => {});
      useAppStore.getState().setToast({
        title: "Success",
        message: `Updated text color for ${selectedTaskIds.length} tasks.`
      });
    } catch (err) {
      console.error(err);
    }
  };

  // Batch baseline update
  const handleSetBaselineSelected = async () => {
    const batch = writeBatch(db);
    const now = new Date().toISOString();

    selectedTaskIds.forEach(id => {
      const t = tasks.find(item => item.id === id);
      if (t && t.start && t.finish) {
        const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', id);
        batch.update(docRef, {
          baselineStart: t.start,
          baselineFinish: t.finish,
          updatedAt: now
        });
      }
    });

    try {
      await batch.commit();
      const deltaItems: TakaDeltaItem[] = [];
      selectedTaskIds.forEach(id => {
        const t = tasks.find(item => item.id === id);
        if (t && t.start && t.finish) {
          deltaItems.push({
            docId: id,
            collectionName: 'taka_tasks',
            action: 'update',
            isPatch: true,
            data: { baselineStart: t.start, baselineFinish: t.finish, updatedAt: now },
            year: t.projectYear ? String(t.projectYear) : undefined,
          });
        }
      });
      if (deltaItems.length > 0) {
        recordBatchDeltas(deltaItems, currentUser?.email || currentUser?.username).catch(() => {});
      }
      useAppStore.getState().setToast({
        title: "Success",
        message: `Saved baseline dates for selected tasks.`
      });
    } catch (err) {
      console.error(err);
    }
  };

  // Batch realtime sync toggle
  const handleToggleRealtimeUpdate = async () => {
    const batch = writeBatch(db);
    const now = new Date().toISOString();

    selectedTaskIds.forEach(id => {
      const t = tasks.find(item => item.id === id);
      if (t) {
        const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', id);
        batch.update(docRef, {
          realtimeUpdate: !t.realtimeUpdate,
          updatedAt: now
        });
      }
    });

    try {
      await batch.commit();
      const deltaItems: TakaDeltaItem[] = [];
      selectedTaskIds.forEach(id => {
        const t = tasks.find(item => item.id === id);
        if (t) {
          deltaItems.push({
            docId: id,
            collectionName: 'taka_tasks',
            action: 'update',
            isPatch: true,
            data: { realtimeUpdate: !t.realtimeUpdate, updatedAt: now },
            year: t.projectYear ? String(t.projectYear) : undefined,
          });
        }
      });
      if (deltaItems.length > 0) {
        recordBatchDeltas(deltaItems, currentUser?.email || currentUser?.username).catch(() => {});
      }
      useAppStore.getState().setToast({
        title: "Success",
        message: `Toggled realtime sync on selected tasks.`
      });
    } catch (err) {
      console.error(err);
    }
  };

  // Toggle Pin project
  const handleTogglePin = async (p: Project) => {
    const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_projects', p.id);
    try {
      await writeBatch(db).update(docRef, {
        isPinned: !p.isPinned,
        updatedAt: new Date().toISOString()
      }).commit();
      recordDelta({
        docId: p.id,
        collectionName: 'taka_projects',
        action: 'update',
        isPatch: true,
        data: { isPinned: !p.isPinned, updatedAt: new Date().toISOString() },
        year: p.year ? String(p.year) : undefined,
        updatedBy: currentUser?.email || currentUser?.username,
      }).catch(() => {});
    } catch (err) {
      console.error(err);
    }
  };

  // Batch delete selected tasks/projects
  const handleDeleteSelected = () => {
    const total = selectedTaskIds.length + selectedProjectIds.length;
    if (total === 0) return;

    setConfirmModal({
      isOpen: true,
      title: "Delete Selected Items",
      message: `Are you sure you want to delete ${total} selected item(s)? This action cannot be undone.`,
      onConfirm: async () => {
        const now = new Date().toISOString();
        const currentTasks = useAppStore.getState().tasks;
        const currentProjects = useAppStore.getState().projects;
        const deletedTaskIds = [...selectedTaskIds];
        const deletedProjectIds = [...selectedProjectIds];
        const taskSet = new Set(deletedTaskIds.map(String));
        const projSet = new Set(deletedProjectIds.map(String));

        // If any deleted task is a parent task, unlink its subtasks so they don't break
        const parentCodeSet = new Set<string>();
        currentTasks.forEach(t => {
          if (taskSet.has(String(t.id)) && t.taskCodePhu) {
            parentCodeSet.add(String(t.taskCodePhu).trim().toLowerCase());
          }
        });

        const childrenToUnlink = currentTasks.filter(t => {
          if (taskSet.has(String(t.id))) return false; // will be deleted anyway
          const rawParent = t.parentTaskId || (t as any).parentId;
          if (rawParent) {
            const rawStr = String(rawParent).trim().toLowerCase();
            return taskSet.has(String(rawParent)) || parentCodeSet.has(rawStr);
          }
          return false;
        });

        // 1. Optimistic UI update & atomic task deletion via deleteTaskStateUnified (0ms)
        if (deletedTaskIds.length > 0) {
          const taskObjsToDelete = currentTasks.filter(t => deletedTaskIds.includes(String(t.id)));
          deleteTaskStateUnified(taskObjsToDelete.length > 0 ? taskObjsToDelete : deletedTaskIds).catch(() => {});
        }

        if (deletedProjectIds.length > 0) {
          const remainingProjects = useAppStore.getState().projects.filter(p => 
            !projSet.has(String(p.id)) && !projSet.has(String(p.no))
          );
          useAppStore.getState().setProjects(remainingProjects);
          try {
            const { localDb } = await import('../../lib/localDb');
            await localDb.projects.bulkDelete(deletedProjectIds);
          } catch (e) {}
        }

        setSelectedTaskIds([]);
        setSelectedProjectIds([]);

        // 2. Commit project deletions to Firestore (task deletions & child unlinks are already committed by deleteTaskStateUnified)
        try {
          if (deletedProjectIds.length > 0) {
            const batch = writeBatch(db);
            deletedProjectIds.forEach(id => {
              const p = currentProjects.find(item => String(item.id) === id || String(item.no) === id);
              const docId = p?.id ? String(p.id) : id;
              const docRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_projects', docId);
              batch.delete(docRef);
            });
            await batch.commit();

            const deltaItems: TakaDeltaItem[] = deletedProjectIds.map(id => {
              const p = currentProjects.find(item => String(item.id) === id || String(item.no) === id);
              const docId = p?.id ? String(p.id) : id;
              return {
                docId,
                collectionName: 'taka_projects' as const,
                action: 'delete' as const,
                isPatch: false,
                data: null
              };
            });
            if (deltaItems.length > 0) {
              recordBatchDeltas(deltaItems, currentUser?.email || currentUser?.username).catch(() => {});
            }
          }

          useAppStore.getState().setToast({
            title: "Success",
            message: `Deleted ${total} item(s).`
          });
        } catch (err) {
          console.error(err);
          useAppStore.getState().setToast({
            title: "Error",
            message: "Failed to delete item(s) on server."
          });
        }
      }
    });
  };

  // Computed sum duration
  const computedSum = useMemo(() => {
    if (sumMode !== 'duration') return null;
    let sum = 0;
    selectedTaskIds.forEach(id => {
      const t = tasks.find(item => item.id === id);
      if (t && t.duration) {
        sum += Number(t.duration) || 0;
      }
    });
    return sum;
  }, [sumMode, selectedTaskIds, tasks]);

  const isAllSelected = useMemo(() => {
    if (paginatedProjects.length === 0) return false;
    return paginatedProjects.every(p => {
      const pTasks = getGroupTasksForProject(p);
      return pTasks.length > 0 && pTasks.every(t => selectedTaskSet.has(t.id));
    });
  }, [paginatedProjects, getGroupTasksForProject, selectedTaskSet]);

  const handleToggleSelectAll = useCallback(() => {
    if (isAllSelected) {
      setSelectedTaskIds([]);
      setSelectedProjectIds([]);
    } else {
      const allIds: string[] = [];
      const projIds: string[] = [];
      paginatedProjects.forEach(p => {
        projIds.push(p.id || p.code);
        const pTasks = getGroupTasksForProject(p);
        pTasks.forEach(t => allIds.push(t.id));
      });
      setSelectedTaskIds(allIds);
      setSelectedProjectIds(projIds);
    }
  }, [isAllSelected, paginatedProjects, getGroupTasksForProject]);

  return {
    selectedTaskIds,
    setSelectedTaskIds,
    selectedProjectIds,
    setSelectedProjectIds,
    selectedTaskSet,
    selectedProjectSet,
    isAllSelected,
    handleToggleSelectAll,
    handleTaskCheckboxChange,
    handleDeleteSelected,
    handleSetTextColor,
    handleSetBaselineSelected,
    handleToggleRealtimeUpdate,
    handleTogglePin,
    handleGroupModularTasks,
    handleUnlinkModularTasks,
    isTaskModularModalOpen,
    setIsTaskModularModalOpen,
    expandedModularTasks,
    setExpandedModularTasks,
    expandedModularSet,
    toggleModularParent,
    sumMode,
    setSumMode,
    computedSum,
    isHighlightDelegationOn,
    setIsHighlightDelegationOn,
    isEditMode,
    setIsEditMode,
    isHideMode,
    setIsHideMode,
    hiddenRows,
    setHiddenRows,
    hiddenRowsSet,
    toggleHideRow,
    getRowVisibilityClass
  };
}

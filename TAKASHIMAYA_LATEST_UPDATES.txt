/**
 * Cross-Check Engine: Tự động kiểm tra chéo và thanh dọn các task rác/cũ (Ghost tasks)
 * Khi một thiết bị (iPad/iPhone) kết nối Online lại sau khi ở trạng thái Offline,
 * hàm này đối soát danh sách task trong RAM/Dexie với `p.taskSummaries` chính thức trên Server.
 * Nếu phát hiện task cục bộ không hề có trên Server (và không nằm trong hàng đợi ghi offline syncQueue),
 * hệ thống tự động dọn dẹp task đó khỏi RAM & Dexie để đưa số lượng task về khớp 100% với Server.
 */
export function reconcileObsoleteLocalTasks(
  tasks: Task[],
  projects: Project[]
): { reconciledTasks: Task[]; purgedCount: number } {
  if (!tasks || tasks.length === 0 || !projects || projects.length === 0) {
    return { reconciledTasks: tasks, purgedCount: 0 };
  }

  const projectSummaryMap = new Map<string, {
    ids: Set<string>;
    codeNames: Set<string>;
    names: Set<string>;
  }>();

  projects.forEach((p) => {
    if (p.taskSummaries && Array.isArray(p.taskSummaries) && p.taskSummaries.length > 0) {
      const ids = new Set<string>();
      const codeNames = new Set<string>();
      const names = new Set<string>();

      p.taskSummaries.forEach((ts) => {
        if (ts.id) ids.add(String(ts.id).toLowerCase());
        const name = ts.name ? String(ts.name).trim().toLowerCase() : '';
        const code = ts.taskCodePhu ? String(ts.taskCodePhu).trim().toLowerCase() : '';
        if (name) names.add(name);
        if (code && name) codeNames.add(`${code}:::${name}`);
      });

      if (p.id) projectSummaryMap.set(String(p.id).toLowerCase(), { ids, codeNames, names });
      if (p.code) projectSummaryMap.set(String(p.code).toLowerCase(), { ids, codeNames, names });
    }
  });

  if (projectSummaryMap.size === 0) {
    return { reconciledTasks: tasks, purgedCount: 0 };
  }

  let purgedCount = 0;
  const purgedIds: string[] = [];

  const reconciledTasks = tasks.filter((t) => {
    if (t.isDeleted) return false;

    const pKeyId = t.projectId ? String(t.projectId).toLowerCase() : '';
    const pKeyCode = t.projectCode ? String(t.projectCode).toLowerCase() : '';
    const summaryInfo = (pKeyId && projectSummaryMap.get(pKeyId)) || (pKeyCode && projectSummaryMap.get(pKeyCode));

    if (!summaryInfo) {
      return true;
    }

    const tId = t.id ? String(t.id).toLowerCase() : '';
    const tName = t.name ? String(t.name).trim().toLowerCase() : '';
    const tCode = t.taskCodePhu ? String(t.taskCodePhu).trim().toLowerCase() : '';

    const hasMatch =
      (tId && summaryInfo.ids.has(tId)) ||
      (tCode && tName && summaryInfo.codeNames.has(`${tCode}:::${tName}`)) ||
      (tName && summaryInfo.names.has(tName));

    if (hasMatch) {
      return true;
    }

    // Local task does not match server taskSummaries! Mark as obsolete ghost task and purge!
    purgedCount++;
    if (t.id) purgedIds.push(t.id);
    return false;
  });

  if (purgedCount > 0 && purgedIds.length > 0) {
    markTasksDeleted(purgedIds);
    purgedIds.forEach((id) => {
      deleteLocalRecord('taka_tasks', id).catch(() => {});
    });
  }

  return { reconciledTasks, purgedCount };
}

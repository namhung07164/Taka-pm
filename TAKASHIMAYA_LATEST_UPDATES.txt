import { doc, getDocFromServer, getDoc } from 'firebase/firestore';
import { db, APP_ID } from './firebase';
import { useAppStore } from '../store/useAppStore';
import { usePerformanceStore } from '../store/usePerformanceStore';
import type { TakaDelta, TakaDeltaItem } from '../types';

const PHYSICAL_COLLECTION: Record<TakaDeltaItem['collectionName'], string> = {
  taka_projects: 'taka_projects',
  taka_tasks: 'taka_tasks',
  taka_costs: 'taka_costs',
  taka_delegations: 'taka_delegation_groups',
};

/**
 * Với chế độ DELTA_FIRST: Mọi tín hiệu invalidate gửi qua taka_deltas đều là thay đổi thực tế
 * từ thiết bị khác (chỉ 1 doc/lần sửa). Luôn cho phép resolve để cập nhật ngay vào Dexie + Zustand,
 * đảm bảo khi người dùng chuyển sang bất kỳ tab nào thì dữ liệu ở tab đó cũng đã mới nhất 100%.
 */
function isRelevant(item: TakaDeltaItem): boolean {
  if (!item || !item.docId || !item.collectionName) return false;
  return true;
}

/**
 * Đọc lại đúng các document vừa nhận tín hiệu invalidate từ thiết bị khác (1 Read / 1 Doc sửa đổi)
 * và chuyển thành synthetic delta để applyIncomingDeltas cập nhật nguyên tử vào Zustand + Dexie.
 */
export async function resolveInvalidations(items: TakaDeltaItem[]): Promise<TakaDelta[]> {
  // Khử trùng lặp theo collectionName + docId
  const unique = new Map<string, TakaDeltaItem>();
  for (const it of items) {
    if (!isRelevant(it)) continue;
    unique.set(`${it.collectionName}:${it.docId}`, it);
  }
  if (unique.size === 0) return [];

  const now = Date.now();
  const nowIso = new Date(now).toISOString();

  const results = await Promise.all(
    Array.from(unique.values()).map(async (it): Promise<TakaDeltaItem | null> => {
      const physical = PHYSICAL_COLLECTION[it.collectionName];
      if (!physical) return null;
      try {
        const ref = doc(db, 'artifacts', APP_ID, 'public', 'data', physical, it.docId);
        let snap;
        try {
          snap = await getDocFromServer(ref);
        } catch {
          snap = await getDoc(ref);
        }
        usePerformanceStore
          .getState()
          .trackGetDoc(physical, false, 1, `Delta invalidate re-read: ${it.docId}`);

        if (!snap.exists()) {
          return {
            docId: it.docId,
            collectionName: it.collectionName,
            action: 'delete',
            isPatch: false,
            data: { id: it.docId, isDeleted: true },
          };
        }
        const data = snap.data() as Record<string, any>;
        if (data?.isDeleted) {
          return {
            docId: it.docId,
            collectionName: it.collectionName,
            action: 'delete',
            isPatch: false,
            data: { id: it.docId, isDeleted: true, ...data },
          };
        }
        return {
          docId: it.docId,
          collectionName: it.collectionName,
          action: 'create',
          isPatch: false,
          data: { id: it.docId, ...data },
          year: String(data.year || data.projectYear || it.year || ''),
        };
      } catch (err) {
        console.warn('[deltaSync] Failed to re-read invalidated doc:', it.collectionName, it.docId, err);
        return null;
      }
    })
  );

  const resolved = results.filter((x): x is TakaDeltaItem => x !== null);
  if (resolved.length === 0) return [];

  return [
    {
      id: `delta_resolved_${now}`,
      action: 'batch_update',
      batchItems: resolved,
      updatedBy: 'delta-invalidate',
      sessionId: '__resolved__',
      timestamp: nowIso,
      numericTime: now,
    },
  ];
}

          const pRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', pu.id);

        }
      });
      
      const allUpdatesToSave = [
        newT,
        ...depUpdates,
        ...updatedParents.filter(pt => pt.id !== newT.id && !depUpdates.some(d => d.id === pt.id)),
        ...predecessorUpdates.filter(pu => pu.id !== t.id && !depUpdates.some(d => d.id === pu.id))

        return createPatchDeltaItem({
          docId: item.id,
          collectionName: 'taka_tasks',
          previous: prev,
          updated: item,
          action: 'update',
          year: (item as any).projectYear ? String((item as any).projectYear) : (t.projectYear ? String(t.projectYear) : undefined),
        });
      });

      saveTaskStateUnified(allUpdatesToSave);

      try {
        const { markExplicitDelta } = await import('../../lib/autoDeltaQueue');
        markExplicitDelta(deltaItems);
        await batch.commit();
        recordBatchDeltas(deltaItems, currentUser.username || currentUser.id).catch(() => {});
      } catch (error: any) {
        console.error("Batch commit failed", error);
        useAppStore.getState().setToast({ title: "Save Failed", message: "Network error or permission denied. Data was not saved.", type: "error" });
        saveTaskStateUnified([t, ...depUpdates.map(d => tasks.find(tsk => tsk.id === d.id) || d)]);
        return;
      }
      
      undoManager.push({
        description: `Update Task Schedule: ${t.name}`,
        revert: async () => {
          const revertBatch = writeBatch(db);
          revertBatch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', t.id), t);
          updatesToDependents.forEach(depTask => {
            const originalDep = tasks.find(tsk => tsk.id === depTask.id);
            if (originalDep) {
              revertBatch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', depTask.id), originalDep);
            }
          });
          revertBatch.commit().catch(console.error);
        }
      });
    } catch (e) {
      console.error(e);
    }
  };

  const handleSaveNotes = async (taskId: string, notes: TaskNote[]) => {
    const t = tasks.find(item => item.id === taskId);
    if (t) {
      await handleUpdateTask(t, { notes });
    }
  };

  return {
    handleDeleteTask,
    handleUpdateTask,
    handleSaveNotes,
  };
}

export async function executeTaskSave({
  formData,
  task,
  projects,
  tasks,
  cloudStatus,
  currentUser,
  setToast,
  onClose,
  setIsSubmitting
}: TaskSaveParams): Promise<boolean> {
  const currentProject = projects.find(p => 
    p.code === formData.projectCode &&
    (String(p.year || '') === String(formData.projectYear || '') || (!p.year && !formData.projectYear)) &&
    (String(p.store || '') === String(formData.projectStore || '') || (!p.store && !formData.projectStore))
  );

  if (!currentProject) {
    useAppStore.getState().setToast({ title: "Notice", message: "Please select a valid project." });
    setIsSubmitting(false);
    return false;
  }

  const currentYear = (!isNaN(Number(currentProject?.year)) && currentProject?.year !== null && currentProject?.year !== '') 
    ? Number(currentProject.year) 
    : (formData.projectYear && !isNaN(Number(formData.projectYear)) ? Number(formData.projectYear) : new Date().getFullYear());

  const targetPrjKey = buildCanonicalProjectKey(
    currentProject.code || formData.projectCode,
    currentProject.year || formData.projectYear,
    currentProject.store || formData.projectStore,
    projects
  );

  // Check for duplicate task name or code in the same project
  const normFormName = normalizeTaskNameForComparison(formData.name, formData.projectCode || currentProject?.code);
  const formTaskCode = formData.taskCodePhu ? String(formData.taskCodePhu).trim().toLowerCase() : '';

  const isDuplicate = tasks.some(t => {
    if (t.id === task?.id || t.isDeleted) return false;
    
    const tPrjKey = buildCanonicalProjectKey(t.projectCode, t.projectYear, t.projectStore, projects);
    const sameProject = (tPrjKey === targetPrjKey);
    if (!sameProject) return false;

    // Check exact name
    if (t.name && String(t.name).trim().toLowerCase() === String(formData.name || '').trim().toLowerCase()) {
      return true;
    }
    // Check normalized name
    const tNorm = normalizeTaskNameForComparison(t.name, t.projectCode);
    if (normFormName && tNorm && normFormName === tNorm) {
      return true;
    }
    // Check taskCodePhu
    if (formTaskCode && t.taskCodePhu && String(t.taskCodePhu).trim().toLowerCase() === formTaskCode) {
      return true;
    }
    return false;
  });

  if (isDuplicate) {
    setToast({ title: "Duplicate Task", message: `Task "${formData.name}" already exists in this project. Duplicate sub-tasks are not allowed.` });
    setIsSubmitting(false);
    return false;
  }

  let id = task ? task.id : `t_${Date.now()}_${generateId().substring(0, 8)}`;
  if (id && id.startsWith('ts_')) {
    let realTask = tasks.find(t => 
      !t.id.startsWith('ts_') &&
      t.projectCode === currentProject.code &&
      (String(t.projectStore || '') === String(currentProject.store || '')) &&
      (String(t.projectYear || '') === String(currentProject.year || '')) &&
      ((task?.taskCodePhu && t.taskCodePhu === task.taskCodePhu) ||
       (task?.name && t.name && t.name.trim().toLowerCase() === task.name.trim().toLowerCase()))
    );
    if (!realTask && currentProject?.code) {
      try {
        const { ensureProjectTasksLoaded } = await import('../../lib/taskStateService');
        const sameIdentity = (x: Task) =>
          String(x.projectStore || '').trim().toLowerCase() === String(currentProject.store || '').trim().toLowerCase() &&
          String(x.projectYear ?? '').trim() === String(currentProject.year ?? '').trim();
        const loaded: Task[] = await ensureProjectTasksLoaded(currentProject.code);
        const loadedSame = loaded.filter(sameIdentity);
        if (loadedSame.length > 0) {
          realTask = loadedSame.find(t =>
            (task?.taskCodePhu && t.taskCodePhu === task.taskCodePhu) ||
            (task?.name && t.name && t.name.trim().toLowerCase() === task.name.trim().toLowerCase())
          );
        }
      } catch (err) {
        console.warn("Failed to find real task for ts_ in local/cache:", err);
      }
    }
    if (realTask) {
      id = realTask.id;
    } else {
      id = `t_${Date.now()}_${generateId().substring(0, 8)}`;
    }
  }
  let newOrderIndex = task?.orderIndex;
  
  if (!task) {
    const existingProjectTasks = tasks.filter(t => 
      t.projectCode === currentProject.code && 
      (String(t.projectYear || '') === String(currentProject.year || '') || (!t.projectYear && !currentProject.year)) &&
      (String(t.projectStore || '') === String(currentProject.store || '') || (!t.projectStore && !currentProject.store))
    );
    let maxOrder = 0;
    existingProjectTasks.forEach(t => {
      const order = t.orderIndex !== undefined ? t.orderIndex : 0;
      if (order > maxOrder) maxOrder = order;
    });
    newOrderIndex = maxOrder + 1;
  }

  if (formData.predecessor !== undefined && (!task || formData.predecessor !== task.predecessor)) {
    if (formData.predecessor) {
      const predContext: PredecessorContext = {
        projectId: formData.projectId || task?.projectId || currentProject?.id,
        projectCode: formData.projectCode || task?.projectCode || currentProject?.code,
        projectYear: formData.projectYear || task?.projectYear || currentProject?.year,
        projectStore: formData.projectStore || task?.projectStore || currentProject?.store,
        projectLocation: formData.location || task?.location || currentProject?.location
      };
      formData.predecessor = resolvePredecessorString(formData.predecessor, tasks, projects, formData.projectCode, predContext, formData.predecessorTaskIds);
      const taskNoMap = buildTaskNoMap(projects, tasks);
      let parts = splitPredecessors(formData.predecessor, taskNoMap);
      if (parts.length >= 2) {
        if (!/FF|SF/i.test(parts[1])) {
          if (!/SS|FS/i.test(parts[1])) parts[1] += "FF";
          else parts[1] = parts[1].replace(/SS|FS/ig, "FF");
        }
        formData.predecessor = parts.join(", ");
      }
    }
    const taskNoMap = buildTaskNoMap(projects, tasks);
    const dummyTask: any = {
      projectId: formData.projectId || task?.projectId || currentProject?.id,
      projectCode: formData.projectCode || task?.projectCode || currentProject?.code,
      projectYear: formData.projectYear || task?.projectYear || currentProject?.year,
      projectStore: formData.projectStore || task?.projectStore || currentProject?.store,
      location: formData.location || task?.location || currentProject?.location
    };
    parsePredecessors(formData.predecessor, taskNoMap, dummyTask);
  }

  try {
    const todayDate = getDefaultDatetimeLocal();
    let finalFormData = { ...formData };
    if (!finalFormData.start) finalFormData.start = todayDate;
    if (!finalFormData.finish) finalFormData.finish = finalFormData.start || todayDate;
    if (!finalFormData.baselineStart) finalFormData.baselineStart = finalFormData.start || todayDate;
    if (!finalFormData.baselineFinish) finalFormData.baselineFinish = finalFormData.finish || todayDate;
    if (finalFormData.duration === undefined || finalFormData.duration === null || String(finalFormData.duration).trim() === '') {
      finalFormData.duration = "0";
    }
    let durationVal = parseFloat(String(finalFormData.duration || "0"));
    if (durationVal < 0) {
      finalFormData.duration = "0";
      finalFormData.finish = finalFormData.start;
    }
    if (finalFormData.durationManual ?? true) {
      finalFormData.durationManual = true;
      if (finalFormData.start && finalFormData.duration !== undefined && finalFormData.duration !== '') {
        const s = parseDateTime(finalFormData.start);
        const days = parseFloat(String(finalFormData.duration));
        if (s && !isNaN(s.getTime()) && !isNaN(days)) {
          const f = new Date(s.getTime() + days * 24 * 60 * 60 * 1000);
          finalFormData.finish = formatToDatetimeLocal(f);
        }
      }
    }
    if (!finalFormData.party) {
      finalFormData.party = currentProject?.party || 'Takashimaya';
    }

    if (cloudStatus === 'offline') {
      let payload = {
        ...finalFormData,
        id,
        projectYear: Number(currentYear) || new Date().getFullYear(),
        comments: task?.comments || finalFormData.comments || [],
        updatedAt: new Date().toISOString(),
        orderIndex: newOrderIndex
      };
      addPendingUpdate({ collectionPath: `artifacts/${APP_ID}/public/data/taka_tasks`, action: 'set', docId: id, data: payload } as any);
      setToast({ title: 'Draft Saved', message: 'Task saved offline. Will sync when connection returns.' });
      onClose();
      setIsSubmitting(false);
      return true;
    }
    
    let dataToSave: any = {
      ...finalFormData,
      id,
      projectYear: Number(currentYear) || new Date().getFullYear(),
      comments: task?.comments || finalFormData.comments || [],
      updatedAt: new Date().toISOString(),
      orderIndex: newOrderIndex
    };

    if (dataToSave.predecessor && dataToSave.predecessor.trim()) {
      dataToSave.realtimeUpdate = true;
    } else {
      dataToSave.realtimeUpdate = false;
    }

    // Explicitly delete fields from Firestore if unlinked or cleared
    if (task?.parentTaskId && !finalFormData.parentTaskId) {
      dataToSave.parentTaskId = deleteField();
      dataToSave.parentId = null;
    }
    if (task?.taskCodePhu && (!finalFormData.taskCodePhu || !finalFormData.taskCodePhu.trim())) {
      dataToSave.taskCodePhu = deleteField();
    }

    Object.keys(dataToSave).forEach(key => {
      if ((dataToSave as any)[key] === undefined) {
        delete (dataToSave as any)[key];
      }
    });

    const oldTaskNoMap = buildTaskNoMap(projects, tasks);
    const newTasksForMap = task 
       ? tasks.map(t => t.id === id ? dataToSave as Task : t) 
       : [...tasks, dataToSave as Task];
    const newTaskNoMap = buildTaskNoMap(projects, newTasksForMap);
    const oldCodeToNewCode = getPredecessorCodeUpdates(oldTaskNoMap, newTaskNoMap, projects, newTasksForMap);

    if (task) {
      const oldCode = (task.taskCodePhu || computeTaskCode(projects, tasks, task)).trim();
      const newCode = (dataToSave.taskCodePhu || computeTaskCode(projects, newTasksForMap, dataToSave as Task)).trim();
      if (oldCode && newCode && oldCode.toLowerCase() !== newCode.toLowerCase()) {
        oldCodeToNewCode.set(oldCode.toLowerCase(), newCode);
      }
    }

    const predecessorUpdatesOrig = updatePredecessorsStrings(oldCodeToNewCode, tasks, projects, oldTaskNoMap);
    const predecessorUpdates = predecessorUpdatesOrig.filter(pu => pu.id !== id);
    const predecessorUpdatesMap = new Map(predecessorUpdates.map(pu => [pu.id, pu]));

    const updatesToDependents = scheduleDependentTasks(dataToSave as Task, tasks, projects);
    
    const scheduledUpdatedTask = updatesToDependents.find(d => d.id === dataToSave.id);
    if (scheduledUpdatedTask) {
      dataToSave = scheduledUpdatedTask as any;
    }

    const batch = writeBatch(db);
    if (task) {
      batch.update(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', id), dataToSave);
    } else {
      batch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', id), dataToSave);
    }

    const depUpdates = updatesToDependents.filter(d => d.id !== id);
    const mappedDepUpdates = new Set<string>();

    depUpdates.forEach(depTask => {
      const depProj = projects.find(p => p.code === depTask.projectCode);
      const depYear = (depTask.projectYear !== undefined && depTask.projectYear !== null && !isNaN(Number(depTask.projectYear)))
        ? Number(depTask.projectYear)
        : (depProj?.year && !isNaN(Number(depProj.year)) ? Number(depProj.year) : new Date().getFullYear());
      const pu = predecessorUpdatesMap.get(depTask.id);
      const data = pu 
        ? { ...depTask, projectYear: depYear, predecessor: pu.predecessor, updatedAt: new Date().toISOString() } 
        : { ...depTask, projectYear: depYear, updatedAt: new Date().toISOString() };
      Object.keys(data).forEach(key => {
        if ((data as any)[key] === undefined) {
          delete (data as any)[key];
        }
      });
      batch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', depTask.id), data, { merge: true });
      mappedDepUpdates.add(depTask.id);
    });

    predecessorUpdates.forEach(pu => {
      if (!mappedDepUpdates.has(pu.id)) {
         batch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', pu.id), { predecessor: pu.predecessor, updatedAt: pu.updatedAt }, { merge: true });
      }
    });

    const optimisticList: any[] = [dataToSave];
    depUpdates.forEach(depTask => {
      const pu = predecessorUpdatesMap.get(depTask.id);
      if (pu) {
        optimisticList.push({ ...depTask, predecessor: pu.predecessor, updatedAt: pu.updatedAt });
      } else {
        optimisticList.push(depTask);
      }
    });
    predecessorUpdates.forEach(pu => {
      if (!mappedDepUpdates.has(pu.id)) {
        optimisticList.push({ id: pu.id, predecessor: pu.predecessor, updatedAt: pu.updatedAt });
      }
    });
    
    saveTaskStateUnified(optimisticList);

    const isNewTask = !task;
    const oldAny = (task || {}) as any;
    const newAny = dataToSave as any;
    const datesOrStatusChanged =
      isNewTask ||
      task?.start !== dataToSave.start ||
      task?.finish !== dataToSave.finish ||
      task?.runningStatus !== dataToSave.runningStatus ||
      task?.approval !== dataToSave.approval ||
      Boolean(oldAny.isCompleted) !== Boolean(newAny.isCompleted) ||
      (oldAny.status || '') !== (newAny.status || '') ||
      task?.projectCode !== dataToSave.projectCode ||
      task?.projectStore !== dataToSave.projectStore ||
      task?.projectYear !== dataToSave.projectYear ||
      depUpdates.length > 0;

    // Close modal INSTANTLY (0ms UI lag)
    onClose();
    setIsSubmitting(false);

    // Offload network commit, delta emission, and project date rollups to non-blocking background microtask
    setTimeout(async () => {
      try {
        const deltaItems = optimisticList.map(item => {
          const prev = tasks.find(t => t.id === item.id);
          return createPatchDeltaItem({
            docId: item.id,
            collectionName: 'taka_tasks',
            previous: prev,
            updated: item,
            action: prev ? 'update' : 'create',
            year: item.projectYear ? String(item.projectYear) : (task?.projectYear ? String(task.projectYear) : undefined),
          });
        });
        const { markExplicitDelta } = await import('../../lib/autoDeltaQueue');
        markExplicitDelta(deltaItems);
        await batch.commit();
        recordBatchDeltas(deltaItems, currentUser?.email || currentUser?.username).catch(() => {});

        if (datesOrStatusChanged) {
          const allTasksCurrent = useAppStore.getState().tasks;
          const targetProjId = dataToSave.projectId ? String(dataToSave.projectId) : (currentProject?.id ? String(currentProject.id) : undefined);
          syncProjectDatesByCode(dataToSave.projectCode, dataToSave.projectStore, dataToSave.projectYear, allTasksCurrent, targetProjId);
          if (task && (task.projectCode !== dataToSave.projectCode || task.projectStore !== dataToSave.projectStore || task.projectYear !== dataToSave.projectYear)) {
            const oldProjId = task.projectId ? String(task.projectId) : targetProjId;
            syncProjectDatesByCode(task.projectCode, task.projectStore, task.projectYear, allTasksCurrent, oldProjId);
          }
        }
      } catch (error: any) {
        console.error("Background task save failed:", error);
        useAppStore.getState().setToast({ title: "Save Failed", message: "Network error or permission denied. Data was not saved.", type: "error" });
      }
    }, 0);

    undoManager.push({
      description: task ? `Update Task: ${formData.name}` : `Create Task: ${formData.name}`,
      revert: async () => {
        const revertBatch = writeBatch(db);
        if (task) {
          revertBatch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', id), { ...task, updatedAt: new Date().toISOString() });
        } else {
          revertBatch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', id), { isDeleted: true, updatedAt: new Date().toISOString() }, { merge: true });
        }
        updatesToDependents.forEach(depTask => {
          const originalDep = tasks.find(tsk => tsk.id === depTask.id);
          if (originalDep) {
            revertBatch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', depTask.id), originalDep);
          }
        });
        predecessorUpdates.forEach(pu => {
          if (!mappedDepUpdates.has(pu.id)) {
             const originalTask = tasks.find(tk => tk.id === pu.id);
             if (originalTask) {
               revertBatch.update(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', originalTask.id), { predecessor: originalTask.predecessor || '' });
             }
          }
        });
        revertBatch.commit().catch(console.error);
      }
    });

    return true;
  } catch (e: any) { 
    console.error(e); 
    useAppStore.getState().setToast({ title: "Notice", message: String(e.message || "An error occurred while saving the task.") });
    setIsSubmitting(false);
    return false;
  }
}

import { Project, Task } from '../../types';
import { useAppStore } from '../../store/useAppStore';
import { removeCodeFromLinkCodesString, extractAllProjectCodes } from '../../lib/utils';
import { doc } from 'firebase/firestore';
import { writeBatch } from '../../lib/firestoreWrites';
import { db, APP_ID } from '../../lib/firebase';
import { undoManager } from '../../lib/undoManager';
import { getProjectCodeChangePredecessorUpdates } from '../../lib/ganttUtils';
import { recordBatchDeltas } from '../../lib/deltaSyncService';

interface HandleProjectIdentityChangeParams {
  project: Project;
  formData: Partial<Project>;
  dataToSave: any;
  docId: string;
  yearNum: number;
  projects: Project[];
  setToast: (toast: any) => void;
}

export async function handleProjectIdentityChange({
  project,
  formData,
  dataToSave,
  docId,
  yearNum,
  projects,
  setToast
}: HandleProjectIdentityChangeParams) {
  const batch = writeBatch(db);
  const projectRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_projects', docId);
  batch.set(projectRef, {
    ...dataToSave,
    updatedAt: new Date().toISOString()
  }, { merge: true });

  let rawTasks = useAppStore.getState().tasks || [];
  const oldPCode = String(project.code || '').trim().toLowerCase();
  const oldPYear = String(project.year || '');
  const oldPStore = String(project.store || '').trim().toLowerCase();

  let relatedTasks = rawTasks.filter(t => {
    const matchesCode = String(t.projectCode || '').trim().toLowerCase() === oldPCode;
    const matchesYear = !t.projectYear || !oldPYear || String(t.projectYear) === oldPYear;
    const matchesStore = !t.projectStore || !oldPStore || String(t.projectStore).trim().toLowerCase() === oldPStore;
    return matchesCode && matchesYear && matchesStore;
  });

  // If memory does not have tasks for this project (e.g. mobile lazy load), check Dexie first, then Firestore only if needed
  if (relatedTasks.length === 0 && project.code && (project.taskCount || (project.taskSummaries && project.taskSummaries.length > 0))) {
    try {
      const { ensureProjectTasksLoaded } = await import('../../lib/taskStateService');
      const loaded: Task[] = await ensureProjectTasksLoaded(project.code);
      if (loaded.length > 0) {
        relatedTasks = loaded.filter(t => {
          const matchesYear = !t.projectYear || !oldPYear || String(t.projectYear) === oldPYear;
          const matchesStore = !t.projectStore || !oldPStore || String(t.projectStore).trim().toLowerCase() === oldPStore;
          return matchesYear && matchesStore;
        });
        const existingIds = new Set(rawTasks.map(t => t.id));
        rawTasks = [...rawTasks, ...loaded.filter(t => !existingIds.has(t.id))];
        useAppStore.getState().setTasks(rawTasks);
      }
    } catch (err) {
      console.warn("Failed to fetch tasks for project identity change from local/cache:", err);
    }
  }

  const { taskUpdates, predecessorUpdates } = getProjectCodeChangePredecessorUpdates(
    project,
    String(formData.code).trim(),
    formData.year,
    formData.store,
    rawTasks,
    projects
  );

  const relatedTaskIds = new Set(relatedTasks.map(t => t.id));

  relatedTasks.forEach(t => {
    const tUpdate = taskUpdates.get(t.id) || {
      projectCode: String(formData.code).trim(),
      projectYear: yearNum,
      projectStore: formData.store
    };
    const pu = predecessorUpdates.find(p => p.id === t.id);
    const payload: any = {
      ...tUpdate,
      updatedAt: new Date().toISOString()
    };
    if (pu && pu.predecessor !== undefined) {
      payload.predecessor = pu.predecessor;
    }
    Object.keys(payload).forEach(k => {
      if (payload[k] === undefined) delete payload[k];
    });
    batch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', t.id), payload, { merge: true });
  });

  predecessorUpdates.forEach(pu => {
    if (!relatedTaskIds.has(pu.id)) {
      const payload: any = {
        predecessor: pu.predecessor,
        updatedAt: pu.updatedAt || new Date().toISOString()
      };
      Object.keys(payload).forEach(k => {
        if (payload[k] === undefined) delete payload[k];
      });
      batch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', pu.id), payload, { merge: true });
    }
  });

  const storeCosts = useAppStore.getState().costs || [];
  const relatedCosts = storeCosts.filter(c => {
    const matchesCode = String(c.projectCode || '').trim().toLowerCase() === oldPCode;
    const matchesYear = !c.projectYear || !oldPYear || String(c.projectYear) === oldPYear;
    const matchesStore = !c.projectStore || !oldPStore || String(c.projectStore).trim().toLowerCase() === oldPStore;
    return matchesCode && matchesYear && matchesStore;
  });

  relatedCosts.forEach(c => {
    const payload: any = {
      projectCode: String(formData.code).trim(),
      projectYear: yearNum,
      projectStore: formData.store,
      updatedAt: new Date().toISOString()
    };
    Object.keys(payload).forEach(k => {
      if (payload[k] === undefined) delete payload[k];
    });
    batch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_costs', c.id), payload, { merge: true });
  });

  const predMap = new Map(predecessorUpdates.map(pu => [pu.id, pu.predecessor]));
  const updatedTasks = rawTasks.map(t => {
    const tUpdate = taskUpdates.get(t.id);
    const newPred = predMap.get(t.id);
    if (tUpdate || newPred !== undefined) {
      return {
        ...t,
        ...(tUpdate || {}),
        ...(newPred !== undefined ? { predecessor: newPred, updatedAt: new Date().toISOString() } : {})
      };
    }
    return t;
  });
  useAppStore.getState().setTasks(updatedTasks);

  const rawProjects = useAppStore.getState().projects || [];
  const savedCode = String(formData.code).trim().toLowerCase();
  const savedYearStr = String(formData.year || '').trim();
  const savedStoreStr = String(formData.store || '').trim().toUpperCase();

  const updatedProjects = rawProjects.map(p => {
    if ((p.id && p.id === project.id) || (p.no !== undefined && project.no !== undefined && p.no === project.no)) {
      return { ...p, ...dataToSave, id: project.id || docId, updatedAt: new Date().toISOString() };
    }
    const pStore = String(p.store || '').trim().toUpperCase();
    const pYear = String(p.year || '').trim();
    if ((!savedStoreStr || !pStore || pStore === savedStoreStr) && (!savedYearStr || !pYear || pYear === savedYearStr)) {
      if (p.linkCodes) {
        const allCodes = extractAllProjectCodes(p);
        if (allCodes.includes(savedCode) && String(p.code || '').trim().toLowerCase() !== savedCode) {
          const cleaned = removeCodeFromLinkCodesString(p.linkCodes, formData.code);
          batch.set(doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_projects', p.id || String(p.no)), {
            linkCodes: cleaned,
            updatedAt: new Date().toISOString()
          }, { merge: true });
          return { ...p, linkCodes: cleaned, updatedAt: new Date().toISOString() };
        }
      }
    }
    return p;
  });
  useAppStore.getState().setProjects(updatedProjects);

  const identityDeltaItems = [
    { docId, collectionName: 'taka_projects' as const, action: 'update' as const, data: dataToSave, year: String(yearNum) },
    ...relatedTasks.map(t => ({
      docId: t.id,
      collectionName: 'taka_tasks' as const,
      action: 'update' as const,
      data: { projectCode: String(formData.code).trim(), projectYear: yearNum, projectStore: formData.store },
      year: String(yearNum)
    })),
    ...relatedCosts.map(c => ({
      docId: c.id,
      collectionName: 'taka_costs' as const,
      action: 'update' as const,
      data: { projectCode: String(formData.code).trim(), projectYear: yearNum, projectStore: formData.store },
      year: String(yearNum)
    }))
  ];
  import('../../lib/autoDeltaQueue')
    .then(({ markExplicitDelta }) => markExplicitDelta(identityDeltaItems))
    .catch(() => {});

  batch.commit().then(() => {
    recordBatchDeltas(identityDeltaItems).catch(console.error);
  }).catch(e => {
    console.error("Batch commit failed", e);
    setToast({ title: "Save Failed", message: "Network error or permission denied. Data was not saved.", type: "error" });
  });

  undoManager.push({
    description: `Update Identity: ${project.code}(${project.year}) -> ${formData.code}(${formData.year})`,
    revert: async () => {
      const revertBatch = writeBatch(db);
      revertBatch.set(projectRef, project, { merge: true });
      relatedTasks.forEach(t => {
        const tRef = doc(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks', t.id);
        revertBatch.set(tRef, {
          projectCode: project.code,
          projectYear: project.year,
          projectStore: project.store,
          ...(t.taskCodePhu !== undefined ? { taskCodePhu: t.taskCodePhu } : {}),
          ...(t.predecessor !== undefined ? { predecessor: t.predecessor } : {})
        }, { merge: true });
      });
      predecessorUpdates.forEach(pu => {
        if (!relatedTaskIds.has(pu.id)) {
          const originalTask = rawTasks.find(st => st.id === pu.id);

import { collection, query, where, limit, getDocsFromCache, getDocsFromServer } from 'firebase/firestore';

/**
 * Returns the standard notification date window:
 * - Past 7 days (to detect overdue/recent tasks)
 * - Future 14 days (to detect upcoming deadlines)
 */
export function getNotificationDateRange(referenceDate: Date = new Date()): { past7: string; future14: string } {
  const past7 = new Date(referenceDate.getTime() - 7 * 24 * 60 * 60 * 1000).toISOString().substring(0, 10);
  const future14 = new Date(referenceDate.getTime() + 14 * 24 * 60 * 60 * 1000).toISOString().substring(0, 10);
  return { past7, future14 };
}

/**
 * Filters a list of tasks for deadline notifications within the target window.
 */
export function filterTasksForNotificationWindow(
  tasks: Task[],
  past7: string,
  future14: string,
  maxCount: number = 50
): Task[] {
  if (!tasks || tasks.length === 0) return [];

  const matched = tasks.filter((t) => {
    if (t.isDeleted) return false;
    if (!t.finish) return false;
    const finishDate = t.finish.substring(0, 10);
    return finishDate >= past7 && finishDate <= future14;
  });

  matched.sort((a, b) => (a.finish || '').localeCompare(b.finish || ''));
  return matched.slice(0, maxCount);
}

/**
 * Loads tasks for notification bell / deadline monitoring.
 *
 * Tối ưu hóa Firestore Read (Phương pháp 1):
 * - Ưu tiên 1 (Cache-First): Đọc 100% từ Dexie Local-First DB (0ms, 0 Firestore Read).
 * - Ưu tiên 2 (Cold-Boot Fallback): Chỉ khi Dexie hoàn toàn rỗng (lần đầu vào app trên máy mới/Incognito),
 *   mới kích hoạt truy vấn 50 tasks từ Firestore, sau đó lưu ngay vào Dexie để lần sau tiếp tục đạt 0 Read.
 */
export async function loadNotificationTasks(options?: {
  referenceDate?: Date;
  maxCount?: number;
  skipFirestoreFallback?: boolean;
}): Promise<Task[]> {
  const { past7, future14 } = getNotificationDateRange(options?.referenceDate);
  const maxCount = options?.maxCount || 50;

  try {
    // 1. Kiểm tra dữ liệu trong Zustand RAM, Dexie Local DB và Project taskSummaries (0 Read Firestore)
    const { useAppStore } = await import('../store/useAppStore');
    const store = useAppStore.getState();
    const storeTasks = store.tasks || [];

    const summaryTasks: Task[] = [];
    if (store.projects) {
      store.projects.forEach((p) => {
        if (p.taskSummaries && Array.isArray(p.taskSummaries)) {
          p.taskSummaries.forEach((ts) => {
            if (ts && ts.id) {
              summaryTasks.push({
                id: ts.id,
                name: ts.name || '',
                start: ts.start || '',
                finish: ts.finish || '',
                status: ts.status || 'On process',
                projectCode: p.code,
                projectStore: p.store,
                projectYear: p.year,
                party: ts.party,
              } as unknown as Task);
            }
          });
        }
      });
    }

    const localTasks = await getLocalTasks();
    const combinedMap = new Map<string, Task>();

    summaryTasks.forEach((t) => combinedMap.set(t.id, t));
    localTasks.forEach((t) => combinedMap.set(t.id, t));
    storeTasks.forEach((t) => combinedMap.set(t.id, t));

    const combinedPool = Array.from(combinedMap.values());

    if (combinedPool.length > 0) {
      const filtered = filterTasksForNotificationWindow(combinedPool, past7, future14, maxCount);

      usePerformanceStore
        .getState()
        .trackGetDoc(
          'local_notif_tasks',
          true,
          filtered.length,
          `Notification tasks from RAM/Dexie/Summaries (0 Firestore Read, pool: ${combinedPool.length})`
        );

      return filtered;
    }

    // 2. Fallback: Nếu hoàn toàn rỗng (Cold Boot / Incognito) và chưa bị vô hiệu hóa
    if (options?.skipFirestoreFallback) {
      return [];
    }

    const qNotifTasks = query(
      collection(db, 'artifacts', APP_ID, 'public', 'data', 'taka_tasks'),
      where('finish', '>=', past7),
      where('finish', '<=', future14),
      limit(maxCount)
    );

    let snapshot;
    try {
      snapshot = await getDocsFromCache(qNotifTasks);
      if (snapshot.empty) {
        snapshot = await getDocsFromServer(qNotifTasks);
      }
    } catch {
      snapshot = await getDocsFromServer(qNotifTasks);
    }

    const loadedNotifTasks: Task[] = [];
    snapshot.forEach((docSnap) => {
      const data = { id: docSnap.id, ...docSnap.data() } as Task;
      if (!data.isDeleted) {
        loadedNotifTasks.push(data);
      }
    });

    // Lưu vào Dexie để các lần kiểm tra tiếp theo không phải đọc Firestore nữa
    if (loadedNotifTasks.length > 0) {
      bulkPersistTasks(loadedNotifTasks).catch((e) => {
        console.warn('[notificationTaskService] Error caching notif tasks to Dexie:', e);
      });
    }

    usePerformanceStore
      .getState()
      .trackGetDoc(
        'firestore_notif_tasks_fallback',
        false,
        loadedNotifTasks.length,
        'Cold-boot notification tasks loaded from Firestore and cached to Dexie'
      );

    return loadedNotifTasks;
  } catch (err) {
    console.warn('[notificationTaskService] Failed to load notification tasks:', err);
    return [];
  }
}

const PREFIX = 'loaded:';
export type MarkerKind =
  | 'projects_year'
  | 'tasks_proj'
  | 'tasks_year'
  | 'costs_year'
  | 'delegations_year';

let cache: Set<string> | null = null;
let loading: Promise<Set<string>> | null = null;

export const markerKey = (kind: MarkerKind, id: string | number) =>
  `${PREFIX}${kind}:${String(id).trim().toLowerCase()}`;

export async function getLoadedMarkers(): Promise<Set<string>> {
  if (cache) return cache;
  if (!loading) {
    loading = (async () => {
      try {
        const rows = await localDb.syncMeta.where('key').startsWith(PREFIX).toArray();
        cache = new Set(rows.map((r) => r.key));
      } catch {
        cache = new Set();
      }
      return cache as Set<string>;
    })();
  }
  return loading;
}

export function hasMarker(set: Set<string>, kind: MarkerKind, id: string | number): boolean {
  return set.has(markerKey(kind, id));
}

export async function markLoaded(kind: MarkerKind, id: string | number): Promise<void> {
  const key = markerKey(kind, id);
  const currentCache = await getLoadedMarkers();
  currentCache.add(key);
  try {
    await localDb.syncMeta.put({ key, value: String(Date.now()), updatedAt: Date.now() });
  } catch {}
}

export async function clearLoadedMarkers(): Promise<void> {
  cache = new Set();
  loading = null;
  try {
    await localDb.syncMeta.where('key').startsWith(PREFIX).delete();
  } catch {}
}

export async function clearAllLoadedMarkers(): Promise<void> {
  return clearLoadedMarkers();
}

  const setCloudStatus = useAppStore((state) => state.setCloudStatus);

  const projectsPool = useRef<Map<string, () => void>>(new Map());
  const prevSigProjects = useRef<string>("");
  const remoteYearsMetaRef = useRef<YearsMetaMap>({});

  // Atomic batching buffers for Projects and Tasks updates to prevent JS main thread freezing
  const pendingProjectsMap = useRef<Map<string, Project> | null>(null);
  const pendingProjectsChanged = useRef<Map<string, Project>>(new Map());
  const pendingProjectsRemoved = useRef<Set<string>>(new Set());
  const pendingDeletedTaskIds = useRef<Set<string>>(new Set());
  const projectsBatchTimer = useRef<NodeJS.Timeout | null>(null);

  const batchProjectsUpdate = useCallback((docChanges: any[], fromCache: boolean = false) => {
    if (!pendingProjectsMap.current) {
      const currentProjects = useAppStore.getState().projects || [];
      const map = new Map<string, Project>();
      currentProjects.forEach(p => map.set(p.id, p));
      pendingProjectsMap.current = map;
    }

    docChanges.forEach((change) => {
      const id = change.doc.id;
      const rawDocData = change.doc.data();
      const data = { id, ...rawDocData } as Project;
      const isTrueDelete = Boolean(data.isDeleted || change.type === "removed");

      if (isTrueDelete) {
        pendingProjectsMap.current!.delete(id);
        pendingProjectsChanged.current.delete(id);
        pendingProjectsRemoved.current.add(id);
      } else {
        if (fromCache && isStaleCacheDoc(pendingProjectsMap.current!.get(id), data)) return;
        const existing = pendingProjectsMap.current!.get(id);

        let resolvedData = existing ? {
          ...data,
          taskCount: data.taskCount !== undefined ? data.taskCount : (existing.taskCount ?? 0),
          taskSummaries: data.taskSummaries !== undefined ? data.taskSummaries : (existing.taskSummaries || []),
        } : data;

        // RAM-truth Protection: Nếu RAM đang có nhiều task thật hơn summary từ server, giữ nguyên truth
        if (!fromCache && existing && Array.isArray(resolvedData.taskSummaries)) {
          const currentTasks = useAppStore.getState().tasks || [];
          const realProjTasks = currentTasks.filter(t => 
            t.projectId === id || 
            (t.projectCode === resolvedData.code && String(t.projectYear) === String(resolvedData.year) && String(t.projectStore).toLowerCase() === String(resolvedData.store).toLowerCase())
          );
          if (realProjTasks.length > resolvedData.taskSummaries.length) {
            // Keep existing summary if it was better or would be updated by self-healing later
            resolvedData.taskSummaries = existing.taskSummaries || resolvedData.taskSummaries;
            resolvedData.taskCount = Math.max(resolvedData.taskCount || 0, existing.taskCount || 0);
          }
        }

        pendingProjectsMap.current!.set(id, resolvedData);
        pendingProjectsChanged.current.set(id, resolvedData);
        pendingProjectsRemoved.current.delete(id);
      }
    });

    if (projectsBatchTimer.current) {
      clearTimeout(projectsBatchTimer.current);
    }

    projectsBatchTimer.current = setTimeout(() => {
      if (pendingProjectsMap.current) {
        const updatedList = Array.from(pendingProjectsMap.current.values());
        updatedList.sort((a, b) => (a.no || 0) - (b.no || 0));
        useAppStore.getState().setProjects(updatedList);

        if (pendingDeletedTaskIds.current.size > 0) {
          const delIdsLower = new Set(
            Array.from(pendingDeletedTaskIds.current).map((tid) => String(tid).trim().toLowerCase())
          );
          const delRawIds = Array.from(pendingDeletedTaskIds.current);
          pendingDeletedTaskIds.current.clear();

          const currentTasks = useAppStore.getState().tasks || [];
          const filteredTasks = currentTasks.filter(
            (t) => !t?.id || !delIdsLower.has(String(t.id).trim().toLowerCase())
          );
          if (filteredTasks.length !== currentTasks.length) {
            useAppStore.getState().setTasks(filteredTasks);
          }
          delRawIds.forEach((tid) => {
            deleteLocalRecord('taka_tasks', tid).catch(() => {});
          });
        }

        // Chỉ ghi Dexie phần thay đổi; xóa khỏi Dexie phần đã bị xóa (chống "sống lại")
        const changed = Array.from(pendingProjectsChanged.current.values());
        const removed = Array.from(pendingProjectsRemoved.current);
        pendingProjectsChanged.current.clear();
        pendingProjectsRemoved.current.clear();
        if (changed.length > 0) offloadPersistSnapshot('taka_projects', changed);
        removed.forEach((id) => { deleteLocalRecord('taka_projects', id).catch(() => {}); });
        const setNotifs = useAppStore.getState().setAllProjectsForNotifs;
        if (setNotifs) setNotifs(updatedList);
        pendingProjectsMap.current = null;
      }
    }, 40);
  }, []);

  // Stage 3 & Delta Pipeline Startup
  useEffect(() => {
    bootstrapLocalStore().catch(() => {});

    fetchYearsMeta().then((meta) => {
      remoteYearsMetaRef.current = meta;
      setMemoryYearsMeta(meta);
    });

    const stopDeltaSync = startDeltaSyncService();
    return () => {
      stopDeltaSync();
    };
  }, []);

  // Sleep/Wake Recovery Listener
  useEffect(() => {
    const handleSleepWakeRecovery = () => {
      // KHÔNG hủy listener khi thức giấc.
      // Hủy xong mà không effect nào tạo lại => mất realtime Projects, và bộ đếm activeListeners
      // không giảm => deltaInvalidateService tưởng doc đã có listener phủ rồi bỏ qua việc đọc lại.
      // Socket chết do iOS treo đã có initNetworkRecovery (syncResilience) xử lý.
      hydrateProjectsFromLocal(
        useAppStore.getState().globalYearFilter,
        useAppStore.getState().globalStoreFilter
      ).catch(() => {});
    };

    window.addEventListener("taka_sleep_wake_recovery", handleSleepWakeRecovery);
    return () => {
      window.removeEventListener("taka_sleep_wake_recovery", handleSleepWakeRecovery);
    };
  }, []);

  // Unmount cleanup
  useEffect(() => {
    return () => {
      projectsPool.current.forEach((u, key) => {
        u();
        usePerformanceStore.getState().unregisterListener(`taka_projects_${key}`);
      });
      projectsPool.current.clear();
      if (projectsBatchTimer.current) clearTimeout(projectsBatchTimer.current);
    };
  }, []);

  // Main Unified Global Delta Sync for Projects
  useEffect(() => {
    if (!querySignature) return;

    if (prevSigProjects.current !== querySignature) {
      projectsPool.current.forEach((u, key) => {
        u();
        usePerformanceStore.getState().unregisterListener(`taka_projects_${key}`);
      });
      projectsPool.current.clear();
      prevSigProjects.current = querySignature;
    }

    const isAllYears = globalYearFilter === "all" || globalYearFilter === "ALL";
    const liveTargetYears = isAllYears ? PRIMARY_ACTIVE_YEARS : [globalYearFilter];

    liveTargetYears.forEach(async (year) => {
      // LUÔN đảm bảo nạp lại dữ liệu từ Dexie Local-First khi người dùng đổi bộ lọc Year hoặc Store,
      // kể cả khi listener của năm đó đã nằm trong projectsPool!
      const localCount = await hydrateProjectsFromLocal(year, globalStoreFilter);

      if (!projectsPool.current.has(year)) {
        const isClosedOrPastYear = Number(year) < new Date().getFullYear();
        const hasLocalData =
          localCount > 0 ||
          (useAppStore.getState().projects || []).some((p) => String(p.year) === String(year));

        const deltaManaged =
          (hasLocalData && isClosedOrPastYear) ||
          (DELTA_FIRST.projects && hasLocalData && hasMarker(await getLoadedMarkers(), 'projects_year', year));
        if (deltaManaged) {
          projectsPool.current.set(year, () => {});
          pullAndApplyDeltas(year, true)
            .then(() => setCloudStatus("online"))
            .catch(() => {});
          usePerformanceStore
            .getState()
            .trackGetDoc(
              `taka_projects_${year}`,
              true,
              localCount,
              "Dexie Local-First Projects Hydration (0 Read full collection)"
            );
          return;
        }

        if (false && isClosedOrPastYear && isYearCacheValid(year, remoteYearsMetaRef.current)) {
          const cached = loadYearDataFromLocal(year);
          if (cached && cached.projects && cached.projects.length > 0) {
            batchProjectsUpdate(
              cached.projects.map((p) => ({
                type: "added",
                doc: { id: p.id, data: () => p },
              }))
            );
            usePerformanceStore
              .getState()
              .trackGetDoc(
                `taka_projects_${year}`,
                true,
                cached.projects.length,
                "Metadata Fingerprint Cache Hit"
              );
            projectsPool.current.set(year, () => {});
            return;
          }
        }

        const colRef = collection(db, "artifacts", APP_ID, "public", "data", "taka_projects");
        const qProjects = buildStoreAndYearQuery(colRef, "store", "year", year, userRole, allowedStores);

        usePerformanceStore.getState().registerListener(`taka_projects_${year}`);
        const unsub = onSnapshot(
          qProjects,
          (snapshot) => {
            usePerformanceStore.getState().trackSnapshot(`taka_projects_${year}`, snapshot);
            const fromCache = snapshot.metadata.fromCache;
            let changesToApply: any[] = snapshot.docChanges();

            // Server snapshot reconciliation: purge any zombie project in Dexie/Zustand for this year that no longer exists on server
            if (!fromCache) {
              const serverIds = new Set<string>();
              snapshot.docs.forEach((d) => serverIds.add(d.id));
              const currentProjectsNow = useAppStore.getState().projects || [];
              const nowMs = Date.now();
              const allowedLower = allowedStores.map((s) => s.toLowerCase());
              const isStoreRestricted = userRole !== "admin" && allowedLower.length > 0 && !allowedLower.includes("all");
              const zombieRemovals: any[] = [];
              currentProjectsNow.forEach((p) => {
                if (
                  p?.id &&
                  String(p.year ?? "").trim() === String(year).trim() &&
                  !serverIds.has(p.id)
                ) {
                  if (isStoreRestricted && p.store && !allowedLower.includes(String(p.store).trim().toLowerCase())) {
                    return;
                  }
                  const updatedMs = tsOf(p.updatedAt);
                  if (!updatedMs || nowMs - updatedMs > 30000) {
                    zombieRemovals.push({
                      type: "removed",
                      doc: { id: p.id, data: () => ({ ...p, isDeleted: true }) },
                    });
                  }
                }
              });
              if (zombieRemovals.length > 0) {
                changesToApply = [...changesToApply, ...zombieRemovals];
              }
            }

            batchProjectsUpdate(changesToApply, fromCache);
            setCloudStatus("online");

            if (DELTA_FIRST.projects && !fromCache) {
              markLoaded('projects_year', year).catch(() => {});
              const u = projectsPool.current.get(year);
              projectsPool.current.set(year, () => {});
              u?.();
              usePerformanceStore.getState().unregisterListener(`taka_projects_${year}`);
            }
          },
          (e) => {
            console.error("Projects listener error:", e);
            setCloudStatus("offline");
          }
        );
        projectsPool.current.set(year, unsub);
      }
    });

    if (isAllYears) {
      SECONDARY_ACTIVE_YEARS.forEach(async (year) => {
        if (projectsPool.current.has(year)) return;
        const secKey = `${querySignature}_${globalStoreFilter}_${year}`;
        if (secondaryYearsFetched.has(secKey)) return;
        secondaryYearsFetched.add(secKey);
        try {
          const colRef = collection(db, "artifacts", APP_ID, "public", "data", "taka_projects");
          const qProjects = buildStoreAndYearQuery(colRef, "store", "year", year, userRole, allowedStores);
          let snapshot;
          try {
            snapshot = await getDocsFromCache(qProjects);
            if (snapshot.empty) {
              snapshot = await getDocsFromServer(qProjects);
            }
          } catch {
            snapshot = await getDocsFromServer(qProjects);
          }

          if (snapshot && !snapshot.empty) {
            const currentProjects = useAppStore.getState().projects || [];
            const projectMap = new Map<string, Project>();
            currentProjects.forEach((p) => projectMap.set(p.id, p));

            snapshot.forEach((docSnap) => {
              const data = { id: docSnap.id, ...docSnap.data() } as Project;
              if (!data.isDeleted) {
                projectMap.set(docSnap.id, data);
              }
            });

            const updatedList = Array.from(projectMap.values());
            updatedList.sort((a, b) => (a.no || 0) - (b.no || 0));
            useAppStore.getState().setProjects(updatedList);
          }
        } catch {
          // Lỗi mạng hoặc cache: cho phép thử lại ở lần chạy sau
          secondaryYearsFetched.delete(secKey);
        }
      });
    }
  }, [querySignature, userRole, allowedStores, globalYearFilter, globalStoreFilter, batchProjectsUpdate, setCloudStatus]);
}

export function useTasksHydration({
  querySignature,
  userRole,
  allowedStores,
}: UseTasksHydrationProps) {
  const isTasksRequested = useAppStore((state) => state.isTasksRequested);
  const expandedProjectIds = useAppStore((state) => state.expandedProjectIds);
  const searchActiveProjectIds = useAppStore((state) => state.searchActiveProjectIds);
  const visibleActiveProjectIds = useAppStore((state) => state.visibleActiveProjectIds);
  const visibleActiveProjectIdsRef = useRef(visibleActiveProjectIds);
  visibleActiveProjectIdsRef.current = visibleActiveProjectIds;

  const globalYearFilter = useAppStore((state) => state.globalYearFilter);
  const setIsTasksLoading = useAppStore((state) => state.setIsTasksLoading);
  const setAllTasksForNotifs = useAppStore((state) => state.setAllTasksForNotifs);
  const setCloudStatus = useAppStore((state) => state.setCloudStatus);

  const tasksPool = useRef<Map<string, () => void>>(new Map());
  const tasksTTLTimeouts = useRef<Map<string, NodeJS.Timeout>>(new Map());
  const knownHydratedProjectCodes = useRef<Set<string>>(new Set());

  const prevQuerySigTasks = useRef<string>("");
  const prevSigTasks = useRef<string>("");

  // Atomic batching buffers for Tasks updates
  const pendingTasksMap = useRef<Map<string, Task> | null>(null);
  const pendingTasksChanged = useRef<Map<string, Task>>(new Map());
  const pendingTasksRemoved = useRef<Set<string>>(new Set());
  const tasksBatchTimer = useRef<NodeJS.Timeout | null>(null);

  const batchTasksUpdate = useCallback((docChanges: any[], fromCache: boolean = false) => {
    if (!pendingTasksMap.current) {
      const currentTasks = useAppStore.getState().tasks || [];
      const map = new Map<string, Task>();
      currentTasks.forEach((t) => map.set(t.id, t));
      pendingTasksMap.current = map;
    }

    docChanges.forEach((change) => {
      const id = change.doc.id;
      const rawDocData = change.doc.data();
      const data = { id, ...rawDocData } as Task;
      const isTrueDelete = Boolean(data.isDeleted || change.type === "removed");

      if (isTrueDelete) {
        markTasksDeleted([id], data.projectId ? String(data.projectId) : undefined);
        pendingTasksMap.current!.delete(id);
        pendingTasksChanged.current.delete(id);
        pendingTasksRemoved.current.add(id);
      } else {
        if (fromCache && isStaleCacheDoc(pendingTasksMap.current!.get(id), data)) return;
        if (!data.isDeleted) {
          const { unmarkTaskDeleted } = require("../../lib/taskDeletedTracker");
          unmarkTaskDeleted(id);
        }
        pendingTasksMap.current!.set(id, data);
        pendingTasksChanged.current.set(id, data);
        pendingTasksRemoved.current.delete(id);
      }
    });

    if (tasksBatchTimer.current) {
      clearTimeout(tasksBatchTimer.current);
    }

    tasksBatchTimer.current = setTimeout(() => {
      if (pendingTasksMap.current) {
        const rawList = Array.from(pendingTasksMap.current.values());
        const projects = useAppStore.getState().projects || [];
        const { dedupedTasks } = deduplicateTasksInMemory(rawList, projects);
        const { cleaned: unrolledTasks } = cleanOrphanTasks(dedupedTasks);
        const rolledUpTasks = rollupParentTaskDatesAndStatus(unrolledTasks, useAppStore.getState().delegationGroups);
        rolledUpTasks.sort((a, b) => {
          if (a.projectCode !== b.projectCode) {
            return (a.projectCode || "").localeCompare(b.projectCode || "");
          }
          return (a.orderIndex || 0) - (b.orderIndex || 0);
        });
        const finalTasks = rolledUpTasks;
        useAppStore.getState().setTasks(finalTasks);

        // Purging logic from taskSummaries removed to prevent data loss (User Requirement)
        if (pendingTasksRemoved.current.size > 0) {
           pendingTasksRemoved.current.clear();
        }
        
        // Self-healing an toàn:
        // - chỉ xét các dự án có task thay đổi trong lô này (không quét toàn bộ P x T)
        // - bỏ qua snapshot từ cache và 15 giây đầu sau khi mở app (dữ liệu cục bộ có thể cũ/thiếu)
        // - KHÔNG chữa theo hướng giảm: ít task hơn taskCount đã lưu nghĩa là có thể mới tải một phần
        if (!fromCache && (getAppBootAgeMs() >= BOOT_SYNC_GUARD_MS || knownHydratedProjectCodes.current.size > 0) && pendingTasksChanged.current.size > 0) {
          const touchedIds = new Set<string>();
          const touchedCodes = new Set<string>();
          pendingTasksChanged.current.forEach((t) => {
            if (t.projectId) touchedIds.add(String(t.projectId));
            if (t.projectCode) touchedCodes.add(String(t.projectCode).trim().toLowerCase());
          });

          const currentProjects = useAppStore.getState().projects || [];

          // Task thuộc dự án khi: đúng projectId; hoặc (task cũ không có projectId) khớp đủ ba trường store, year, code
          // (hoặc khớp vắt qua 2 năm nếu mã dự án p.code là duy nhất trong p.store)
          const belongsToProject = (t: Task, p: any, isUniqueCodeInStore: boolean): boolean => {
            if (t.projectId) return String(t.projectId) === String(p.id);
            const codeMatches = String(t.projectCode || "").trim().toLowerCase() === String(p.code || "").trim().toLowerCase();
            if (!codeMatches) return false;
            const storeMatches = !p.store || !t.projectStore || String(t.projectStore).trim().toLowerCase() === String(p.store).trim().toLowerCase();
            if (!storeMatches) return false;
            if (isUniqueCodeInStore) return true;
            return String(t.projectYear ?? "").trim() === String(p.year ?? "").trim();
          };

          let projectsNeedUpdate = false;
          const healedProjects = currentProjects.map((p) => {
            const codeKey = String(p.code || "").trim().toLowerCase();
            if (!touchedIds.has(String(p.id)) && !(codeKey && touchedCodes.has(codeKey))) return p;
            if (getAppBootAgeMs() < BOOT_SYNC_GUARD_MS && !knownHydratedProjectCodes.current.has(String(p.id))) return p;

            const sameCodeStoreProjects = currentProjects.filter(
              (proj) =>
                !proj.isDeleted &&
                String(proj.code || "").trim().toLowerCase() === codeKey &&
                (!p.store || !proj.store || String(proj.store).trim().toLowerCase() === String(p.store).trim().toLowerCase())
            );
            const isUniqueCodeInStore = sameCodeStoreProjects.length <= 1;

            const pTasks = finalTasks.filter((t) => !t.isDeleted && belongsToProject(t, p, isUniqueCodeInStore));
            if (pTasks.length === 0) return p;

            const existingMaxCount = Math.max(p.taskCount || 0, Array.isArray(p.taskSummaries) ? p.taskSummaries.length : 0);
            if (pTasks.length < existingMaxCount) return p;

            const isFullyHydratedProject =
              knownHydratedProjectCodes.current.has(String(p.id)) ||
              tasksPool.current.has(`proj_${p.id}`) ||
              tasksPool.current.has(`year_${p.year}`);

            if (!isFullyHydratedProject) return p;

            const computed = calculateProjectComputedDates(p, pTasks, useAppStore.getState().delegationGroups);
            const hasDiscrepancy =
              p.taskCount !== computed.taskCount ||
              !p.taskSummaries ||
              p.taskSummaries.length !== (computed.taskSummaries?.length || 0) ||
              (p.computedTakStart || '') !== (computed.computedTakStart || '') ||
              (p.computedTakCompl || '') !== (computed.computedTakCompl || '') ||
              (p.computedPlanProgress ?? 0) !== (computed.computedPlanProgress ?? 0) ||
              (p.computedActProgress ?? 0) !== (computed.computedActProgress ?? 0);
            if (!hasDiscrepancy) return p;

            projectsNeedUpdate = true;
            syncProjectDatesToFirestore(p.id, computed).catch(() => {});

            return {
              ...p,
              ...computed,
            };
          });
          if (projectsNeedUpdate) {
            useAppStore.getState().setProjects(healedProjects);
          }
        }
        
        const finalIds = new Set(finalTasks.map(t => t.id));
        const changed = Array.from(pendingTasksChanged.current.values()).filter(t => finalIds.has(t.id));
        const removed = Array.from(pendingTasksRemoved.current);
        pendingTasksChanged.current.clear();
        pendingTasksRemoved.current.clear();
        if (changed.length > 0) offloadPersistSnapshot('taka_tasks', changed);
        removed.forEach((id) => { deleteLocalRecord('taka_tasks', id).catch(() => {}); });

        const setNotifs = useAppStore.getState().setAllTasksForNotifs;
        if (setNotifs) setNotifs(finalTasks);
        pendingTasksMap.current = null;
      }
    }, 200);
  }, []);

  // Sleep/Wake Recovery Listener
  useEffect(() => {
    const handleSleepWakeRecovery = () => {
      // 1. KHÔNG hủy listener khi thức giấc.
      // Hủy xong mà không effect nào tạo lại => mất realtime tasks/costs/delegations
      // Socket chết do iOS treo đã có initNetworkRecovery (syncResilience) xử lý bằng deltaResync
      
      if (useAppStore.getState().isTasksRequested) {
        hydrateTasksFromLocal(useAppStore.getState().globalYearFilter).catch(() => {});
      }
    };

    window.addEventListener("taka_sleep_wake_recovery", handleSleepWakeRecovery);
    return () => {
      window.removeEventListener("taka_sleep_wake_recovery", handleSleepWakeRecovery);
    };
  }, []);

  // Cleanup on unmount
  useEffect(() => {
    return () => {
      tasksPool.current.forEach((u) => u());
      tasksPool.current.clear();
      tasksTTLTimeouts.current.forEach((t) => clearTimeout(t));
      tasksTTLTimeouts.current.clear();
      if (tasksBatchTimer.current) clearTimeout(tasksBatchTimer.current);
    };
  }, []);

  // Lightweight Notification Query for Tasks (0 Read from Dexie with Safe Fallback)
  useEffect(() => {
    let isCancelled = false;
    let intervalId: NodeJS.Timeout | null = null;

    if (!querySignature || isTasksRequested) {
      return;
    }

    const fetchNotifs = async () => {
      if (isCancelled) return;
      try {
        const loadedNotifTasks = await loadNotificationTasks();
        if (isCancelled) return;
        setAllTasksForNotifs(loadedNotifTasks);
      } catch (e) {
        console.warn("[useTasksHydration] Notification lightweight tasks query error:", e);
      }
    };

    fetchNotifs();

    const POLL_INTERVAL_MS = 4 * 60 * 60 * 1000; // 4 hours
    intervalId = setInterval(fetchNotifs, POLL_INTERVAL_MS);

    return () => {
      isCancelled = true;
      if (intervalId) clearInterval(intervalId);
    };
  }, [querySignature, isTasksRequested, setAllTasksForNotifs]);

  // Main Tasks Sync Logic
  useEffect(() => {
    if (!querySignature) return;

    const activeProjectIds = isTasksRequested
      ? [...new Set([...expandedProjectIds, ...searchActiveProjectIds])]
      : [];
    const currentSig = `${querySignature}_${activeProjectIds.join(",")}_${isTasksRequested}`;

    if (prevQuerySigTasks.current !== querySignature) {
      if (prevQuerySigTasks.current !== "") {
        tasksPool.current.forEach((u) => u());
        tasksPool.current.clear();
        tasksTTLTimeouts.current.forEach((t) => clearTimeout(t));
        tasksTTLTimeouts.current.clear();
      }
      prevQuerySigTasks.current = querySignature;
    }
    prevSigTasks.current = currentSig;

    if (activeProjectIds.length === 0) {
      Array.from(tasksPool.current.keys()).forEach((key) => {
        if (!tasksTTLTimeouts.current.has(key)) {
          const timeoutId = setTimeout(() => {
            const unsub = tasksPool.current.get(key);
            if (unsub) unsub();
            tasksPool.current.delete(key);
            tasksTTLTimeouts.current.delete(key);
            usePerformanceStore.getState().unregisterListener(`taka_tasks_${key}`);
          }, TTL_DURATION_MS);
          tasksTTLTimeouts.current.set(key, timeoutId);
        }
      });
      setIsTasksLoading(false);
      return;
    }

    let isCancelled = false;
    const runTasksSync = async () => {
      const currentProjects = useAppStore.getState().projects || [];
      const keysToHydrate = extractKeysToHydrate(activeProjectIds, currentProjects);

      await Promise.all([
        hydrateTasksForProjects(Array.from(keysToHydrate)),
        hydrateDelegationsForProjects(activeProjectIds),
      ]);
      if (isCancelled) return;
      const markers = await getLoadedMarkers();

      setIsTasksLoading(true);

      const projectsByYear = groupActiveProjectsByYear(activeProjectIds, currentProjects, globalYearFilter);

      const neededKeys = new Set<string>();
      let needTasksDeltaSync = false;

      projectsByYear.forEach((projs, year) => {
        if (projs.length > 10) {
          const yearKey = `year_${year}`;
          neededKeys.add(yearKey);

          if (tasksTTLTimeouts.current.has(yearKey)) {
            clearTimeout(tasksTTLTimeouts.current.get(yearKey)!);
            tasksTTLTimeouts.current.delete(yearKey);
          }

          projs.forEach((p) => {
            const pKey = `proj_${p.id}`;
            if (tasksPool.current.has(pKey)) {
              const u = tasksPool.current.get(pKey);
              if (u) u();
              tasksPool.current.delete(pKey);
              usePerformanceStore.getState().unregisterListener(`taka_tasks_${pKey}`);
            }
          });

          if (!tasksPool.current.has(yearKey)) {
            const currentTasks = useAppStore.getState().tasks || [];
            const hasLocalYearTasks = currentTasks.some(
              (t) => String(t.projectYear || "") === String(year)
            );
            const isClosedOrPastYear = Number(year) < new Date().getFullYear();

            const deltaManaged =
              (hasLocalYearTasks && isClosedOrPastYear) ||
              (DELTA_FIRST.tasks && hasLocalYearTasks && hasMarker(markers, 'tasks_year', year));
            if (deltaManaged) {
              tasksPool.current.set(yearKey, () => {});
              setIsTasksLoading(false);
              needTasksDeltaSync = true;
              usePerformanceStore
                .getState()
                .trackGetDoc(
                  `taka_tasks_${yearKey}`,
                  true,
                  currentTasks.length,
                  "Dexie Local-First Tasks (0 Read full collection)"
                );
            } else {
              const colRef = collection(db, "artifacts", APP_ID, "public", "data", "taka_tasks");
              const qTasks = buildStoreAndYearQuery(
                colRef,
                "projectStore",
                "projectYear",
                year,
                userRole,
                allowedStores
              );

              usePerformanceStore.getState().registerListener(`taka_tasks_${yearKey}`);
              const unsub = onSnapshot(
                qTasks,
                (snapshot) => {
                  usePerformanceStore.getState().trackSnapshot(`taka_tasks_${yearKey}`, snapshot);
                  setIsTasksLoading(false);
                  const fromCache = snapshot.metadata.fromCache;
                  let changesToApply: any[] = snapshot.docChanges();
                  batchTasksUpdate(changesToApply, fromCache);
                  setCloudStatus("online");

                  if (DELTA_FIRST.tasks && !fromCache) {
                    markLoaded('tasks_year', year).catch(() => {});
                    const u = tasksPool.current.get(yearKey);
                    tasksPool.current.set(yearKey, () => {});
                    u?.();
                    usePerformanceStore.getState().unregisterListener(`taka_tasks_${yearKey}`);
                  }
                },
                (e) => {
                  setIsTasksLoading(false);
                  console.error("Tasks consolidated listener error:", e);
                  setCloudStatus("offline");
                }
              );
              tasksPool.current.set(yearKey, unsub);
            }
          }
        } else {
          const visibleSet = new Set((visibleActiveProjectIdsRef.current || []).map((id) => String(id)));
          const sortedProjs = [...projs].sort((a, b) => {
            const aVis = visibleSet.has(a.id) || visibleSet.has(a.code) ? 1 : 0;
            const bVis = visibleSet.has(b.id) || visibleSet.has(b.code) ? 1 : 0;
            return bVis - aVis;
          });

          sortedProjs.forEach((p) => {
            const projKey = `proj_${p.id}`;
            const yearKey = `year_${year}`;

            if (tasksPool.current.has(yearKey)) {
              neededKeys.add(yearKey);
              return;
            }

            neededKeys.add(projKey);

            if (tasksTTLTimeouts.current.has(projKey)) {
              clearTimeout(tasksTTLTimeouts.current.get(projKey)!);
              tasksTTLTimeouts.current.delete(projKey);
            }

            if (!tasksPool.current.has(projKey)) {
              const currentTasks = useAppStore.getState().tasks || [];
              const localProjTasks = currentTasks.filter((t) => doesTaskBelongToProject(t, p));
              const hasLocalProjTasks =
                localProjTasks.length > 0 ||
                knownHydratedProjectCodes.current.has(String(p.id));
              const isClosedOrPastYear = Number(year) < new Date().getFullYear();
              const isZeroTaskProject =
                (p.taskCount || 0) === 0 &&
                Array.isArray(p.taskSummaries) &&
                p.taskSummaries.length === 0 &&
                localProjTasks.length === 0;

              const deltaManaged =
                isZeroTaskProject ||
                (hasLocalProjTasks && isClosedOrPastYear) ||
                (DELTA_FIRST.tasks && hasLocalProjTasks && (hasMarker(markers, 'tasks_proj', p.id) || hasMarker(markers, 'tasks_proj', p.code) || hasMarker(markers, 'tasks_year', year)));
              if (deltaManaged) {
                tasksPool.current.set(projKey, () => {});
                setIsTasksLoading(false);
                if (!isZeroTaskProject) needTasksDeltaSync = true;
                usePerformanceStore
                  .getState()
                  .trackGetDoc(
                    `taka_tasks_${projKey}`,
                    true,
                    localProjTasks.length,
                    "Dexie Local-First Project Tasks (0 Read)"
                  );
              } else {
                const codeList = [p.code];
                if (p.linkCodes) {
                  const extra = String(p.linkCodes).split(/[,;/]+/).map(s => s.trim()).filter(Boolean);
                  extra.forEach(c => { if (!codeList.includes(c)) codeList.push(c); });
                }
                const qTasks = codeList.length > 1
                  ? query(
                      collection(db, "artifacts", APP_ID, "public", "data", "taka_tasks"),
                      where("projectCode", "in", codeList.slice(0, 10))
                    )
                  : query(
                      collection(db, "artifacts", APP_ID, "public", "data", "taka_tasks"),
                      where("projectCode", "==", p.code)
                    );

                usePerformanceStore.getState().registerListener(`taka_tasks_${projKey}`);
                const unsub = onSnapshot(
                  qTasks,
                  (snapshot) => {
                    knownHydratedProjectCodes.current.add(String(p.id));
                    usePerformanceStore.getState().trackSnapshot(`taka_tasks_${projKey}`, snapshot);
                    setIsTasksLoading(false);
                    const fromCache = snapshot.metadata.fromCache;
                    const allProjectsNow = useAppStore.getState().projects || [];
                    const sameCodeStoreProjects = allProjectsNow.filter(
                      (proj) =>
                        !proj.isDeleted &&
                        String(proj.code || "").trim().toLowerCase() === String(p.code || "").trim().toLowerCase() &&
                        (!p.store || !proj.store || String(proj.store).trim().toLowerCase() === String(p.store).trim().toLowerCase())
                    );
                    const isUniqueCodeInStore = sameCodeStoreProjects.length <= 1;

                    let compatibleChanges = snapshot.docChanges().filter((ch) => {
                      const dt: any = ch.doc.data();
                      if (dt && dt.projectId && String(dt.projectId) !== String(p.id)) return false;
                      if (dt && dt.projectStore && p.store &&
                          String(dt.projectStore).trim().toLowerCase() !== String(p.store).trim().toLowerCase()) return false;
                      if (ch.type === "removed") return true;
                      if (dt && dt.projectId && String(dt.projectId) === String(p.id)) return true;
                      if (!isUniqueCodeInStore) {
                        const dy = dt.projectYear;
                        if (dy !== undefined && dy !== null && dy !== "" && p.year &&
                            String(dy).trim() !== String(p.year).trim()) return false;
                      }
                      return true;
                    });

                    batchTasksUpdate(compatibleChanges, fromCache);
                    setCloudStatus("online");

                    if (DELTA_FIRST.tasks && !fromCache) {
                      markLoaded('tasks_proj', p.id).catch(() => {});
                      const u = tasksPool.current.get(projKey);
                      tasksPool.current.set(projKey, () => {});
                      u?.();
                      usePerformanceStore.getState().unregisterListener(`taka_tasks_${projKey}`);
                    }
                  },
                  (e) => {
                    setIsTasksLoading(false);
                    console.error("Tasks listener error:", e);
                    setCloudStatus("offline");
                  }
                );
                tasksPool.current.set(projKey, unsub);
              }
            }
          });
        }
      });

      if (needTasksDeltaSync) {
        pullAndApplyDeltas(undefined, true)
          .then(() => setCloudStatus("online"))
          .catch(() => {});
      }

      Array.from(tasksPool.current.keys()).forEach((key) => {
        if (!neededKeys.has(key)) {
          if (!tasksTTLTimeouts.current.has(key)) {
            const timeoutId = setTimeout(() => {
              const unsub = tasksPool.current.get(key);
              if (unsub) unsub();
              tasksPool.current.delete(key);
              tasksTTLTimeouts.current.delete(key);
              usePerformanceStore.getState().unregisterListener(`taka_tasks_${key}`);
            }, TTL_DURATION_MS);
            tasksTTLTimeouts.current.set(key, timeoutId);
          }
        }
      });
    };

    runTasksSync().catch((err) => {
      console.warn("[useTasksHydration] runTasksSync error:", err);
    });

    return () => {
      isCancelled = true;
    };
  }, [
    querySignature,
    userRole,
    allowedStores,
    isTasksRequested,
    expandedProjectIds,
    searchActiveProjectIds,
    globalYearFilter,
    batchTasksUpdate,
    setIsTasksLoading,
    setCloudStatus,
  ]);
}

import { useEffect, useRef, useCallback } from "react";
import { collection, onSnapshot, getDocsFromCache, getDocsFromServer } from "firebase/firestore";
import { useAppStore } from "../../store/useAppStore";
import { usePerformanceStore } from "../../store/usePerformanceStore";
import { db, APP_ID } from "../../lib/firebase";
import { Cost } from "../../types";
import { PRIMARY_ACTIVE_YEARS, SECONDARY_ACTIVE_YEARS } from "../../lib/utils";
import { buildStoreAndYearQuery } from "./firebaseQueryHelpers";
import { hydrateCostsFromLocal, offloadPersistSnapshot } from "../../lib/backgroundSyncWorker";
import { deleteLocalRecord } from "../../lib/localDb";
import { pullAndApplyDeltas } from "../../lib/deltaSyncService";
import { getLoadedMarkers, hasMarker, markLoaded } from "../../lib/loadedMarkers";

// Delta-first: sau lần tải đầy đủ đầu tiên, hủy onSnapshot lớn, dựa vào Dexie + kênh delta.
// Bật TỪNG CÁI sau khi đã test. Mặc định tắt = hành vi hiện tại.
const DELTA_FIRST = { projects: true, tasks: true, costs: true, delegations: true };


interface UseCostsHydrationProps {
  querySignature: string | null;
  userRole?: string;
  allowedStores: string[];
}

const TTL_DURATION_MS = 15 * 60 * 1000; // 15 minutes

// Chuyển updatedAt (ISO string / Firestore Timestamp / number) về ms
const tsOf = (v: any): number => {
  if (!v) return 0;
  const n =
    typeof v === "string" ? Date.parse(v)
    : typeof v?.toMillis === "function" ? v.toMillis()
    : Number(v);
  return Number.isFinite(n) ? n : 0;
};
// Snapshot lấy từ cache cục bộ có thể cũ hơn dữ liệu đã nhận qua delta -> không cho ghi đè.
// Chỉ áp dụng cho snapshot fromCache; snapshot từ server luôn được tin (tránh lệch đồng hồ)
const isStaleCacheDoc = (existing: any, incoming: any): boolean => {
  if (!existing) return false;
  const e = tsOf(existing.updatedAt);
  const i = tsOf(incoming?.updatedAt);
  return e > 0 && i > 0 && i < e;
};

export function useCostsHydration({
  querySignature,
  userRole,
  allowedStores,
}: UseCostsHydrationProps) {
  const isCostsRequested = useAppStore((state) => state.isCostsRequested);
  const globalYearFilter = useAppStore((state) => state.globalYearFilter);
  const setIsCostsLoading = useAppStore((state) => state.setIsCostsLoading);
  const setCloudStatus = useAppStore((state) => state.setCloudStatus);

  const costsPool = useRef<Map<string, () => void>>(new Map());
  const costsTTLTimeouts = useRef<Map<string, NodeJS.Timeout>>(new Map());
  const prevSigCosts = useRef<string>("");

  // Atomic batching buffers for Costs updates
  const pendingCostsMap = useRef<Map<string, Cost> | null>(null);
  const pendingCostsChanged = useRef<Map<string, Cost>>(new Map());
  const pendingCostsRemoved = useRef<Set<string>>(new Set());
  const costsBatchTimer = useRef<NodeJS.Timeout | null>(null);

  const batchCostsUpdate = useCallback((docChanges: any[], fromCache: boolean = false) => {
    if (!pendingCostsMap.current) {
      const currentCosts = useAppStore.getState().costs || [];
      const map = new Map<string, Cost>();
      currentCosts.forEach((c) => map.set(c.id, c));
      pendingCostsMap.current = map;
    }

    docChanges.forEach((change) => {
      const id = change.doc.id;
      const rawDocData = change.doc.data();
      const data = { id, ...rawDocData } as Cost;
      const isTrueDelete = Boolean(data.isDeleted || change.type === "removed");

      if (isTrueDelete) {
        pendingCostsMap.current!.delete(id);
        pendingCostsChanged.current.delete(id);
        pendingCostsRemoved.current.add(id);
      } else {
        if (fromCache && isStaleCacheDoc(pendingCostsMap.current!.get(id), data)) return;
        pendingCostsMap.current!.set(id, data);
        pendingCostsChanged.current.set(id, data);
        pendingCostsRemoved.current.delete(id);
      }
    });

    if (costsBatchTimer.current) {
      clearTimeout(costsBatchTimer.current);
    }

    costsBatchTimer.current = setTimeout(() => {
      if (pendingCostsMap.current) {
        const updatedList = Array.from(pendingCostsMap.current.values());
        useAppStore.getState().setCosts(updatedList);
        
        const changed = Array.from(pendingCostsChanged.current.values());
        const removed = Array.from(pendingCostsRemoved.current);
        pendingCostsChanged.current.clear();
        pendingCostsRemoved.current.clear();
        
        if (changed.length > 0) offloadPersistSnapshot('taka_costs', changed);
        removed.forEach((id) => { deleteLocalRecord('taka_costs', id).catch(() => {}); });
        
        pendingCostsMap.current = null;
      }
    }, 80);
  }, []);

  // Sleep/Wake Recovery Listener
  useEffect(() => {
    const handleSleepWakeRecovery = () => {
      // 1. KHÔNG hủy listener khi thức giấc.
      
      if (useAppStore.getState().isCostsRequested) {
        hydrateCostsFromLocal(useAppStore.getState().globalYearFilter).catch(() => {});
      }
    };

    window.addEventListener("taka_sleep_wake_recovery", handleSleepWakeRecovery);
    return () => {
      window.removeEventListener("taka_sleep_wake_recovery", handleSleepWakeRecovery);
    };
  }, []);

  // Cleanup on unmount
  useEffect(() => {
    return () => {
      costsPool.current.forEach((u, key) => {
        u();
        usePerformanceStore.getState().unregisterListener(`taka_costs_${key}`);
      });
      costsPool.current.clear();
      costsTTLTimeouts.current.forEach((t) => clearTimeout(t));
      costsTTLTimeouts.current.clear();
      if (costsBatchTimer.current) clearTimeout(costsBatchTimer.current);
    };
  }, []);

  useEffect(() => {
    if (!querySignature || !isCostsRequested) {
      if (!isCostsRequested) {
        setIsCostsLoading(false);
        Array.from(costsPool.current.keys()).forEach((key) => {
          if (!costsTTLTimeouts.current.has(key)) {
            const timeoutId = setTimeout(() => {
              const unsub = costsPool.current.get(key);
              if (unsub) unsub();
              costsPool.current.delete(key);
              costsTTLTimeouts.current.delete(key);
              usePerformanceStore.getState().unregisterListener(`taka_costs_${key}`);
            }, TTL_DURATION_MS);
            costsTTLTimeouts.current.set(key, timeoutId);
          }
        });
      }
      return;
    }

    costsTTLTimeouts.current.forEach((t) => clearTimeout(t));
    costsTTLTimeouts.current.clear();

    if (prevSigCosts.current !== querySignature) {
      costsPool.current.forEach((u, key) => {
        u();
        usePerformanceStore.getState().unregisterListener(`taka_costs_${key}`);
      });
      costsPool.current.clear();
      prevSigCosts.current = querySignature;
    }

    const isAllYears = globalYearFilter === "all" || globalYearFilter === "ALL";
    const liveTargetYears = isAllYears ? PRIMARY_ACTIVE_YEARS : [globalYearFilter];

    setIsCostsLoading(true);

    liveTargetYears.forEach(async (year) => {
      if (!costsPool.current.has(year)) {
        const localCount = await hydrateCostsFromLocal(year);
        const currentCosts = useAppStore.getState().costs || [];
        const hasLocalCosts =
          localCount > 0 || currentCosts.some((c) => String(c.projectYear || "") === String(year));
        const isClosedOrPastYear = Number(year) < new Date().getFullYear();

        const deltaManaged =
          (hasLocalCosts && isClosedOrPastYear) ||
          (DELTA_FIRST.costs && hasLocalCosts && hasMarker(await getLoadedMarkers(), 'costs_year', year));
        if (deltaManaged) {
          costsPool.current.set(year, () => {});
          setIsCostsLoading(false);
          pullAndApplyDeltas(year, true)
            .then(() => setCloudStatus("online"))
            .catch(() => {});
          usePerformanceStore
            .getState()
            .trackGetDoc(
              `taka_costs_${year}`,
              true,
              localCount,
              "Dexie Local-First Costs (0 Read full collection)"
            );
          return;
        } else {
          const colRef = collection(db, "artifacts", APP_ID, "public", "data", "taka_costs");
          const qCosts = buildStoreAndYearQuery(
            colRef,
            "projectStore",
            "projectYear",
            year,
            userRole,
            allowedStores
          );

          usePerformanceStore.getState().registerListener(`taka_costs_${year}`);
          const unsub = onSnapshot(
            qCosts,
            (snapshot) => {
              usePerformanceStore.getState().trackSnapshot(`taka_costs_${year}`, snapshot);
              setIsCostsLoading(false);
              const fromCache = snapshot.metadata.fromCache;
              batchCostsUpdate(snapshot.docChanges(), fromCache);

              if (DELTA_FIRST.costs && !fromCache) {
                markLoaded('costs_year', year).catch(() => {});
                const u = costsPool.current.get(year);
                costsPool.current.set(year, () => {});
                u?.();
                usePerformanceStore.getState().unregisterListener(`taka_costs_${year}`);
              }
            },
            (e) => {
              setIsCostsLoading(false);
              console.error("Costs listener error:", e);
            }
          );
          costsPool.current.set(year, unsub);
        }
      } else {
        setIsCostsLoading(false);
      }
    });

    if (isAllYears) {
      SECONDARY_ACTIVE_YEARS.forEach(async (year) => {
        if (costsPool.current.has(year)) return;
        try {
          const colRef = collection(db, "artifacts", APP_ID, "public", "data", "taka_costs");
          const qCosts = buildStoreAndYearQuery(
            colRef,
            "projectStore",
            "projectYear",
            year,
            userRole,
            allowedStores
          );
          let snapshot;
          try {
            snapshot = await getDocsFromCache(qCosts);
            if (snapshot.empty) {
              snapshot = await getDocsFromServer(qCosts);
            }
          } catch {
            snapshot = await getDocsFromServer(qCosts);
          }

          if (snapshot && !snapshot.empty) {
            const liveCosts = useAppStore.getState().costs || [];
            const costMap = new Map<string, Cost>();
            liveCosts.forEach((c) => costMap.set(c.id, c));

            snapshot.forEach((docSnap) => {
              const data = { id: docSnap.id, ...docSnap.data() } as Cost;
              if (!data.isDeleted) {
                costMap.set(docSnap.id, data);
              }
            });

            const updatedList = Array.from(costMap.values());
            useAppStore.getState().setCosts(updatedList);
          }
        } catch {}
      });
    }
  }, [
    querySignature,
    userRole,
    allowedStores,
    isCostsRequested,
    globalYearFilter,
    setIsCostsLoading,
    setCloudStatus,
  ]);
}

  const isDelegationsRequested = useAppStore((state) => state.isDelegationsRequested);
  const globalYearFilter = useAppStore((state) => state.globalYearFilter);
  const setIsDelegationsLoading = useAppStore((state) => state.setIsDelegationsLoading);
  const setCloudStatus = useAppStore((state) => state.setCloudStatus);

  const delegationsPool = useRef<Map<string, () => void>>(new Map());
  const delegationsTTLTimeouts = useRef<Map<string, NodeJS.Timeout>>(new Map());
  const yearDelegationsCache = useRef<Map<string, Map<string, DelegationGroup>>>(new Map());

  const prevQuerySigDelegations = useRef<string>("");
  const prevSigDelegations = useRef<string>("");

  // Atomic batching buffers for Delegations updates
  const pendingDelegationsMap = useRef<Map<string, DelegationGroup> | null>(null);
  const pendingDelegationsChanged = useRef<Map<string, DelegationGroup>>(new Map());
  const pendingDelegationsRemoved = useRef<Set<string>>(new Set());
  const delegationsBatchTimer = useRef<NodeJS.Timeout | null>(null);

  const batchDelegationsUpdate = useCallback((docChanges: any[], fromCache: boolean = false) => {
    if (!pendingDelegationsMap.current) {
      const currentDelegations = useAppStore.getState().delegationGroups || [];
      const map = new Map<string, DelegationGroup>();
      currentDelegations.forEach((g) => map.set(g.id, g));
      pendingDelegationsMap.current = map;
    }

    docChanges.forEach((change) => {
      const id = change.doc.id;
      const rawDocData = change.doc.data();
      const data = { id, ...rawDocData } as DelegationGroup;
      const isTrueDelete = Boolean(data.isDeleted || change.type === "removed");

      if (isTrueDelete) {
        pendingDelegationsMap.current!.delete(id);
        pendingDelegationsChanged.current.delete(id);
        pendingDelegationsRemoved.current.add(id);
      } else {
        if (fromCache && isStaleCacheDoc(pendingDelegationsMap.current!.get(id), data)) return;
        pendingDelegationsMap.current!.set(id, data);
        pendingDelegationsChanged.current.set(id, data);
        pendingDelegationsRemoved.current.delete(id);
      }
    });

    if (delegationsBatchTimer.current) {
      clearTimeout(delegationsBatchTimer.current);
    }

    delegationsBatchTimer.current = setTimeout(() => {
      if (pendingDelegationsMap.current) {
        const updatedList = Array.from(pendingDelegationsMap.current.values());
        useAppStore.getState().setDelegationGroups(updatedList);
        
        const changed = Array.from(pendingDelegationsChanged.current.values());
        const removed = Array.from(pendingDelegationsRemoved.current);
        pendingDelegationsChanged.current.clear();
        pendingDelegationsRemoved.current.clear();
        
        if (changed.length > 0) offloadPersistSnapshot('taka_delegations', changed);
        removed.forEach((id) => { deleteLocalRecord('taka_delegations', id).catch(() => {}); });
        
        pendingDelegationsMap.current = null;
      }
    }, 100);
  }, []);

  // Sleep/Wake Recovery Listener
  useEffect(() => {
    const handleSleepWakeRecovery = () => {
      // KHÔNG hủy listener và KHÔNG xóa cache năm khi thức giấc (xem giải thích ở B1).
      if (useAppStore.getState().isDelegationsRequested) {
        hydrateDelegationsFromLocal(useAppStore.getState().globalYearFilter).catch(() => {});
      }
    };

    window.addEventListener("taka_sleep_wake_recovery", handleSleepWakeRecovery);
    return () => {
      window.removeEventListener("taka_sleep_wake_recovery", handleSleepWakeRecovery);
    };
  }, []);

  // Cleanup on unmount
  useEffect(() => {
    return () => {
      delegationsPool.current.forEach((u, key) => {
        u();
        usePerformanceStore.getState().unregisterListener(`taka_delegation_groups_${key}`);
      });
      delegationsPool.current.clear();
      delegationsTTLTimeouts.current.forEach((t) => clearTimeout(t));
      delegationsTTLTimeouts.current.clear();
      yearDelegationsCache.current.clear();
      if (delegationsBatchTimer.current) clearTimeout(delegationsBatchTimer.current);
    };
  }, []);

  useEffect(() => {
    if (!querySignature) return;

    const isAllYears = globalYearFilter === "all" || globalYearFilter === "ALL";
    const targetYears = isDelegationsRequested
      ? isAllYears
        ? PRIMARY_ACTIVE_YEARS
        : [globalYearFilter]
      : [];
    const currentSig = `${querySignature}_${targetYears.join(",")}_${isDelegationsRequested}`;

    if (prevQuerySigDelegations.current !== querySignature) {
      if (prevQuerySigDelegations.current !== "") {
        delegationsPool.current.forEach((u, key) => {
          u();
          usePerformanceStore.getState().unregisterListener(`taka_delegation_groups_${key}`);
        });
        delegationsPool.current.clear();
        delegationsTTLTimeouts.current.forEach((t) => clearTimeout(t));
        delegationsTTLTimeouts.current.clear();
        yearDelegationsCache.current.clear();
      }
      prevQuerySigDelegations.current = querySignature;
    }
    prevSigDelegations.current = currentSig;

    const activeYearsSet = new Set(targetYears);
    Array.from(delegationsPool.current.keys()).forEach((year) => {
      if (!activeYearsSet.has(year)) {
        if (!delegationsTTLTimeouts.current.has(year)) {
          const timeoutId = setTimeout(() => {
            const unsub = delegationsPool.current.get(year);
            if (unsub) unsub();
            delegationsPool.current.delete(year);
            delegationsTTLTimeouts.current.delete(year);
            yearDelegationsCache.current.delete(year);
            usePerformanceStore.getState().unregisterListener(`taka_delegation_groups_${year}`);
            if (
              delegationsPool.current.size === 0 &&
              !useAppStore.getState().isDelegationsRequested
            ) {
              useAppStore.getState().setDelegationGroups([]);
            }
          }, TTL_DURATION_MS);
          delegationsTTLTimeouts.current.set(year, timeoutId);
        }
      } else {
        if (delegationsTTLTimeouts.current.has(year)) {
          clearTimeout(delegationsTTLTimeouts.current.get(year)!);
          delegationsTTLTimeouts.current.delete(year);
        }
      }
    });

    if (targetYears.length === 0) {
      return;
    }

    setIsDelegationsLoading(true);

    targetYears.forEach(async (year) => {
      if (!delegationsPool.current.has(year)) {
        const localCount = await hydrateDelegationsFromLocal(year);
        const isClosedOrPastYear = Number(year) < new Date().getFullYear();
        const currentDelegations = useAppStore.getState().delegationGroups || [];
        const hasLocalDelegations =
          localCount > 0 ||
          currentDelegations.some((d) => String(d.projectYear || "") === String(year));

        const deltaManaged =
          (hasLocalDelegations && isClosedOrPastYear) ||
          (DELTA_FIRST.delegations && hasLocalDelegations && hasMarker(await getLoadedMarkers(), 'delegations_year', year));
        if (deltaManaged) {
          delegationsPool.current.set(year, () => {});
          setIsDelegationsLoading(false);
          pullAndApplyDeltas(year, true)
            .then(() => setCloudStatus("online"))
            .catch(() => {});
          usePerformanceStore
            .getState()
            .trackGetDoc(
              `taka_delegation_groups_${year}`,
              true,
              localCount,
              "Dexie Local-First Delegations (0 Read full collection)"
            );
          return;
        }

        delegationsPool.current.set(year, () => {});
        setIsDelegationsLoading(false);

        if (!yearDelegationsCache.current.has(year)) {
          yearDelegationsCache.current.set(year, new Map());
        }
        const yearMap = yearDelegationsCache.current.get(year)!;
        if (localCount > 0) {
          const localItems = await getLocalDelegations(year);
          localItems.forEach((d) => {
            if (d.id) yearMap.set(d.id, d);
          });
          pullAndApplyDeltas(year)
            .then(() => setCloudStatus("online"))
            .catch(() => {});
        }

        const colRef = collection(
          db,
          "artifacts",
          APP_ID,
          "public",
          "data",
          "taka_delegation_groups"
        );
        const qDelegations = buildStoreAndYearQuery(
          colRef,
          "projectStore",
          "projectYear",
          year,
          userRole,
          allowedStores
        );

        usePerformanceStore.getState().registerListener(`taka_delegation_groups_${year}`);
        const unsub = onSnapshot(
          qDelegations,
          (snapshot) => {
            usePerformanceStore
              .getState()
              .trackSnapshot(`taka_delegation_groups_${year}`, snapshot);
            setIsDelegationsLoading(false);
            batchDelegationsUpdate(snapshot.docChanges(), snapshot.metadata.fromCache);

            if (DELTA_FIRST.delegations && !snapshot.metadata.fromCache) {
              markLoaded('delegations_year', year).catch(() => {});
              const u = delegationsPool.current.get(year);
              delegationsPool.current.set(year, () => {});
              u?.();
              usePerformanceStore.getState().unregisterListener(`taka_delegation_groups_${year}`);
            }
          },
          (e) => {
            delegationsPool.current.delete(year);
            setIsDelegationsLoading(false);
            console.error("Delegation groups listener error:", e);
          }
        );
        delegationsPool.current.set(year, unsub);
      }
    });
  }, [
    querySignature,
    userRole,
    allowedStores,
    isDelegationsRequested,
    globalYearFilter,
    setIsDelegationsLoading,
    setCloudStatus,
  ]);
}
